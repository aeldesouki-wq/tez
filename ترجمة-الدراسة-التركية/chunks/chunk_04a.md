# [الفصل الأول: دراسة حياة المؤلف برهان الدين النسفي وآثاره]{dir="rtl"}

## [1.1.‎ التعريف بالمؤلف]{dir="rtl"}

### [1.1.1.‎ اسمه ونسبه وكنيته وألقابه]{dir="rtl"}

[هو محمد بن محمد]{dir="rtl"}[^63] [بن محمد النسفي، كنيته أبو
الفضل]{dir="rtl"}[^64][، وقيل: أبو الفضائل]{dir="rtl"}[^65][، وقيل: أبو
عبد الله]{dir="rtl"}[^66][، ولقبه الذي غلب عليه حتى صار عَلَما: برهان
الدين، ونسبته إلى \"نسف\" إحدى مدن ما وراء النهر.]{dir="rtl"}

### [1.1.2.‎ مولده ونشأته]{dir="rtl"}

[ولد النسفي سنة 600هـ تقريبا على ما تفيده المصادر]{dir="rtl"}[^67][، ولم
تُعيّن هذه المصادر بلدة مولده، ولا مكان تلقيه تعليمه الأولي، ولا شيئا عن
الأسرة التي نشأ فيها.]{dir="rtl"}

[وأقصى ما يمكن أن يقال في ذلك: إن نسبته إلى \"نسف\" تجعل ولادته بها أو
نشأته فيها أمرا محتملا، غير أن النسبة وحدها لا تنهض دليلا كافيا على
تعيين المولد؛ إذ قد تلحق الرجل من جهة آبائه، وقد تلحقه من جهة الإقامة
الطارئة أو الدراسة، وقد يُنسب إلى البلد من لم يُولد بها. فالأولى الوقوف
عند حد الاحتمال، وترك الجزم في هذه المسألة حتى يظهر نص صريح يقطع
فيها.]{dir="rtl"}

## [1.2.‎ مسيرته العلمية]{dir="rtl"}

### [1.2.1.‎ طلبه للعلم ومراحل تعليمه]{dir="rtl"}

[لم يتبين لي من المصادر المتاحة شيء عن مراحل تعلمه الأولى، ولم أقف ممن
تلقى عنهم إلا على واحد سيأتي ذكره في شيوخه. غير أن مما يمكن تأكيده أنه
تلقى العلم على أيدي كبار عصره، وأنه أخذ عن جماعة من العلماء المتفننين في
بدايات طلبه؛ وذلك يبدو جليا بالنظر إلى مكانة مؤلفاته وتنوعها في فنون
متعددة، من التفسير إلى الجدل والفلسفة، أي إنه برع في المعقول والمنقول
معا، وهذا لا يكون إلا بالتتلمذ على علماء أفذاذ في كل هذه
الفنون.]{dir="rtl"}

[ومما تجدر الإشارة إليه أن الأستاذ الدكتور شكري أوزن رجّح أن يكون النسفي
قد تلقى العلم في بداياته على ركن الدين العميدي أو على أحد
تلامذته]{dir="rtl"}[^68][، بناء على سيره على منهج العميدي في علم الجدل
وتأثره به، وهو ما يبدو واضحا في المؤلَّف قيد التحقيق.]{dir="rtl"}

### [1.2.2.‎ شيوخه]{dir="rtl"}

[ذكر ابن الفوطي أن البرهان روى لهم عن مجد الدين أبي المحامد محمد بن
مسعود بن الحسن الخوارزمي]{dir="rtl"}[^69][، فهو بذلك شيخه، وهو الشيخ
الوحيد الذي وقفت على ذكره للنسفي في المصادر، وقد وصفه ابن الفوطي بأنه
كان من الأفراد المشهورين والأئمة المعدودين.]{dir="rtl"}

[ولا يعني هذا أن النسفي لم يقرأ إلا على الخوارزمي، بدليل كونه إماما في
فنون كثيرة لا تخفى صعوبتها، كالمنطق والفلسفة والخلاف العالي، وكلها مما
يستلزم شيخا حاذقا لفتح مغاليقها وحل معضلها.]{dir="rtl"}

### [1.2.3.‎ تلاميذه]{dir="rtl"}

[ذكر الذهبي أنه \"تخرج به خلق\"]{dir="rtl"}[^70][، وهذه دلالة على كثرة
من قرأ العلم على يديه، خاصة أنه تجول في حواضر إسلامية كثيرة، ودرّس في دار
الملك بدهلي، وفي بغداد سنين طوالا. ومع هذا لم تذكر المصادر إلا قليلا من
تلاميذه، وممن وقفت عليهم:]{dir="rtl"}

1.  [شرف الدين هارون بن شمس الدين الجويني، ابن الصاحب الأعظم (ت 685هـ)،
    الذي عمل مدرسا في المدرسة النظامية سنة 671هـ]{dir="rtl"}[^71][، وقد
    درس على الإمام بعد رجوعه من الحج سنة خمس وسبعين وستمائة تقريبا،
    وأهداه النسفي كتابه \"شرح النكت الأربعين\". وهذا يُرجّح كون الشرح من
    آخر كتبه، ويوضح سبب عدم بروز حواش عليه، بخلاف شروح المتن المتعددة،
    والله أعلم.]{dir="rtl"}

2.  [الحافظ المحدث علم الدين البرزالي (ت 739هـ)، وقد أجازه النسفي ببغداد
    سنة أربع وثمانين وستمائة]{dir="rtl"}[^72][.]{dir="rtl"}

3.  [ابن الفوطي (ت 723هـ)، وقد صرّح بالرواية عن النسفي والسماع
    منه]{dir="rtl"}[^73][.]{dir="rtl"}

4.  [قاضي القضاة جلال الدين أحمد بن الحسن ابن أنوشروان (ت 745هـ)، وقد
    قرأ على برهان الدين الخلاف بدمشق]{dir="rtl"}[^74][.]{dir="rtl"}

5.  [الحسن بن يوسف بن المطهر الحلي، المعروف بالعلامة الحلي (ت
    726هـ)]{dir="rtl"}[^75][، وقد أهدى كتابه \"الأسرار الخفية في العلوم
    العقلية\" إلى هارون بن الصاحب الأعظم تلميذ الإمام البرهان
    النسفي]{dir="rtl"}[^76][، وذُكر في مقدمة كتابه \"نهج الحق\" أنه لم
    يقرأ على النسفي كتبه في الجدل فحسب، بل قرأ عليه الحديث
    أيضا]{dir="rtl"}[^77][.]{dir="rtl"}

[وتجدر الإشارة إلى أن المصادر السنية تخلو من ذكر هذه التلمذة، وليس في
خلوها منها نفي لها بالضرورة، لكنه إذا ضُمّ إلى تصريح الحلي بالإجازة ترجّح
أن يكون الحلي قد أُجيز من النسفي إجازةً دون تلمذة حقيقية مستمرة. وعلى كلا
الاحتمالين فإن صلة عالم شيعي بحجم الحلي بإمام من أئمة السنة الكبار في
هذا الوقت تبرز سعة أفق الإمام النسفي، وتكشف عن الجو العلمي المنفتح الذي
كانت عليه بغداد في ذلك الزمان.]{dir="rtl"}

6.  [شمس الدين السمرقندي (ت 690هـ)، وقد صرّح بنفسه أنه قرأ على الإمام متن
    \"النكت الأربعين\" بتبريز]{dir="rtl"}[^78][، وإن لم يُكتب اسم المؤلف
    في بداية المخطوط ولا في نهايته، فإن إحالته على كتابيه \"الأنوار\"
    و\"القسطاس\"]{dir="rtl"}[^79] [تدل على أنه له، وتؤكد تلمذته على يد
    الإمام.]{dir="rtl"}

7.  [برهان الدين الجعبري (ت 732هـ)]{dir="rtl"}[^80][، فقد نقل الباحث
    عيسى خيري الجعبري عن \"عوالي مشيخته\" أنه صرّح بقراءته على الإمام
    النسفي، وأن من جملة ما قرأ عليه كتاب \"مصابيح السنة\"
    للبغوي]{dir="rtl"}[^81][.]{dir="rtl"}

[ولم أقف على \"عوالي مشيخته\" لأتحقق بنفسي من ذكره الإمام برهان الدين
النسفي في جملة من قرأ عليهم، غير أن قرائن تسند هذا النقل،
منها:]{dir="rtl"}

-   [أن الإمام الجعبري وافق وجودُه في بغداد وجودَ الإمام النسفي بها، وهذا
    يرجّح التقاءهما، خاصة وأن الإمام النسفي كان من أعلام العلماء في تلك
    الفترة ببغداد.]{dir="rtl"}

-   [أن الإمام الجعبري ذكر في ثبت مصنفاته ما يؤكد اهتمامه بعلم الجدل،
    فقد اختصر \"منتهى السول والأمل في علمي الأصول والجدل\" لابن الحاجب
    (ت 646هـ)، وسمّى مختصره: \"منتهى النهول والعلل مختصر مختصر الأصول
    والجدل\"، كما كتب \"المرتجل والمنتظر في الجدل وعلم
    النظر\"]{dir="rtl"}[^82][.]{dir="rtl"}

8.  [محمود بن إبراهيم المدعو بالشمس الخوارزمي، فقد ورد قوله في مقدمة
    شرحه على النكت: \"أما بعد، فقد رأيت النكات الأربعين التي بلغت من بسط
    الأرض الأقطار، حتى وثبن الجبال وخضن البحار، وهي التي أنشأها العلامة
    المحقق الداهية، أستاذي وأستاذ البشر، المبرز في علم النظر\... الهادي
    إلى التفنن، المهدي برهان الدين النسفي، أطال الله بقاءه، ورزقنا مرة
    أخرى لقاءه\"]{dir="rtl"}[^83][.]{dir="rtl"}

[وهذا تصريح منه بلقيا الإمام النسفي والتتلمذ على يديه، وقد وقع الفراغ من
كتابة شرحه سنة ست وسبعمائة كما دُوّن في آخر المخطوط، أي بعد وفاة الإمام
النسفي بتسع عشرة سنة تقريبا، لذا يمكننا الجزم بأنه قرأ عليه؛ لوجود
المعاصرة مع التصريح باللقيا.]{dir="rtl"}

9.  [علاء الدين النسفي، شيخ جلال الدين الكتاني ثم الكشاني، كما ورد في
    شرح سعد الدين بن علي السمرقندي على متن النكت]{dir="rtl"}[^84] [أنه
    قرأ المتن على شيخه جلال الدين الكشاني، وشيخِه حميد الحق والدين، وهما
    قرآ على علاء الدين النسفي الذي قرأ على المصنف برهان الدين النسفي
    رحمه الله]{dir="rtl"}[^85][.]{dir="rtl"}

10. [علي بن هارون بن محمد السمرقندي، فقد جاء في خاتمة نسخة \"المقدمة
    البرهانية\" الموجودة في مكتبة برلين أن ناسخها محمد بن عبد الرحمن
    الطوسي قد أجازه إجازة قراءة وحفظ وفهم وإتقان شيخُه علي بن هارون
    السمرقندي، الذي سمع الكتاب بدوره من مصنفه برهان الدين النسفي؛ فهذا
    تصريح من تلميذه بأنه تلميذ النسفي. وقد نُسخ هذا الكتاب عام 697هـ، أي
    بعد وفاة المؤلف بعشر سنوات تقريبا، وهذا مرجّح لكون السمرقندي تلميذا
    حقيقيا له؛ لقرب العهد]{dir="rtl"} [^86] [، ولم أقف على ترجمة
    له.]{dir="rtl"}

### [1.2.4.‎ رحلاته في طلب العلم ونشره]{dir="rtl"}

[لا نستطيع بحسب ما بين أيدينا من مصادر أن نعرف أين وُلد بالتحديد، ولا أين
تلقى تعليمه الأولي، لكن من خلال مطالعة كتبه يمكننا الجزم بأنه تلقى
تعليما قويا في بداية طلبه؛ وذلك لأنه برع في علوم كثيرة، وكتبه فيها من
الدقة بمكان، كما أن رحلاته المتعددة، وأخذ أئمة كبار كالبرزالي والسمرقندي
والصاحب هارون عنه، وإجازته للعلامة الحلي، كل ذلك يدل على جلالة قدره في
العلوم عامة، وفي علم الخلاف خاصة.]{dir="rtl"}

[وبالجمع بين ما تفرق في المصادر يمكننا أن نرجّح أنه انتقل من نسف إلى
تبريز بعدما أتقن العلوم وذاع صيته، فأقرأ هناك، وممن أقرأهم شمس الدين
السمرقندي، وربما انتقل بعدها إلى الهند للتدريس في دار الملك
بدهلي]{dir="rtl"}[^87][، ولعل هذا كان في زمن ناصر الدين محمود بن شمس
الدين إلتتمش، أو غياث الدين بلبن]{dir="rtl"}[^88] [على الأرجح. وتثبت
المصادر ذهابه إلى رحلة الحج، ولعل زيارته لدمشق كانت في رحلة الذهاب أو
العودة منه، كما أنه قدم بغداد حاجا سنة خمس وسبعين
وستمائة]{dir="rtl"}[^89][، وختم رحلة حياته الحافلة هذه في بغداد التي
تثبت المصادر وفاته بها، أي أنه بقي فيها مدرسا ما يقارب ثنتي عشرة
سنة.]{dir="rtl"}

## [1.3.‎ مكانته ومذهبه]{dir="rtl"}

### [1.3.1.‎ مذهبه الفقهي والعقدي]{dir="rtl"}

[النسفي حنفي المذهب، كما يتضح من تبنيه مذهب الحنفية في المخطوط الذي أعمل
على تحقيقه، ومن كلام من ترجم له. أما مذهبه الكلامي فتأكيد نسبته إلى
الماتريدية أو الأشعرية من الصعوبة بمكان؛ إذ إن بعض أقواله تشير إلى تبنيه
المذهب الماتريدي، ومن ذلك ما ورد عن قبوله إيمان المقلد واعتباره
صحته]{dir="rtl"}[^90][، وهو مما خالف فيه الماتريدية الأشعرية؛ ويمكننا في
المقابل أن نقول: إنه تأثر بالأشعرية من خلال اهتمامه بكتب الإمامين الرازي
والغزالي الأشعريين وتأثره بها. وإن كانت الكتب القديمة التي ترجمت له لم
تعتمد له مذهبا كلاميا واضحا --- على حد بحثي.]{dir="rtl"}

[كذلك فإن نقله عن الجنيد البغدادي ودفاعه عن سنية الحلاج يمكن اعتبارهما
دليلا على كونه متصوفا سنيا، لا يرى في الانتساب إلى التصوف
بأسا]{dir="rtl"}[^91][.]{dir="rtl"}

[ومما يبرز مبلغ قبوله لدى أصحاب المذاهب كلها في هذا الزمن، سنيّها وغير
سنيّها، ويدل على سعة أفقه: أن من الشافعية من قرأ عليه كالجعبري، ومن
الشيعة من أخذ عنه إجازةً كالعلامة الحلي، وقد أجمعوا على الثناء على علمه
وتواضعه وزهده، مما يبرز علو كعبه علميا وتربويا.]{dir="rtl"}

### [1.3.2.‎ مكانته العلمية وأثره في علم الجدل]{dir="rtl"}

[برع الإمام البرهان في علوم متعددة، كالتفسير والفقه والكلام والخلاف
والمنطق والجدل والفلسفة، وكتبه فيها عظيمة الفائدة، ومن الدقة بمكان؛
فنجده اختصر تفسير الإمام الرازي في تفسيره \"الواضح\"، واختصر كتاب الرازي
في الكلام وسماه \"مختصر نهاية العقول\"، وله في شرح الأسماء الحسنى
كتابان: \"مشارق الأنوار\"، و\"المنتخب من كتاب المقصد الأقصى\" وهو اختصار
لكتاب الغزالي]{dir="rtl"}[^92][. وقد شرح كتاب \"الإشارات والتنبيهات\"
لابن سينا، وكتب \"أساس الكياسة\" في الفلسفة وشرحه، وكتب \"المنطق
الكبير\".]{dir="rtl"}

[وكتبه في الخلاف من الكتب المشهورة التي اعتُمد عليها في التأليف بعده،
وصارت من الكتب المعتمدة في التدريس في المدارس فترة طويلة، كما أنه لم
يكتف بالتأصيل النظري للجدل، بل أردفه بالتطبيق العملي للمسائل الجدلية،
كما في كتاب \"شرح النكت الأربعين\" موضع التحقيق.]{dir="rtl"}

[وقد جاء في وصفه أنه \"شيخ الفلسفة ببغداد\"]{dir="rtl"}[^93][، وأنه
\"المتكلم صاحب التصانيف في الخلاف\"]{dir="rtl"}[^94][، ونُقل عن ابن
الفوطي قوله عنه: \"هو شيخنا، المحقق، المدقق، العلامة، الحكيم، له
التصانيف المشهورة، كان في الخلاف والفلسفة، مُتّع بحواسه، وكان
زاهدا\"]{dir="rtl"}[^95][. وفوق هذه الأوصاف فهو: حنفي
منطقي]{dir="rtl"}[^96][، مفسر محدث أصولي]{dir="rtl"}[^97][.]{dir="rtl"}

[وفوق ذلك يمكن بيان مزيد أهميته حسب ما يمكن تلخيصه من كلام ميلر في كتابه
فيما يلي:]{dir="rtl"}

1.  [يُعد النسفي أول من صاغ قوانين محددة للمناظرة تبيّن من له الحق في بدء
    المناظرة وطريقة سيرها، وقبل النسفي كانت القواعد تقتصر على تقاليد
    تتوزع في كتب الخلاف دون تدوين منهجي
    واضح]{dir="rtl"}[^98][.]{dir="rtl"}

2.  [استبعد النسفي كلمة \"الجدل\" من مقدمته مستبدلا بها \"المناظرة\"،
    وهذا دليل رغبته في تحرير المناظرة من مجرد السؤال والجواب، ولذا فقد
    أعاد تعريف أدوار المتناظرَين، فلم يعد المستدل مجرد مجيب، بل سماه
    \"المناظر\"، وصار السائل \"خصما\" لا مجرد مستفسر، بل يملك حق
    الاعتراض]{dir="rtl"}[^99][.]{dir="rtl"}

3.  [وضع صيغا منضبطة مجردة للمناقضة والمعارضة، مما سهّل على المتناظرين
    الاحتكام إلى قوانين منضبطة مشتركة]{dir="rtl"}[^100][.]{dir="rtl"}

4.  [وضع ضبطا زمنيا ومنهجيا بين الغصب والمعارضة، وميّز
    بينهما]{dir="rtl"}[^101][.]{dir="rtl"}

5.  [أحدث اندماجا عميقا بين المنطق والجدل الفقهي، فأدخل مصطلحات المنطق
    الصوري وقضاياه إلى البنية التأسيسية للمناظرة]{dir="rtl"}[^102][، كما
    ألحق فصلا مفصلا لمعالجة أحكام التناقض وشروطه التسعة (وهي الوحدة في:
    الجزء، والكل، والمكان، والزمان، والإضافة، والقوة، والفعل، والكيف،
    والكم) لتأصيل البنية التناقضية للقضايا
    الجدلية]{dir="rtl"}[^103][.]{dir="rtl"}

6.  [صاغ تعريفات منطقية دقيقة للمفاهيم الثلاثة التي تقوم عليها المناظرة،
    وهي: الدليل، والاستدلال، والتعليل]{dir="rtl"}[^104][.]{dir="rtl"}

[وفوق كل ما سبق يمكن اعتبار النسفي مؤسسا حقيقيا لعلم آداب البحث
والمناظرة مع السمرقندي؛ فالسمرقندي لم يقرأ عليه ويأخذ علمه شفاهةً فحسب،
