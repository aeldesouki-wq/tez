# الباب: السؤال والجواب، و«لا أدري» و«سلوني»، وآداب المستفتي
## (حمد السؤال وذمّ الحياء منه، حسن السؤال وآدابه في الدرس، النهي عن كثرة السؤال وعمّا لم يقع، صناعة الجواب، قول «لا أدري»، ابتداء العالم و«سلوني»، آداب المستفتي)

> **المقروء لهذا الباب بنصه** (الصفحة المطبوعة بحسب وسم الشاملة، وحيث وقع في سجل واحد وسمٌ لصفحة تالية (⦗ص⦘) أخذتُ صفحة الوسم لما بعده؛ وبوسم الجزء حيث تتكرر الأرقام):
> - جامع بيان العلم (ع‑ب): ج١ ص٣٧٣–٣٨٧ (باب حمد السؤال والإلحاح في طلب العلم)، ص٤٥٥–٤٧٥ (هيبة المتعلم للعالم؛ ابتداء العالم جلساءه بالفائدة وقوله سلوني)، ص٤٨٥–٤٨٧ (فتوى الصغير بين يدي الكبير)، ص٥١٥–٥٢٢ (علموا ولا تعنتوا؛ ترك المماراة؛ حقوق العالم)، ص٥٧٨–٥٨٤ (من باب جامع آداب العالم والمتعلم: أرجوزة اللؤلؤي ووصية يحيى بن خالد)؛ ج٢ ص٨٢٦–٨٤٣ (ما يلزم العالم إذا سئل عما لا يدريه) كاملاً؛ وج٢ ص١٠١٥–١٠٢١ (روابط ١٢٧٣–١٢٨١: لخبر مطر «زاملة»)، وج٢ ص١٠٥٣–١٠٧٥ (روابط ١٣٥٥–١٣٩٩: من أثناء باب ذم الرأي، فيه نصوص السؤال عما لم يكن والأغلوطات). **ومما أضفتُه من القراءة الجديدة (بعد مقابلته بالنص الخام):** ج١ ص٤٧٩–٤٨٤ (باب طرح العالم المسألة على المتعلم، ويُعالَج في الباب الثامن م١٩ فلا أنقل منه هنا)، وج١ ص٧٥٣ وص٧٥٧–٧٥٩ وص٧٧٣–٧٧٨ (من باب معرفة أصول العلم: «العلم ثلاثة»، وجواب سحنون في «لا أدري»، و«أرأي أم علم؟»)، وج٢ ص٨١٥–٨٢١ (معرفة الاختلاف شرط الفتوى)، وص٨٤٨–٨٤٩، وص٨٨٠، وج٢ ص١٠٣٧ وص١٠٤٢ وص١٠٥٢–١٠٥٤ (من أول باب ذم الرأي وكثرة المسائل)، وص١٠٥٩–١٠٦٠، وص١٠٧٦–١٠٧٧ (أرأيت)، وج٢ ص١١٢٠–١١٢٨ (باب تدافع الفتوى وذم من سارع إليها) كاملاً.
> - الجامع لأخلاق الراوي (خ‑ج): ج١ ص١٩٦–١٩٧ (إعادة الاستفهام)، ص٢٠٢–٢٢٧ (أدب السؤال للمحدث إلى كتابة الأطراف)؛ وج٢ ص١١١–١١٢ (تفسير الراوي ما يرويه). **الأسانيد لم تُدقَّق** إلا ما يعيّن الراوي. وأضفتُ من القراءة الجديدة: ج١ ص٢٢٣–٢٢٧ (ما ينبغي أن يُسأل عنه الراوي)، وج١ ص٣١٧، وج٢ ص٩٦.
> - الفقيه والمتفقه (خ‑ف): ج٢ ص١١–٣٥ (السؤال عن الحادثة قبل وقوعها)، ص٦١–٧٢ (السؤال والجواب)، ص٧٧–٨٢ (أول تقسيم الأسئلة)، ص٢٠١–٢٠٤، ص٢٧٤–٢٨٣ (إلقاء المسائل؛ الحدث في حلقة النظر)، ص٢٩١–٣٠٤ وص٣١١–٣١٧ (من «تنبيه الفقيه على مراتب أصحابه» ما يخص السؤال)، ص٣٤٩–٣٧٢ (الزجر عن التسرع إلى الفتوى؛ الإحجام عن الجواب)، ص٣٧٥–٤٠٥ (أدب المستفتي وما يفعله المفتي إلى نصف ص٤٠٥)، ص٤٢١–٤٢٦ (رجوع المفتي؛ ولا أنقل منه)؛ وج١ ص٤٤٥ (سطر واحد). وما بعد ص٤٠٥ إلى ص٤٠٧ مسحٌ بالكلمات المفتاحية لا قراءة متصلة. وأضفتُ من القراءة الجديدة: ج١ ص٤٥٥–٤٦٦ (من باب ذم القياس: مسروق والشعبي والرأيتيون)، وج٢ ص٣٢٥ وص٣٥٠ وص٣٥٣، وص٤١٦–٤٢٠؛ وما يخص أضرب السؤال والجواب من ج٢ ص٧٧–١١١ (وقد بُسط في الباب العاشر م٢٢–٢٥).
> - تذكرة السامع (ج): ص٣٤، ٤١–٤٥، ٥٣–٥٥، ١٠١–١٠٨، ١٥٦–١٥٩.
> - المعيد (ع): ص١٠٥، ١٢٠–١٢١، ١٢٨–١٣١، ١٥٠–١٥٣، ١٧٤–١٧٧ (آداب المفتي: ما نقله عن ب‑ص من أخبار لا أدري)، ٢٠٢–٢١٦ (بقية آداب المفتي وآداب المستفتي)، ص٢٧١.
> - حلية طالب العلم (ح): ص١٦٢–١٦٤، ١٨٣–١٨٤، ١٨٨–١٨٩.
> - آداب العلماء والمتعلمين للزيدي (ز‑ي، ترقيم آلي لا يطابق المطبوع): ص٨ وص٢١.
> - أدب الدنيا والدين (ماو): ص٥٩، ٦٩–٧٠، ٧٣–٧٦. الذريعة (ر): ص١٢٣ وص١٥٥ (من القراءة الجديدة)، وص١٧٧، ١٨٢. إحياء الغزالي (غ): ج١ ص٩، ٢٧، ٥١، ٥٨، ٦٨–٧١؛ وج٢ ص١٠٣، ١١٥، ١٢٢ (من مبحث مثارات الشبهة، لمواضع المستفتي ولا أدري). أدب الإملاء والاستملاء (سم): ص٦٢–٦٣، وسطر في ص١٣٣. أدب المفتي والمستفتي لابن الصلاح (ب‑ص): ص٧٤–٨٠، ١٠٩، ١٣١–١٥٧ (كيفية الفتوى وآدابها)، ١٥٨–١٧١ (المستفتي). الإعلام لمحمد المقدم (مق، معاصر): ص٢٢١–٢٣٢ وص٢٤٢–٢٦٠، وأضفتُ من القراءة الجديدة متن ص٢٧٢–٢٧٥ وهوامش ص٢٣٠ و٢٤٢ و٢٤٤ و٢٥١ و٢٥٦–٢٥٧ و٢٦٠ و٢٧٤ (وهي في الأغلب نقول المقدم عن النووي والشاطبي والخطابي وغيرهم، ولم أراجع أصولها).
> - ما أُخذ من تلخيص نصوص متعرَّف عليها ضوئياً (يُوسَم بعبارة (OCR، بلا صفحة) ولا يُنقل منه نصٌّ بين علامتي التنصيص): الكافي للكليني (باب حق العالم)، والذريعة للمرتضى (فصل صفة المفتي والمستفتي)، وجامع السعادات للنراقي (آداب التعليم)، ومنية المريد للشهيد الثاني (ش؛ وقد قوبل على نص OpenITI النظيف المشكول، وهو بلا ترقيم صفحات).
>
> **لم يُقرأ لهذا الباب بنصه:** من جامع بيان العلم ما سوى المواضع المذكورة أعلاه؛ وقد قُرئ الكتاب كله في ملفات القراءة (الروابط ١–١٥٨٥)، ولم آخذ لهذا الباب من بقيته إلا ما أثبتّه، بعد بحث آلي في نصه عن «لا أدري» و«لا أعلم». الجامع للخطيب خارج المواضع المذكورة. الفقيه والمتفقه خارج المواضع المذكورة (وما قُرئ منه في ملف الجدل ج٢ ص٨–١١٣ وفي آداب الفقيه ج٢ ص٢٣٧–٢٨٩ لم آخذ منه إلا ما يخص السؤال). ابن جماعة والعلموي وأبو زيد خارج المواضع المذكورة. الشوكاني (ص١٣٢، ٢٠٨، ٢٢٢–٢٢٤) واليعقوبي (ص٢٣٨): ملخصات لا نصوص، وتتمة قراءتهما لا مادة فيها لهذا الباب. الراغب: قُرئ في التتمة كله، وأخذتُ منه ص١٢٣ وص١٥٥ مع ص١٧٧ وص١٨٢. الماوردي ومسكويه: قُرئا كاملين في التتمة، وما يخص الباب عند الماوردي في ص٥٩ وص٦٩–٧٦ هنا وفي ص٢٧٦–٢٧٧ (الباب الثامن م٢١) ولا مادة عند مسكويه. ما في المقدم منسوباً إلى ابن حجر والشاطبي والقاسمي وابن القيم والنووي والخطابي **هو نقل المقدم عنهم ولم أراجع أصولهم**؛ وما في ع منسوباً إلى الصيمري وأبي إسحاق والنووي، وما في ب‑ص منسوباً إلى أبي المظفر السمعاني والقفال وابن سريج والحليمي والصيمري: نقلٌ لم أراجع أصوله. الزرنوجي والمنسوب للطوسي: لم أجد فيما قرأته من تقاريرهما شيئاً مستقلاً في هذا الباب. منية المريد للشهيد الثاني: اطلعتُ على نص OpenITI النظيف لبنيته وعناوين أموره، فأثبتُّ ما يخص السؤال (م٥ وم١٠ وم٢٣ وم٥١) دون نصٍّ بين علامتي التنصيص. الكافي: لم أقرأ منه إلا عناوين أبواب كتاب فضل العلم وباب حق العالم، ومنها باب سؤال العالم وتذاكره وباب النهي عن القول بغير علم: لم يُقرأ منهما إلا العنوان.
>
> **تنبيهات منهجية:**
> - (أ) **الأقدمية بالوفاة:** الكليني (٣٢٩؛ الكافي، OCR) ثم المرتضى (٤٣٦؛ الذريعة، OCR) ثم الماوردي (٤٥٠) ثم ابن عبد البر والخطيب (كلاهما ٤٦٣ فلا أحكم بأقدمية لأحدهما) ثم الراغب (٥٠٢) فالغزالي (٥٠٥) فالسمعاني (٥٦٢) فابن الصلاح (٦٤٣) فابن جماعة (٧٣٣) فالشهيد الثاني (٩٦٥) فالعلموي (٩٨١) فالزيدي (١٠٥٠) فالنراقي (١٢٠٩) فأبو زيد (ت ١٤٢٩هـ) ثم المقدم (معاصر). وأقدمية الكتاب لا تعني أنه أصل لما بعده: فبعض ما عند ب‑ص وع يطابق الخطيب لفظاً أو معنى، وأنبه على ذلك حيث ظهر لي، ولا أجزم بالأخذ إلا حيث صرّح المصنِّف.
> - (ب) **رموز مضافة إلى جدول المقدمة:** **ع‑ب** ابن عبد البر، **خ‑ج** الجامع للخطيب، **سم** السمعاني، **ب‑ص** ابن الصلاح، **مق** المقدم؛ وحيث تتكرر الأرقام بين الجزأين ذكرتُ الجزء.
> - (ج) **العلموي وابن جماعة والزيدي:** كثير مما في ع وز‑ي في هذا الباب نقلٌ من ج أو من ب‑ص؛ وأبيّن ما غيّراه أو زاداه. وقد صرّح ع بالنقل عن ابن الصلاح (ص٢٠٩) وعن الخطيب (ص٢١٤) والسمعاني (ص٢١٤) والنووي (ص١٣٠).
> - (د) **أحكام الأحاديث** ليست من عملي؛ ومن ضعّف حديثاً فحكمه باسمه. وما يرد منسوباً إلى النبي في هذه الكتب يُنقل كما أوردوه.
> - (هـ) **الاقتباس:** كل «» قوبل آلياً بنص المصدر بعد التطبيع وراجعتُ صفحاته آلياً، وعلى عيّنة يدوياً. **«أبو المظفر السمعاني»** المنقول عنه في ب‑ص وع لم أستطع من المقروء ربطه بصاحب «أدب الإملاء والاستملاء» (سم)، فلا أخلط بينهما.
> - (و) **تداخل مع أبواب سابقة:** عولجت في الباب الثالث (٠٥) مسائل: ألا يستفهم قبل تمام الكلام (١٤) وترك الحياء في السؤال (١٥)، وفي باب آداب العالم في درسه (٠٧) مسائل: الإنصاف وسماع السؤال (٢٦) و«لا أدري» (٢٨) واختبار الفهم و«هل فهمت» (٣٣). أعيدها هنا بإيجاز مع الإحالة حيث لا زيادة في مصدر جديد، وأبسطها حيث جاءت مصادر لم تُقرأ هناك.
> - (ز) **تداخل مع الأبواب اللاحقة:** ما تكرر بنصه في باب لاحق تركتُ مسألته كاملةً في الأنسب وأبقيتُ هنا إحالةً صريحة مع ما انفرد به هذا الباب باسم صاحبه: أضرب السؤال الأربعة وحسن الاستماع (الباب العاشر م٢٢ وم١٩)، وتفسير الغريب (الباب الحادي عشر م١٧)، وآداب المستفتي وما يفعله المفتي بالرقعة وما ليس من فنّه (الباب الثالث عشر م٧٥–٨٨ وم٥٢ وم٧٣–٧٤). وقد أحالت الأبواب ٠٤ و٠٥ و٠٨ و١٠ إلى مسائل بأرقامها من هذا الباب (م١–١٢ وم١٤–١٥ وم١٨ وم٢٠ وم٣٨)، فلم أغيّر الترقيم؛ وأضفتُ المسائل الجديدة بأرقام ٤٩ وما بعدها في مواضعها.

## القسم الأول: أصل السؤال وحمده وآدابه عند الطالب

## ١ — السؤال شفاء العي ومفتاح العلم
**المسألة:** السؤال طريق العلم وعلاج الجهل؛ فالعلم خزائن مفتاحها السؤال، ومن سكت عن جهله زاد عماه.
**المصادر:** ماو ص٧٠ [أقدم مقروء]؛ ع‑ب ص٣٧٣–٣٨٠ وخ‑ف ص٦١–٦٢ [٤٦٣]؛ ر ص١٢٣ [٥٠٢]؛ غ ص٩؛ ج ص١٥٧؛ ع ص١٥٢.
**الزيادات:**
- **الماوردي:** يورد عن النبي «هلا سألوا إذا لم يعلموا فإنما شفاء العي السؤال» (ماو ص٧٠)، وحديث «العلم خزائن ومفتاحه السؤال» ثم «فإنما يؤجر في العلم ثلاثة: القائل والمستمع والآخذ» (ماو ص٧٠)، وبيت بشار بن برد «شفاء العمى طول السؤال وإنما ... دوام العمى طول السكوت على الجهل» (ماو ص٥٩).
- **ابن عبد البر:** مطلع الباب «شفاء العي السؤال» (ع‑ب ص٣٧٣)، وحديث صاحب الجرح المحتلم الذي أُمر بالاغتسال فمات: «قتلوه قتلهم الله ألم يكن شفاء العي السؤال» (ع‑ب ص٣٧٥)؛ وابن مسعود «زيادة العلم الابتغاء ودرك العلم السؤال» (ع‑ب ص٣٧٤)؛ وابن شهاب في ثلاث روايات: «العلم خزانة مفتاحها المسألة» (ع‑ب ص٣٧٤) و«العلم خزائن ومفاتيحها السؤال» (ع‑ب ص٣٧٩) و«إن العلم خزائن وتفتحها المسألة» (ع‑ب ص٣٨٠)؛ والخليل «العلوم أقفال والسؤالات مفاتيحها» (ع‑ب ص٣٨٠)؛ وينشد الأصمعي «شفاء العمى طول السؤال وإنما ... تمام العمى طول السكوت على الجهل» (ع‑ب ص٣٨٠).
- **الخطيب (الفقيه):** يفتتح باب السؤال والجواب بحديث مسنَد إلى علي: «فإنه يؤجر فيه أربعة: السائل والمعلم والمستمع والمحب لهم» (خ‑ف ص٦١)، ثم ابن شهاب «إنما هذا العلم خزائن وتفتحها المسألة» (خ‑ف ص٦٢)، وأبو يزيد النهشلي «العلم قفل ومفتاحه المسألة» (خ‑ف ص٦٢).
- **الراغب:** يورد عن النبي «سائل العلماء، وخالط الحكماء، وجالس الكبراء» (ر ص١٢٣)، وينقل شرحاً عن بعض العلماء «مساءلة العلماء ترغبك من الله في ثوابه، وتخوفك من عقابه» (ر ص١٢٣)، ويرى بعضهم أن الأجدر أن تكون هذه الثلاث بإزاء منازل الظالم والمقتصد والسابق (ر ص١٢٣).
- **الغزالي:** «العلم خزائن مفاتيحها السؤال ألا فاسألوا فإنه يؤجر فيه أربعة» (غ ص٩)، ويضيف «لا ينبغي للجاهل أن يسكت على جهله ولا للعالم أن يسكت على علمه» (غ ص٩).
- **ابن جماعة:** يورد «ولبعض العرب: وليس العمى طول السؤال وإنما ... تمام العمى طول السكوت على الجهل» (ج ص١٥٧)؛ **العلموي:** «وقال ابن شهاب: العلم خزائن ومفتاحه المسألة» (ع ص١٥٢).
> **تحرير:** (أ) عدد من يُؤجر: ثلاثة عند ماو (القائل والمستمع والآخذ)، وأربعة عند خ‑ف وغ (السائل والمعلم أو العالم، والمستمع، والمحب لهم)؛ وهو عند خ‑ف مسنَد إلى علي ثم عن النبي، وعند غ مرفوع بلا إسناد. (ب) ألفاظ ابن شهاب: خزانة/خزائن، مفتاحها/مفاتيحها/ومفتاحه/تفتحها، وكلها عن ابن شهاب نفسه في ع‑ب وخ‑ف وع. (ج) صدر البيت: «شفاء العمى طول السؤال» عند ماو (لبشار) وعند ع‑ب (ينشده الأصمعي، ص٣٨٠)، «وليس العمى طول السؤال» عند ج (لبعض العرب، ص١٥٧)؛ وعجزه «دوام العمى» عند ماو (ص٥٩) و«تمام العمى» عند ع‑ب (ص٣٨٠) وج (ص١٥٧)؛ ولا أرجّح.

## ٢ — ذمّ الحياء في السؤال
**المسألة:** لا يمنع الحياء الطالبَ من السؤال؛ فمن استتر عن طلب العلم بالحياء حُرم، والحياء المحمود ما لم يحل بين المرء والطلب. (المسألة موجزة في ٠٥ م١٥؛ وتبسط هنا بنصوص ع‑ب وخ‑ف.)
**المصادر:** ع‑ب ص٣٧٣–٣٨٦ [٤٦٣؛ أقدم مقروء في هذه المسألة، وفي الباب ماو (٤٥٠) أقدم منه في م١ وم٣]؛ خ‑ف (خ‑ف ص٣٠٠)؛ ج ص١٥٧–١٥٨؛ ز‑ي ص٢١؛ ع ص١٥٢.
**الزيادات:**
- **ابن عبد البر:** عائشة في نساء الأنصار «لم يمنعهن الحياء أن يسألن عن أمر دينهن» (ع‑ب ص٣٧٣)؛ «واستحيا علي رضي الله عنه أن يسأل عن المذي لمكان رسول الله» فأمر المقداد وعماراً (ع‑ب ص٣٧٤)؛ وعبد العزيز بن عمر بن عبد العزيز: «كنت أستحي أن أسأل عنها فكبرت وفي جهالتها» (ع‑ب ص٣٨٢)؛ وبطاقة ابن المبارك لرجل استحيا أن يسأل: «فأعنت الشيخ بالسؤال تجده سلسا يلقاك بالراحتين» (ع‑ب ص٣٨١). وفي وصية علي: «ولا يستحي جاهل أن يسأل ولا يستحي عالم إن لم يعلم أن يقول: الله أعلم» (ع‑ب ص٣٨٣)، وفي رواية الشعبي «ولا يستحي إذا كان لا يعلم أن يتعلم» (ع‑ب ص٣٨٣)؛ وعلي «قرنت الهيبة بالخيبة، والحياء بالحرمان» (ع‑ب ص٣٨٣)؛ والحسن «من استتر عن طلب العلم بالحياء لبس للجهل سربالَه» (ع‑ب ص٣٨٣)؛ والخليل «الجهل منزلة بين الحياء والأنفة» (ع‑ب ص٣٨٤). ويقيّد ابن عبد البر في أبياته: «إن الحياء لخير كله أبدا ... ما لم يحل بين نفس المرء والطلب» (ع‑ب ص٣٨٦).
- **ابن جماعة:** «قال عمر رضي الله عنه: من رق وجهه رق علمه» (ج ص١٥٧)، ومجاهد «لا يتعلم العلم مستحيي ولا مستكبر» (ج ص١٥٧)، وأم سليم «إن الله لا يستحيي من الحق» (ج ص١٥٧)؛ ثم زيادته المبنية على سؤال الشيخ: «وكما لا ينبغي للطالب أن يستحيي من السؤال فكذلك لا يستحيي من قوله لم أفهم إذا سأله الشيخ» (ج ص١٥٧)، ويقيّد جوابه: «فإن سأله فلا يقول نعم، حتى يتضح له المعنى اتضاحا جليا كيلا يفوته الفهم ويدركه بكذبه الإثم» (ج ص١٥٨).
- **العلموي:** «ومن رق وجهه عند السؤال، ظهر نقصه عند اجتماع الرجال» (ع ص١٥٢)، ويكرر «ولا يستحي من قوله لم أفهم» (ع ص١٥٢) وقول الخليل (ع ص١٥٢). **الزيدي:** يقتصر على «وقد قيل: من رق وجهه عند السؤال، ظهر نقصه عند اجتماع الرجال» (ز‑ي ص٢١) ويسقط نسبة عمر ومجاهد وأم سليم.
> **تحرير النسبة:** «من رق وجهه رق علمه»: للحسن عند ع‑ب (ص٣٨٣)، وبصيغة «وكان يقال: من رق وجهه عند السؤال رق علمه عند الرجال» عند ع‑ب أيضاً (ص٣٨٤)، ولابن عمر عند خ‑ف («من رق وجهه رق علمه»، خ‑ف ص٣٠٠)، ولعمر عند ج (ص١٥٧)، وبلا عزو عند ع (ص١٥٢). ووصية علي «لا يستحي جاهل أن يسأل» تقع عند ع‑ب برواية عكرمة (ص٣٨٣) وبرواية الشعبي (ص٣٨٣) بلفظ مختلف في الشطر الثاني («أن يقول: الله أعلم» / «أن يتعلم»). ولا أرجّح.
> **ملحوظة:** تقييد ج لترك الحياء بقوله «ولا يسأل عن شيء في غير موضعه» (ص١٥٧) في م٦.

## ٣ — حسن السؤال نصف العلم
**المسألة:** حسن السؤال شطر العلم؛ وبه نال السلف ما نالوا.
**المصادر:** ماو ص٧٠ [أقدم مقروء]؛ ع‑ب ص٣٧٨، ٣٨٢؛ خ‑ج ج١ ص٢١٣ وخ‑ف ص٦٣–٦٤ [٤٦٣]؛ ع ص١٥٢؛ ح ص١٨٨.
**الزيادات:**
- **الماوردي:** سُئل ابن عباس بم نلت العلم فقال «بلسان سئول وقلب عقول» (ماو ص٧٠)، ويروي عن ابن عمر مرفوعاً «حسن السؤال نصف العلم» (ماو ص٧٠).
- **ابن عبد البر:** دغفل النسابة «حفظت هذا بقلب عقول ولسان سئول» (ع‑ب ص٣٧٨)؛ ووهب بن منبه وسليمان بن يسار «حسن المسألة نصف العلم، والرفق نصف العيش» (ع‑ب ص٣٨٢)؛ والأصمعي «بكثرة سؤالي وتلقفي الحكمة الشرود» (ع‑ب ص٣٨٢).
- **الخطيب:** في الفقيه حديث ابن عمر المرفوع «الاقتصاد في النفقة نصف المعيشة، والتودد إلى الناس نصف العقل، وحسن السؤال نصف العلم» (خ‑ف ص٦٤)، وعن ميمون بن مهران «التودد إلى الناس نصف العقل، وحسن المسألة نصف العلم» (خ‑ف ص٦٤)، وابن عباس «ما سألني أحد عن مسألة، إلا عرفت: فقيه أو غير فقيه» (خ‑ف ص٦٤)؛ وفي الجامع ينسب إلى ميمون «حسن المسألة نصف الفقه» (خ‑ج ج١ ص٢١٣).
- **العلموي:** يورد الحديث المرفوع بلا إسناد: «الاقتصاد في النفقة نصف المعيشة، والتودد إلى الناس نصف العقل، وحسن السؤال نصف العلم» (ع ص١٥٢).
- **أبو زيد:** «وللعلم ست مراتب: أولها: حسن السؤال» (ح ص١٨٨)، ينقلها عن ابن القيم؛ وتفصيل صيغ المراتب في ٠٥ م١٢.
> **تحرير:** المنسوب إلى ميمون بن مهران «نصف العلم» عند خ‑ف (ص٦٤) و«نصف الفقه» عند خ‑ج (ص٢١٣)، وكلاهما للخطيب. والمرفوع عن ابن عمر ورد عند ماو وخ‑ف وع. ولا أرجّح.

## ٤ — «اذهب فتعلم كيف تسأل»
**المسألة:** من اضطرب سؤاله ردّه العالم ليتعلم كيف يسأل ثم يسأل؛ وعلى السائل أن ينظر كيف يسأل.
**المصادر:** خ‑ج ج١ ص٢١٣ وخ‑ف ص٦٥، ٣٨٩ [٤٦٣]؛ ح ص١٨٨.
**الزيادات:**
- **الخطيب (الجامع):** جاء ابن عجلان زيد بن أسلم فسأله عن شيء فخلط عليه فقال له زيد: «اذهب فتعلم كيف تسأل، ثم تعال فسل» (خ‑ج ج١ ص٢١٣). **(الفقيه):** برواية مالك عن زيد: «اذهب فتعلم كيف تسأل، فإذا تعلمت فسل» (خ‑ف ص٦٥).
- **الخطيب (الفقيه):** عن ابن عباس: «إذا سأل أحدكم، فلينظر كيف يسأل، فإنه ليس أحد إلا وهو أعلم بما سأل عنه من المسئول» (خ‑ف ص٣٨٩).
- **أبو زيد:** «التزم أدب المباحثة من حسن السؤال، فالاستماع فصحة الفهم للجواب» (ح ص١٨٨)، ثم «فكن واضحا في السؤال، وقل: ما رأيك في الفتوى بكذا، ولا تسم أحداً» (ح ص١٨٨)، وينهى عن أن يقول بعد الجواب «لكن الشيخ فلان قال لي كذا» (ح ص١٨٨).
> **تحرير:** صيغة زيد بن أسلم تختلف بين «ثم تعال فسل» (خ‑ج) و«فإذا تعلمت فسل» (خ‑ف)، والراوي في الأول مالك عن ابن عجلان وفي الثاني مالك عن زيد؛ وهما للخطيب. وقد نصّ ج على قريب من نهي أبي زيد: «وهكذا لا يقول: قال فلان خلاف هذا» (ج ص١٠١) في مخاطبة الشيخ.

## ٥ — إيجاز السؤال وتحريره وتعيين المسؤول عنه
**المسألة:** يحسن السائلُ سؤاله بالإيجاز وتحرير العبارة وتعيين المسؤول عنه، لئلا يلتبس على المجيب.
**المصادر:** خ‑ف ص٦٣؛ خ‑ج ج١ ص٢١٣؛ ع‑ب ص٥٨٣–٥٨٤ [٤٦٣]؛ ج ص١٠٤.
**الزيادات:**
- **الخطيب (الفقيه):** «وينبغي أن يوجز السائل في سؤاله، ويحرر كلامه، ويقلل ألفاظه» ثم «فإن ذلك يدل على حسن معرفته» (خ‑ف ص٦٣).
- **الخطيب (الجامع):** تعيين الحديث المسؤول عنه: «فإن كان للحديث طرق متسعة نص السائل على أحسنها» (خ‑ج ج١ ص٢١٣).
- **ابن عبد البر:** في أرجوزة اللؤلؤي «فكن لما سمعته مستفهما ... إن أنت لا تفهم منه الكلما» (ع‑ب ص٥٨٣)، ووصية يحيى بن خالد لابنه جعفر «ولا تستح أن تستفهم إذا لم تفهم» (ع‑ب ص٥٨٤).
- **ابن جماعة:** في خطاب الشيخ: «وكذلك ينبغي أن يقول في موضع لم ولا نسلم فإن قيل لنا كذا أو فإن منعنا ذلك» (ج ص١٠٤) «ليكون مستفهما للجواب سائلا له بحسن أدب ولطف عبارة» (ج ص١٠٤).
- **الشهيد الثاني** (ش، OCR، بلا صفحة): يعدّ من الأمور المشتركة بين المعلم والمتعلم في الدرس تهذيب السؤال قبل إيراده، ومعه ترك السؤال تعنتاً (وانظر م١٠)؛ ولا يسمّي مصنفاً قبله.

## ٤٩ — مراتب الأبحاث الأربعة: هل، وما، وأي، ولم
**المسألة:** ترتيب الأسئلة العلمية في الشيء الواحد: يُسأل عن وجوده، ثم جنسه، ثم ما يميّزه عن غيره، ثم غرضه، ولا تصح معرفة اللاحق إلا بالسابق.
**المصادر:** ر ص١٥٥ [٥٠٢؛ المقروء الوحيد لهذه القسمة].
**الزيادات:**
- **الراغب:** «لأن الأبحاث أربعة: بحث عن وجود الشيء بهل هو، وبحث عن جنسه بما هو» (ر ص١٥٥)، «وبحث عما يباين به غيره بأي شيء هو، وبحث عن الغرض بلم هو؟» (ر ص١٥٥)؛ ثم «وهذه الأبحاث ينبني بعضها على بعض» (ر ص١٥٥)، «فلا تصح معرفة الثاني إلا بمعرفة الأول» (ر ص١٥٥) وهكذا إلى الرابع. ويحتج بقول المؤمنين {ربنا ما خلقت هذا باطلا} على أنهم عرفوا «الغرض المقصود بخلقه» (ر ص١٥٥) وهو آخر الأبحاث.
> **تحرير:** سياق الراغب معرفة الموجودات والاستدلال على الخالق، لا آداب السائل في الدرس؛ وأذكره لصلته بترتيب ما يُسأل عنه، ولا أحكم أن الراغب أراد به ضبط أسئلة المتعلم. وقد جاء عنده قبل ذلك (ر ص١٢٣) أن مساءلة العلماء من منازل الاتصال بأهل العلم (م١).

## ٦ — وقت السؤال: بعد تمام الكلام، وفي موضعه
**المسألة:** لا يسأل الطالب في أثناء كلام الشيخ، ولا في غير موضع السؤال، ولا وهو قائم أو مشغول؛ بل يتحين وقته. (وتقدمت في ٠٥ م١٤ و١٦؛ وتبسط هنا.)
**المصادر:** خ‑ج ج١ ص٢١١–٢١٢؛ خ‑ف ص٢٠٢، ٢٩٦–٢٩٨، ٣٨٠–٣٨١ [٤٦٣]؛ ر ص١٧٧؛ غ ص٥١؛ ج ص١٥٧؛ ع ص١٥٢؛ ح ص١٦٢.
**الزيادات:**
- **الخطيب (الجامع):** «ومن الأدب إذا روى المحدث حديثا، فعرض للطالب في خلاله شيء أراد السؤال عنه» ثم «أن لا يسأل عنه في تلك الحال، بل يصبر حتى ينهي الراوي حديثه» (خ‑ج ج١ ص٢١١)؛ وقصة عمر مع تميم الداري إذ جلس إليه فقال تميم «اتقوا زلة العالم» «فكره عمر أن يسأله عنه فيقطع على القوم» (خ‑ج ج١ ص٢١١) ثم أمر ابن عباس أن يسأله إذا فرغ؛ وابن عباس «إذا رأيته نائما لم أوقظه، وإذا رأيته مغموما لم أسأله» (خ‑ج ج١ ص٢١٢)؛ وأبو الطفيل «لكل مقام مقال» (خ‑ج ج١ ص٢١٢)؛ وابن المبارك لرجل سأله وهو يمشي «ليس هذا من توقير العلم» (خ‑ج ج١ ص٢١٢).
- **الخطيب (الفقيه):** «ولا ينبغي أن يستفهم من الفقيه حكم الفصل الذي يذكره له قبل أن يتمم الفقيه ذكره» (خ‑ف ص٢٠٢)؛ «وليتحين أن يسأله عند طيب نفسه» (خ‑ف ص٢٩٦)، «وليتق سؤاله عند الغضب» (خ‑ف ص٢٩٧)، «وكذلك لا يسأله حين يشتد فرحه» (خ‑ف ص٢٩٧)، «فليصدف عن سؤاله في تلك الحال» إذا رآه مشغول القلب (خ‑ف ص٢٩٨)؛ وفي أدب المستفتي «ولا يسأله قائما» (خ‑ف ص٣٨٠) «وإن رآه في هم قد عرض له ... أمسك عنه» (خ‑ف ص٣٨١).
- **الراغب:** يفرّق بين النهي عن السؤال قبل أوانه والأمر بالسؤال: «وليس ذلك نهيا عما حث الله تعالى عليه بقوله» (ر ص١٧٧) «وإنما هو نهي عن نوع من العلم الذي لم تبلغه منزلته بعد» (ر ص١٧٧)، في قصة موسى والخضر. **الغزالي** يقيّد الأمر بالسؤال بإذن المعلم: «ولكن فيما يأذن المعلم في السؤال عنه» (غ ص٥١) و«فإن السؤال عما لم تبلغ مرتبتك إلى فهمه مذموم» (غ ص٥١).
- **ابن جماعة:** «ولا يسأل عن شيء في غير موضعه إلا لحاجة أو علم بإيثار الشيخ ذلك» (ج ص١٥٧). **العلموي:** «ألا يسأل عن شيء في غير موضعه، ففاعل ذلك لا يستحق جوابا» (ع ص١٥٢)، «ويغتنم سؤاله عند طيب نفسه وفراغه» (ع ص١٥٢). **أبو زيد:** «متجنباً الإكثار من السؤال، ولا سيما مع شهود الملأ» (ح ص١٦٢).
> **تحرير:** الغزالي يقيّد السؤال بالمحل (ما يأذن المعلم في السؤال عنه)، والخطيب بالوقت (بعد تمام الكلام وطيب النفس)، وج بالموضع والإيثار، وع بالموضع مع التعليل بعدم استحقاق الجواب، وأبو زيد بالملأ؛ ولا تعارض صريح بينها. والراغب والغزالي كلاهما يقف عند قصة الخضر وموسى، والراغب أقدم (٥٠٢ مقابل ٥٠٥)، وكلام الراغب يُخرج الخضرَ من النهي عن السؤال المطلوب.

## ٧ — ملاطفة الشيخ الممتنع، وترك المماراة معه
**المسألة:** إن كان المحدث أو العالم ممن يتمنع بالرواية ويتعسر في التحديث، لاطفه الطالب في المسألة ورفق به؛ ولا يماريه فيحرم علمه.
**المصادر:** خ‑ج ج١ ص٢٠٨–٢١١؛ ع‑ب ص٥١٧–٥١٩؛ خ‑ف ص٢٩٥ [٤٦٣].
**الزيادات:**
- **الخطيب (الجامع):** «فينبغي للطالب أن يلاطفه في المسألة، ويرفق به ويخاطبه بالسؤدد، والتفدية، ويديم الدعاء له» (خ‑ج ج١ ص٢٠٨)؛ الزهري عن أبي سلمة «لو رفقت بابن عباس لاستخرجت منه علما كثيرا» (خ‑ج ج١ ص٢٠٩)، وفي الرواية الأخرى «كان أبو سلمة يسأل ابن عباس قال فكان يخزن عنه» (خ‑ج ج١ ص٢٠٩) «وكان عبيد الله بن عبد الله يلاطفه فكان يغره غرا» (خ‑ج ج١ ص٢٠٩)؛ وعمرو بن قيس الملائي يجلس بين يدي المحدث ويخفض جناحه ويقول «علمني رحمك الله مما علمك الله» (خ‑ج ج١ ص٢١٠)؛ وعن علي «ليس من أخلاق المؤمن التملق، ولا الحسد، إلا في طلب العلم» (خ‑ج ج١ ص٢١١).
- **ابن عبد البر:** ميمون بن مهران «لا تمار عالما ولا جاهلا؛ فإنك إذا ماريت عالما خزن عنك علمه» (ع‑ب ص٥١٧)؛ والزهري «كان أبو سلمة يماري ابن عباس فحرم بذلك علما كثيرا» (ع‑ب ص٥١٨)؛ وابن جريج «لم أخرج الذي قد استخرجت من عطاء إلا برفقي به» (ع‑ب ص٥١٩).
- **الخطيب (الفقيه):** «وينبغي أن يلاطف الفقيه إذا سأله ويحسن خطابه، وإن فداه بأبويه فلا بأس بذلك» (خ‑ف ص٢٩٥).
> **تحرير:** أبو سلمة وابن عباس خبر واحد يرويه خ‑ج وع‑ب بألفاظ متقاربة (يخزن عنه / حُرم علماً كثيراً)؛ وعبارة «يماري ابن عباس» عند ع‑ب مرويةٌ عن الزهري (ص٥١٨) وعن الشعبي (ص٥٢١)، فلا أنسب الخبر لواحد؛ أما لفظ «ماريت» فهو لميمون بن مهران في موضع آخر (ص٥١٧). وما يخص إعنات الشيخ والإلحاح عليه في م٩. وانظر الباب العاشر، المسألة ٣٠ (مماراة الأعلم والشيخ في الحلقة).

## ٨ — لا يعيد الاستفهام لما فهمه ولا يكرر
**المسألة:** يتقي الطالب إعادة السؤال عما فهمه أو سمعه، لأنه يضيّع الزمن ويضجر الشيخ؛ وله أن يسأل الإعادة إذا لم يسمع أو لم يفهم.
**المصادر:** خ‑ج ج١ ص١٩٦–١٩٧ [٤٦٣]؛ ج ص١٠٦؛ ع ص١٥١–١٥٢. (وفي ٠٣/٠٤ م٨ ما يتعلق بإصغاء الطالب؛ وهنا السؤال فقط.)
**الزيادات:**
- **الخطيب:** «وليتق إعادة الاستفهام لما قد فهمه، وسؤال التكرار لما قد سمعه وعلمه؛ فإن ذلك يؤدي إلى إضجار الشيوخ» (خ‑ج ج١ ص١٩٦)؛ شريك لمن قال أفهمني «ليس علي أن أفهمك، إنما علي أن أحدثك» (خ‑ج ج١ ص١٩٦)؛ ووكيع «من استفهم وهو يفهم، فهو طرف من الرياء» (خ‑ج ج١ ص١٩٧)؛ والزهري «نقل الصخر أهون من تكرار الحديث» (خ‑ج ج١ ص١٩٧).
- **ابن جماعة:** «ولا ينبغي للطالب أن يكرر سؤال ما يعلمه ولا استفهام ما يفهمه؛ فإنه يضيع الزمان وربما أضجر الشيخ» (ج ص١٠٦)، ويضيف: «فله أن يسأل الشيخ إعادته وتفهيمه بعد بيان عذره بسؤال لطيف» (ج ص١٠٦).
- **العلموي:** ينقل ذلك وزيادة قول الزهري «إعادة الحديث أشد من نقل الصخر» (ع ص١٥١)، «ولا ينبغي أن يكرر ما يعلمه، ولا استفهام ما يفهمه، فإنه يضيع الزمان، وربما أضجر الشيخ» (ع ص١٥١)، ثم «فله أن يسأل الشيخ إعادته أو تفهيمه بعد بيان عذره بسؤال لطيف» (ع ص١٥٢).
- **الخطيب (الفقيه):** «وإذا استفهم المتعلم الفقيه فأفهمه، ثم عاد فاستفهمه، جاز للفقيه أن يزيده» (خ‑ف ص٣١٥) «فله أن يأخذه بلسانه» إن راجعه بعد ذلك (خ‑ف ص٣١٦).
- **ابن عبد البر (استثبات الخبر):** عائشة لعروة بعد أن حدثها بحديث عن عبد الله بن عمرو: «انطلق إلى عبد الله فاستثبت منه الحديث الذي حدثتني به عنه» (ع‑ب ج٢ ص١٠٣٧)، فسأله عروة وعاد إليها فعجبت وقالت «والله لقد حفظ عبد الله بن عمرو» (ع‑ب ج٢ ص١٠٣٧).
> **تحرير:** خ‑ج (وكيع: «طرف من الرياء») يذم الاستفهام عما فُهم، وخ‑ف يجيز للفقيه أن يزيد المتعلم إذا أعاد الاستفهام ثم يأخذه بلسانه بعد ذلك؛ وج وع يجيزان الإعادة لعذر. وكلها في استفهام ما يُفهم، لا في الاستيضاح فيما لم يُفهم (م٢). وسؤال عائشة هنا استثبات للخبر عند مصدره لا سؤال عما فُهم، فلا يدخل في كراهة الخطيب.

## ٩ — الإكثار من السؤال والإلحاح: بين الحثّ والنهي
**المسألة:** هل يُكثر الطالب من السؤال ويُلحّ فيه، أم يخفف عن الشيخ؟ النصوص تحمل الاتجاهين.
**المصادر:** الكافي (الكليني ت ٣٢٩، OCR، بلا صفحة)؛ ع‑ب ص٣٨٠–٣٨١، ٥١٩، ٥٨٠ وج١ ص٧٧٤–٧٧٥ [٤٦٣]؛ خ‑ج ج١ ص٢١٤–٢١٥ وج٢ ص٩٦؛ ماو ص٧٠؛ غ ص٥١؛ ج ص١٥٧؛ ح ص١٦٢.
**الزيادات:**
- **ما يحثّ:** الخليل «ولا تجزع بتفريع السؤال؛ فإنه ينبهك على علم ما لم تعلم» (ع‑ب ص٣٨١)؛ وابن المبارك لرجل مستحيٍ «فأعنت الشيخ بالسؤال تجده سلسا يلقاك بالراحتين» (ع‑ب ص٣٨١)؛ والماوردي «وليس كثرة السؤال فيما التبس إعناتا» (ماو ص٧٠).
- **ما ينهى:** علي في حق العالم: «من حق العالم ألا تكثر عليه بالسؤال، ولا تعنته بالجواب وأن لا تلح عليه إذا كسل» (ع‑ب ص٥١٩)، وفي رواية أخرى عند ع‑ب نفسه «ولا تلح عليه في السؤال» (ع‑ب ص٥٨٠)، وعند غ «أن من حق العالم أن لا تكثر عليه بالسؤال ولا تعنته في الجواب ولا تلح عليه إذا كسل» (غ ص٥١). وعند الخطيب باب «كراهة إملال الشيوخ إذا أجاب المحدث الطالب إلى مسألته وحدثه، فيجب أن يأخذ منه العفو ولا يضجره» (خ‑ج ج١ ص٢١٤)، ويحيى القطان «كنت آخذ العفو في الحديث» (خ‑ج ج١ ص٢١٤)، ومالك لمن سأله خامساً «يا هذا، ما هذا بإنصاف» (خ‑ج ج١ ص٢١٥)، وابن سيرين «إنك إن كلفتني ما لم أطق ساءك ما سرك مني من خلق» (خ‑ج ج١ ص٢١٥).
- **الكليني** (الكافي، باب حق العالم، OCR، بلا صفحة): يروي وصية لأمير المؤمنين في حق العالم على المتعلم، أولها ألا يكثر عليه السؤال ولا يؤذيه بقوله، وفيها ألا يكثر من ذكر قال فلان وقال فلان خلافاً لقوله، وألا يضجر بطول صحبته؛ وهي قريبة من وصية علي عند ع‑ب (ص٥١٩) وغ (ص٥١)، ولم أطابق ألفاظها بالمطبوع لأنها OCR.
- **ما يجمع الأمرين (ع‑ب ج١ ص٧٧٤–٧٧٥):** أحمد وقد عاوده السائل في مسألة زكاة الدنانير والدراهم: «برأي أستعفي منها، وأخبرك أن فيها اختلافا» (ص٧٧٤)؛ فلما ألح عليه: «وتلح علي تقول: فما تقول أنت فيها؟» (ص٧٧٥)، وقال له رجل «لا بد أن نعرف مذهبك في هذه المسألة لحاجتنا إليها» (ص٧٧٥) فغضب وقال «أي شيء بد إذا هاب الرجل شيئا يحمل على أن يقول فيه» (ص٧٧٥)؛ ويقول الأثرم «ولم يزل به السائل حتى جعل يجنح لقول من لا يرى الجمع بينهما» (ص٧٧٥).
- **الخطيب (الجامع) [زيادة]:** في باب «من روى حديثا ذكر أنه سأل شيخه عنه حتى حدثه به» يروي عن شعبة: «سألت طلحة بن مصرف عن هذا الحديث، أكثر من عشرين مرة ولو كان غيري قال ثلاثين مرة» (خ‑ج ج٢ ص٩٦)، ولا يعقّب عليه بكراهة.
- **ابن جماعة:** «وإذا سكت الشيخ عن الجواب لم يلح عليه» (ج ص١٥٧). **أبو زيد:** «أو الإلحاح عليه في جواب، متجنباً الإكثار من السؤال، ولا سيما مع شهود الملأ» (ح ص١٦٢)، ويعلل: «فإن هذا يوجب لك الغرور وله الملل» (ح ص١٦٢).
> **خلاف وتحرير:** (أ) ع‑ب نفسه يجمع الأمر بإعنات الشيخ بالسؤال (ابن المبارك، ص٣٨١) والنهي عن الإكثار والإلحاح (علي، ص٥١٩ و٥٨٠)، ومادة (ع ن ت) ترد في الباب بمعنيين: «فأعنت الشيخ بالسؤال» (ص٣٨١) «ولا تعنته بالجواب» (ص٥١٩)؛ ولا أحكم أنهما متعارضان لاحتمال اختلاف الحال (طالب حيي أُمر بالإقدام، وطالب مُلحّ نُهي)، ولم أجد نصاً يصرّح بهذا الجمع إلا ما ذكره الماوردي في م١٠. (ب) الخطيب يوازن في الجامع بين باب «أدب السؤال» (خ‑ج ج١ ص٢٠٢) وباب «كراهة إملال الشيوخ» (خ‑ج ج١ ص٢١٤)، ويصف الإضجار بأنه «يغير الأفهام، ويفسد الأخلاق، ويحيل الطباع» (خ‑ج ج١ ص٢١٧). (ج) لا مقدار محدد للأسئلة في أي من المصادر؛ إنما ورد عدد الأحاديث عند التحديث (الخامس عند مالك، ثلاثة عند أبي قلابة، ص٢٠٧)، وليس ذلك عدد أسئلة. (د) أحمد عند ع‑ب يمتنع عن الجواب برأيه مع إلحاح السائل ثم يغضب منه (ص٧٧٤–٧٧٥)، فهو إلحاح على الفتوى لا على الرواية؛ وفي المقابل يذكر الخطيب سؤال شعبة طلحةَ أكثر من عشرين مرة بلا كراهة (خ‑ج ج٢ ص٩٦)؛ فلا يظهر تعارض، ولا أرجّح.

## ١٠ — السؤال تفقهاً لا تعنتاً، وتوفيق الماوردي
**المسألة:** يسأل الطالب ليعلم ما جهل، لا ليُعنت العالم ويُحرجه؛ وقد جاء الأمر بالسؤال والنهي عن كثرته، فجمع الماوردي بينهما بقصد السائل.
**المصادر:** ماو ص٦٩–٧٠ [أقدم مقروء]؛ ع‑ب ص٤٦٤، ٤٦٨، ٥١٥ وخ‑ف ص١٩، ٣٠، ٨٠، ٨٢ [٤٦٣]؛ ع ص١٠٥، ١٥٢؛ ح ص١٨٨؛ ش (OCR، بلا صفحة)؛ مق هامش ص٢٣٠ (نقل عن النووي)؛ وع‑ب ج٢ ص١٠٥٩ في التحرير.
**الزيادات:**
- **الماوردي:** يصف طالباً قوياً في نفسه «فقصد من يعلمه بالإعنات له والاعتراض عليه إزراء به وتبكيتا له» (ماو ص٦٩) ويورد «أعلمه الرماية كل يوم ... فلما اشتد ساعده رماني» (ماو ص٦٩)؛ ثم يقول «وإنما أمر بالسؤال من قصد به علم ما جهل، ونهى عنه من قصد به إعنات ما سمع» (ماو ص٧٠) بعد أن أورد «أنهاكم عن قيل وقال وكثرة السؤال وإضاعة المال» (ماو ص٧٠) و«إياكم وكثرة السؤال فإنما هلك من قبلكم بكثرة السؤال» (ماو ص٧٠).
- **ابن عبد البر:** علي لابن الكواء «ويلك سل تفقها ولا تسل تعنتا» ثم «أعمى سأل عن عمياء» (ع‑ب ص٤٦٤)؛ وفي موضع آخر «إنك لذهاب في التيه سل عما ينفعك أو يعينك» فقال ابن الكواء «إنما نسأل عما لا نعلم» (ع‑ب ص٤٦٨)؛ وحديث «علموا ولا تعنتوا؛ فإن المعلم خير من المعنت» (ع‑ب ص٥١٥) ويشير إلى رواية «تعلموا ولا تعنتوا فإن المتعلم خير من المعنت» (ع‑ب ص٥١٥).
- **الخطيب:** في تقسيم الأسئلة: من قال «لا أسلم لك» ولم ينقل الكلام «كان متعنتا مطالبا للمسئول بما لا يجب عليه» (خ‑ف ص٨٠)، وفي السؤال عن وجه الدليل: «وإن كان الدليل ظاهرا جليا لم يجز هذا السؤال، وكان السائل عنه متعنتا أو جاهلا» (خ‑ف ص٨٢)؛ ويحمل تحريج عمر على أنه «قصد به السؤال على سبيل التعنت والمغالطة، لا على سبيل التفقه وابتغاء الفائدة» (خ‑ف ص١٩)، وينقل عن المزني «وكما روي عن علي من إنكاره على ابن الكواء أن يسأل تعنتا، وأمره أن يسأل تفقها» (خ‑ف ص٣٠).
- **العلموي:** «ألا يسأل أحدا تعنتا وتعجزا، فإنه لا يستحق جوابا» (ع ص١٠٥). **أبو زيد** ينقل عن ابن القيم: «وقيل: إذا جلست إلى عالم، فسل تفقها لا تعنتاً» (ح ص١٨٨).
- **الشهيد الثاني** (ش، OCR، بلا صفحة): يعدّ من أمور الدرس المشتركة بين المعلم والمتعلم ترك السؤال تعنتاً ويشرح معه المراء وعلاجه، وتهذيب السؤال قبل إيراده؛ ولا يسمّي مصنفاً قبله.
- **المقدم** (هامش ص٢٣٠): في من طلب العلم ليجاري به العلماء: «وحق من فعل هذا أن يعرض عنه، ولا يجاب إلا بالسكوت» (مق هامش ص٢٣٠)، ثم ينقل عن النووي: «السائل تعنتا وتعجيزا لا يستحق جوابا» (مق هامش ص٢٣٠، عن المجموع ١/٣٩؛ نقل مق ولم أراجع أصله).
> **تحرير:** الماوردي (٤٥٠) أقدم من وقفت عليه يعلّل الأمر والنهي بقصد السائل ويصرّح بذلك؛ وع‑ب وخ‑ف يسندان العبارة إلى علي (٤٦٣)، والخطيب يزيد التمييز الفقهي بين السؤال المشروع والتعنت في التناظر. ولفظ المعنِّت/المتعنّت لا يتحد معناه عند الجميع: ففي حديث «علموا ولا تعنتوا» الخطاب بصيغة التعليم عند ع‑ب وهو ظاهر في المعلم، وفي رواية «تعلموا ولا تعنتوا» في المتعلم؛ ولم أجد في ع‑ب ترجيحاً بين الروايتين.
> **تحرير (من القراءة الجديدة):** عبارة النووي في نقل المقدم «السائل تعنتا وتعجيزا لا يستحق جوابا» قريبة من عبارة ع «ألا يسأل أحدا تعنتا وتعجزا، فإنه لا يستحق جوابا» (ع ص١٠٥)، وقد صرّح ع بالنقل عن النووي في موضع آخر (ص١٣٠) ولم أجد تصريحاً بذلك في ص١٠٥ نفسها، فلا أجزم بأنها مأخوذة منه. وفي حديث «أنهاكم عن قيل وقال وكثرة السؤال» تردد مالك في معنى كثرة السؤال فقال: «فلا أدري أهو هذا أم السؤال في مسألة الناس في الاستعطاء؟» (ع‑ب ج٢ ص١٠٥٩)، فيحتمل أن يكون المراد سؤال المال لا سؤال العلم؛ ولم أجد عند الماوردي (ص٦٩–٧٠) تعرضاً لهذا الاحتمال، وتوفيقه مبني على سؤال العلم.


## ١١ — الاستحياء من سؤال الشيخ: يسأل بواسطة غيره
**المسألة:** من غلبه الحياء عن مواجهة العالم بسؤاله ألقى مسألته إلى من يأنس به ليسأل عنها؛ فيكون الحياء مانعاً من المواجهة لا من العلم. (وأصل النهي عن الحياء في م٢.)
**المصادر:** ع‑ب ج١ ص٣٧٤ [٤٦٣]؛ خ‑ف ج٢ ص٢٩٩، ٣٠١–٣٠٢ [٤٦٣ كذلك فلا أحكم بأقدمية].
**الزيادات:**
- **ابن عبد البر:** «واستحيا علي رضي الله عنه أن يسأل عن المذي لمكان رسول الله صلى الله عليه وسلم من ابنته التي كانت عنده» فأمر المقداد وعماراً أن يسألا (ع‑ب ص٣٧٤).
- **الخطيب (الفقيه):** يصوغ القاعدة: «ألقى مسألته إلى من يأنس به وينبسط عليه» ليسأل الفقيه عنها ويخبره بحكمها (خ‑ف ص٢٩٩)، ويستشهد بقول علي للمقداد: «فلولا أن ابنته عندي لاستفتيته» (خ‑ف ص٢٩٩). ويزيد حالين: أبا موسى حين قال لعائشة إنه يستحي أن يسألها فقالت «لا تستحي فإنما أنا أمك» (خ‑ف ص٣٠١)، والأسود ومسروقاً إذ أتيا عائشة «فاستحيينا، فقمنا قبل أن نسألها» فقالت لهما «سلا عما بدا لكما» (خ‑ف ص٣٠٢).
> **تحرير:** الخبر عن علي والمقداد عند ع‑ب والخطيب واحد في المعنى والتعليل (وجود ابنة النبي عنده)، فلا خلاف؛ وما زاده الخطيب من خبر عائشة مع أبي موسى ومع الأسود ومسروق ليس عند ع‑ب في الموضع الذي قرأته (ص٣٧٣–٣٨٧).

## ١٢ — «سل سؤال الأحمق» واستئذان العالم قبل السؤال
**المسألة:** وردت عبارتان تتعلقان بصفة السؤال: أمر بأن يسأل المرء «سؤال الأحمق» ويحفظ «حفظ الأكياس»، وسؤال لعالم عن إذنه في السؤال. ولم أجد عند الخطيب شرحاً للعبارة الأولى.
**المصادر:** خ‑ف ج٢ ص٣٠٠–٣٠٤ [٤٦٣]. (لم أجد العبارة في بقية المصادر المقروءة لهذا الباب.)
**الزيادات:**
- **الخطيب (الفقيه):** يقدّم لها بقوله «والأولى: أن يكون هو السائل للفقيه عن الأمور التي تصلح دينه» (خ‑ف ص٣٠٠)؛ ثم إبراهيم لمنصور: «يا منصور، سل سؤال الحمقى، واحفظ حفظ الأكياس» (خ‑ف ص٣٠٢)، ومثله عن الأعمش «سل سؤال الأحمق، واحفظ حفظ الأكياس» (خ‑ف ص٣٠٣)؛ وعبد العزيز بن عمر بن عبد العزيز: «كنت أستحي أن يرى مثلي يسأل عن مثلها» فبقيت جهالتها فيه (خ‑ف ص٣٠٣).
- **الخطيب في الاستئذان:** قول رجل لإياس بن معاوية «أتأذن في مسألة؟» فأجابه «والله، ما استربت بك حتى استأذنتني» (خ‑ف ص٣٠٤)، وشرطه «فإن كنت لا تؤذي جليسك ولا تفضح عالمك فهاتها» (خ‑ف ص٣٠٤).
> **تحرير:** الخطيب يسوق عبارة «سؤال الأحمق» بلا بيان لمعناها ولا يضعها في سياق الحياء صراحة، وإن جاءت بعد أخبار الحياء (ص٣٠٠–٣٠٣) وقبل خبر إياس؛ وكل ربط بينها وبين الحياء استنتاج لا نص فيه. أما قول إياس فقد قابله الخطيب بما سبق في م٦ من اختيار وقت السؤال. وفي حمقى السائلين وجواب العالم عليهم: م٥٠ هنا.

## ١٣ — ما يُسأل عنه: النافع المعتنى به دون الغريب
**المسألة:** يحسن أن يكون سؤال الطالب فيما يعنيه ويكثر نفعه ووقوعه، ويقلّ فيما لا طائل تحته من العضل والغرائب.
**المصادر:** ماو ص٧٠ [٤٥٠: «في موضعه»]؛ ع‑ب ج١ ص٤٦٨ وص٧٥٧–٧٥٨ وج٢ ص١٠٦٢ وخ‑ف ج٢ ص٣٠٣–٣٠٤ وخ‑ج ج١ ص٢٢٤–٢٢٧ [٤٦٣]؛ ع ص١٧٥؛ مق ص٢٤٨ (معاصر، ينقل عن السلف).
**الزيادات:**
- **الماوردي:** «وإذا كان السؤال في موضعه أزال الشكوك ونفى الشبهة» (ماو ص٧٠).
- **ابن عبد البر:** علي لابن الكواء: «سل عما ينفعك أو يعينك» (ع‑ب ج١ ص٤٦٨)؛ وابن عباس في أصحاب النبي: «ما كانوا يسألون إلا عما ينفعهم» (ع‑ب ج٢ ص١٠٦٢) مع تعقيب ابن عبد البر: «ليس في الحديث من الثلاث عشرة مسألة إلا ثلاث» (ع‑ب ج٢ ص١٠٦٢).
- **الخطيب (الفقيه):** «وينبغي أن تكون مساءلته عما يكثر نفعه ويقل المساءلة عن العضل والأغلوطات» (خ‑ف ص٣٠٣)، ويورد حديث المغيرة أن النبي «ينهى عن قيل وقال، وكثرة السؤال، وإضاعة المال» (خ‑ف ص٣٠٤). **الخطيب (الجامع):** يروي أن رجلاً جاء شعبة من مسيرة ستة أشهر يسأله عن حديث فقال شعبة «يسألني عن حديث لا يحل حراما، ولا يحرم حلالا» ثم أملى عليه حديث النائم عن الصلاة وقال «وإلا فقد ذهبت رحلتك باطلا» (خ‑ج ج١ ص٢٢٦).
- **العلموي:** ابن عباس لعكرمة: «فمن جاءك يسألك عما يعنيه فأفته، ومن سألك عما لا يعنيه فلا تفته» (ع ص١٧٥)، وهو بنصه في المقدم (مق ص٢٤٨) الذي يزيد جواب أبي الدرداء للسائل «فما تصنع بازدياد حجة الله عليك» (مق ص٢٤٨).
- **ابن عبد البر (من القراءة الجديدة):** مالك «العلم والحكمة نور يهدي الله به من يشاء، وليس بكثرة المسائل» (ع‑ب ج١ ص٧٥٧)، و«ليس الفقه بكثرة المسائل ولكن الفقه يؤتيه الله من يشاء من خلقه» (ع‑ب ج١ ص٧٥٨)؛ وابن مسعود «ليس العلم عن كثرة الحديث، إنما العلم خشية الله» (ع‑ب ج١ ص٧٥٨).
- **الخطيب (الجامع) [زيادة، في باب ما ينبغي أن يُسأل الراوي عنه من أحاديثه]:** بعد خبر يزيد بن هارون في حديث شريك ينبّه على أن شريكاً «ليس من قدماء شيوخه، ولا أثباتهم» (خ‑ج ج١ ص٢٢٤)، ويروي عن المقرئ أن أبا حنيفة حدّث عن مرجئ فقال «أبيعكم اللحم مع العظام» (خ‑ج ج١ ص٢٢٤)، ثم يقول: «فينبغي للطالب أن يسأل الراوي عن عيون أحاديثه التي ثبتت أسانيدها، وتقدم سماعه لها» (خ‑ج ج١ ص٢٢٤)؛ وبعد خبر شعبة والمغربي: «وإذا لم يكن الطالب ممن يعرف الأحاديث التي يسأل المحدث عنها، استعان بمن حضر المجلس من أهل الحفظ، والمعرفة» (خ‑ج ج١ ص٢٢٦)، وإن لم يحضر الشيخ أحد منهم «فينبغي للطالب أن يقدم الاستخبار عن ذلك بعض حفاظ الحديث قبل حضوره المجلس، ويعلق أطراف الأحاديث حتى يسأل الراوي عنها» (خ‑ج ج١ ص٢٢٦)؛ وابن مهدي في موت الثوري: «وجدنا في حجزته رقاعا فيها أطراف ليسأل عنها» (خ‑ج ج١ ص٢٢٧)، وإبراهيم لحماد وقد جعل يسأله عن أطراف: «ما هذا أصلحك الله، إنما هي أطراف» (خ‑ج ج١ ص٢٢٧)، ثم «لا بأس بكتابة الأطراف» (خ‑ج ج١ ص٢٢٧).
- وقول ابن شبرمة فيما لا يجمل بالسائل السؤال عنه (خ‑ف ص٤١٧)، وقول مالك للسائل «لو سألت عما تنتفع به» (خ‑ف ص٤١٨): في الباب الثالث عشر، المسألة ٦؛ وحمقى السائلين عند الخطيب (خ‑ف ص٤١٩): في م٥٠ هنا.
> **تحرير:** ابن عباس عند ع‑ب وصف أصحاب النبي في السؤال، وعند ع (ومنه مق) وصية لمن يفتي؛ فالموضوع واحد (النفع)، والمخاطَب مختلف (السائل / المفتي). وخبر شعبة يجعل معيار النفع في الحديث ما يُحلّ ويحرّم، وفي الجامع للخطيب أن معيار الطالب الراحل أن يسأل عن الأحاديث الثابتة المتقدم سماعها لا عن النازل من حديث الراوي (خ‑ج ج١ ص٢٢٤)، وهو غير معيار شعبة (ما يُحلّ ويحرّم)؛ ولا أرجّح. ولا أجد في المصادر المقروءة تحديداً آخر للنافع.

## القسم الثاني: الإكثار من المسائل، والسؤال عما لم يقع

## ١٤ — الكراهة: ما رُوي في ترك السؤال عن الحادثة قبل وقوعها
**المسألة:** رُوي عن النبي وعمر وابن عمر وزيد بن ثابت وأبي بن كعب وعمار ومعاذ وابن شهاب ومالك ما يدل على كراهة الكلام في الحادثة قبل نزولها؛ وساق ذلك الخطيب ثم ردّه (م١٥)، وساقه ابن عبد البر في سياق ذم الرأي.
**المصادر:** خ‑ف ج٢ ص١١–١٦، ٢٣ وع‑ب ج٢ ص١٠٥٣–١٠٦٨ [٤٦٣ كلاهما فلا أحكم بأقدمية]؛ ب‑ص ص١٠٩ [٦٤٣]؛ ع ص١٧٥؛ مق ص٢٤٩–٢٥٠ (معاصر).
**الزيادات:**
- **الخطيب (الفقيه):** حديث أبي هريرة «فإنه إنما هلك من كان قبلكم بكثرة سؤالهم، واختلافهم على أنبيائهم» (خ‑ف ص١١)، وسهل بن سعد «كره رسول الله صلى الله عليه وسلم المسائل وعابها» (خ‑ف ص١١)؛ وعمر: «أحرج عليكم أن تسألونا عما لم يكن، فإن لنا فيما كان شغلا» (خ‑ف ص١٢)؛ وابن عمر: «لا تسألوا عما لم يكن، فإني سمعت عمر يلعن السائل عما لم يكن» (خ‑ف ص١٢)؛ وزيد بن ثابت: «دعوه حتى يكون» (خ‑ف ص١٣)؛ وأبي بن كعب: «فأجمنا حتى يكون، فإذا كان اجتهدنا لك رأينا» (خ‑ف ص١٤)؛ وعمار: «فدعونا حتى يكون، فإذا كان تجشمناه لكم» (خ‑ف ص١٥)؛ وابن شهاب لموسى بن علي: «ما سمعت فيه بشيء وما نزل بنا» (خ‑ف ص١٥)؛ ومالك: «أدركت هذه البلدة وإنهم ليكرهون هذا الإكثار الذي فيه اليوم» (خ‑ف ص١٥)؛ ومعاذ: «لا تعجلوا بالبلاء قبل نزوله» (خ‑ف ص٢٣).
- **ابن عبد البر** (في ج٢ ص١٠٥٣–١٠٦٨ التي قرأتها): ابن عمر عن عمر «لا تسألوا عما لم يكن؛ فإني سمعت عمر، يلعن من سأل عما لم يكن» (ع‑ب ص١٠٥٥)؛ وعمر على المنبر «أحرج بالله على كل امرئ سأل عن شيء لم يكن» (ع‑ب ص١٠٦١)؛ ومعاذ مرفوعاً «لا تعجلوا بالبلية قبل نزولها» (ع‑ب ص١٠٦٣)؛ وعمر «إنه لا يحل لأحد أن يسأل عما لم يكن» (ع‑ب ص١٠٦٤)؛ وأبي بن كعب لمسروق «أكانت هذه بعد؟» (ع‑ب ص١٠٦٥)؛ وزيد «أوقعت؟» (ع‑ب ص١٠٦٥) و«آلله أكان هذا؟» (ع‑ب ص١٠٦٨)؛ وابن شهاب لعبد الملك بن مروان «أكان هذا يا أمير المؤمنين؟» (ع‑ب ص١٠٦٧). ويسوقها على لسان من يذمّ الرأي: «قالوا: ألا ترى أنهم كانوا يكرهون الجواب في مسائل الأحكام ما لم تنزل» (ع‑ب ص١٠٦٣).
- **ابن الصلاح:** «وإذا سأل العامي عن مسألة لم تقع لم تجب مجاوبته» (ب‑ص ص١٠٩).
- **العلموي:** يورد قول ابن عمر «لا تسأل عما لم يكن» (ع ص١٧٥). **المقدم:** «وقد وردت آثار عن السلف فيها النهي عن السؤال عما لم يقع حتى يقع» (مق ص٢٤٩).
- **ابن عبد البر (تتمة):** عبدة بن أبي لبابة «يتكاثرون بالمسائل كما يتكاثر أهل الدراهم بالدراهم» (ع‑ب ج٢ ص١٠٥٩)؛ وحديث سعد «أعظم المسلمين في المسلمين جرما من سأل عن شيء لم يحرم على المسلمين فحرم عليهم من أجل مسألته» (ع‑ب ج٢ ص١٠٦٠).
- **المقدم** (هامش ص٢٤٤): ينقل من الموطأ أن عمر خرج في ركب فيهم عمرو بن العاص حتى وردوا حوضاً، فقال عمرو لصاحب الحوض: «يا صاحب الحوض هل ترد حوضك السباع؟» فقال عمر: «يا صاحب الحوض! لا تخبرنا، فإنا نرد على السباع، وترد علينا» (مق هامش ص٢٤٤)؛ وفي متنه عن ابن القيم الأثر نفسه بصيغة «يا صاحب الميزاب! لا تخبرنا» (مق ص٢٤٤)، ويقول المقدم في هامشه «لم أقف على تخريجه» (مق هامش ص٢٤٤).
> **تحرير:** (أ) لفظ ابن عمر عن عمر: «يلعن السائل عما لم يكن» عند خ‑ف (ص١٢)، و«يلعن من سأل عما لم يكن» عند ع‑ب (ص١٠٥٥)، وفي رواية أخرى عند خ‑ف «كان يلعن، أو يسب من سأل عما لم يكن» (ص١٣). (ب) قول أبي بن كعب: «أكان بعد» و«فأجمنا» عند خ‑ف (ص١٤)، و«أكانت هذه بعد؟» و«فأجمني» عند ع‑ب (ص١٠٦٥). (ج) راوي قول عمر: عمرو بن مرة عند خ‑ف (ص١٢)، وطاوس عند ع‑ب (ص١٠٦١، ١٠٦٤). (د) ع‑ب يسوق هذه الآثار في باب «ما جاء في ذم القول في دين الله تعالى بالرأي والظن والقياس على غير أصل وعيب الإكثار من المسائل دون اعتبار» (ج٢ ص١٠٣٧)، وقد عرض قبلها اختلاف العلماء في الرأي المذموم (ص١٠٥٢) ثم قال: «وقال آخرون وهم جمهور أهل العلم» (ص١٠٥٤)، «واحتجوا على صحة ما ذهبوا إليه من ذلك بأشياء منها» (ص١٠٥٤)، فهذه الآثار حجة الجمهور عنده في الرأي المذموم؛ وبعدها يحكي «قالوا: ألا ترى أنهم كانوا يكرهون الجواب في مسائل الأحكام ما لم تنزل» (ص١٠٦٣). وضابطه في م١٦؛ ولا أنسب إليه حكماً مستقلاً في جواز الكلام في الحادثة قبل وقوعها ما لم يصرّح به.

## ١٥ — الجواب عن الكراهة: الخطيب يجيز، وحجة المزني
**المسألة:** أجاز الخطيب الكلام في الحادثة قبل وقوعها، وعدّ أخبار المنع منتهية أو محمولة على قصد التعنت أو على الورع، وساق حجة المزني.
**المصادر:** خ‑ف ج٢ ص١٦–٣٥ [٤٦٣؛ المقروء الوحيد لهذا الجواب بنصه]؛ مق ص٢٥٩–٢٦٠ (معاصر، ينقل عن القاسمي).
**الزيادات:**
- **الخطيب — الوجوه:** (١) كراهة النبي «إشفاقا على أمته ورأفة بها» (خ‑ف ص١٦)، «وهذا المعنى قد ارتفع بموت رسول الله صلى الله عليه وسلم» (خ‑ف ص١٧). (٢) سؤال رافع بن خديج عن الذبح بالقصب في الغد، قال: «فلم يعب رسول الله صلى الله عليه وسلم مسألة رافع عما لم ينزل به» (خ‑ف ص١٨)؛ وسؤال الأشعث بن قيس المعترض «تسأل رسول الله صلى الله عليه وسلم عن أمر لم يحدث بعد؟» (خ‑ف ص١٨) ثم «فلم يمنع رسول الله صلى الله عليه وسلم هذا الرجل من مسألته» (خ‑ف ص١٩). (٣) تحريج عمر: «قصد به السؤال على سبيل التعنت والمغالطة، لا على سبيل التفقه وابتغاء الفائدة» (خ‑ف ص١٩)، ومثّل بضرب صبيغ: «ولهذا ضرب صبيغ بن عسل ونفاه» (خ‑ف ص١٩). (٤) فعل زيد وأبي وعمار: «توقوا القول برأيهم خوفا من الزلل» (خ‑ف ص٢٢)، «وهذا فعل أهل الورع والمشفقين على دينهم» (خ‑ف ص٢٣). (٥) الإجماع: «فكان ذلك إجماعا منهم على أنه جائز غير مكروه ومباح غير محظور» (خ‑ف ص٢٢).
- **الخطيب — كلام أحمد وتعليقه:** عن أحمد «من عرض نفسه للفتيا فقد عرضها لأمر عظيم، إلا أنه قد تجيء الضرورة» وأن الإمساك «أحب إلي لا شك» (خ‑ف ص٢٩)، ثم يعقّب الخطيب: «الإمساك أقرب إلى السلامة» (خ‑ف ص٢٩) ويرجّح ثواب المجتهد الناصح.
- **المزني (في نقل الخطيب):** «يقال لمن أنكر السؤال في البحث عما لم يكن لم أنكرتم ذلك؟» (خ‑ف ص٣٠)، ثم «فقد ارتفع ذلك برفع رسول الله صلى الله عليه وسلم» (خ‑ف ص٣٠)؛ ومن حججه أن على كل مسلم «أن يطلب الفرائض في الطهارة والصلاة والزكاة والصيام» قبل نزول ما يلزمه (خ‑ف ص٣١)؛ ومثال المجوسي الراغب في الإسلام الذي يقول «علموني الدخول في الإسلام» (خ‑ف ص٣٤) ثم يسأل عن أحكام الطهارة والزكاة والقصاص، وسؤالهم الساخر «أم تقولون: لا نخبرك حتى تنزل بك نازلة» (خ‑ف ص٣٥).
- **المقدم:** ينقل عن القاسمي عن بعض المفسرين: «لابد من تقييد النهي في هذه الآية بما لا تدعو إليه حاجة» (مق ص٢٥٩)، ثم يردّ القاسمي بأن الآية بقيدها «غنية عن أن تقيد بقيد آخر» (مق ص٢٥٩) (نقل مق عن القاسمي، ولم أراجع أصله).
> **تحرير:** الخطيب يرتّب الحجة على أن علة الكراهة النبوية (خشية فرض ما شق) زالت بوفاة النبي، ثم يحمل نهي الصحابة على التعنت (عمر) أو الورع (زيد وأبي وعمار)؛ ولا يذكر في ما قرأته أن أحداً من الصحابة صرّح بخلاف هذا الحمل. وعند أحمد في نقله تفضيل الإمساك مع إقرار الضرورة، وقد عقّب عليه الخطيب بترجيح بذل الاجتهاد، فلا يظهر اتفاقهما.

## ١٦ — ضابط المحمود والمذموم من الأسئلة (ابن عبد البر، ثم نقل المقدم عن ابن حجر والشاطبي)
**المسألة:** ما حدّ ما يُكره من السؤال وما يُحمد؟ أقدم ضابط وقفت عليه ما ساقه ابن عبد البر في تحديد «الرأي المذموم» الذي وردت فيه آثار كراهة المسائل؛ ثم ما نقله المقدم عن ابن حجر والشاطبي، وفيه قسمة أوفى.
**المصادر:** ع‑ب ج٢ ص١٠٥٢–١٠٥٤ [٤٦٣، أقدم مقروء]؛ مق ص٢٥٢–٢٥٨ (معاصر؛ ما فيه منسوباً إلى ابن حجر والشاطبي هو نقل المقدم عنهم ولم أراجع أصولهم).
**الزيادات:**
- **ابن عبد البر:** «اختلف العلماء في الرأي المقصود إليه بالذم والعيب في هذه الآثار المذكورة في هذا الباب» (ع‑ب ج٢ ص١٠٥٢)؛ فقالت طائفة: «الرأي المذموم هو البدع المخالفة للسنن في الاعتقاد كرأي جهم وسائر مذاهب أهل الكلام» (ع‑ب ص١٠٥٢)، وقال آخرون «وهم جمهور أهل العلم»: «الرأي المذموم في هذه الآثار عن النبي صلى الله عليه وسلم وعن أصحابه والتابعين هو القول في أحكام شرائع الدين بالاستحسان والظنون، والاشتغال بحفظ المعضلات والأغلوطات» (ع‑ب ص١٠٥٤)، ويصف ذلك بقوله «فاستعمل فيها الرأي قبل أن تنزل وفرعت وشققت قبل أن تقع، وتكلم فيها قبل أن تكون بالرأي المضارع للظن» (ع‑ب ص١٠٥٤)، ويقول إن «في الاشتغال بهذا والاستغراق فيه تعطيل السنن» (ع‑ب ص١٠٥٤).
- **ابن حجر (بواسطة مق):** البحث فيما لا نص فيه قسمان؛ الأول: «أن يبحث عن دخوله في دلالة النص على اختلاف وجوهها» وهذا «مطلوب لا مكروه» (مق ص٢٥٢)؛ والثاني: أن يدقق في الفروق بما لا أثر له في الشرع، وهو «الذي ذمه السلف» (مق ص٢٥٢). ويحكم على من سدّ باب المسائل بأنه «يقل فهمه وعلمه»، وعلى من توسّع في التفريع للمباهاة بأنه «يذم فعله» (مق ص٢٥٣).
- **الشاطبي (بواسطة مق):** «الإكثار من الأسئلة مذموم» (مق ص٢٥٥)، ثم يعدّ عشرة مواضع تُكره فيها الأسئلة: ما لا ينفع في الدين، وبعد بلوغ الحاجة من العلم، ومن غير احتياج في الوقت، وصعاب المسائل، وعلة الحكم التعبدي، والتكلف والتعمق، ومعارضة الكتاب والسنة بالرأي، والمتشابهات، وما شجر بين السلف، والتعنت والإفحام (مق ص٢٥٥–٢٥٨). ويختم بقوله: «وليس النهي فيها واحدا، بل فيها ما تشتد كراهيته، ومنها ما يخف، ومنها ما يحرم» (مق ص٢٥٨). ومن أمثلته قول مالك لمن سأل عن الاستواء: «الاستواء معلوم، والكيف مجهول، والسؤال عنه بدعة» (مق ص٢٥٧).
- **أمثلة في متن المقدم وهوامشه للمواضع الخامس والسادس والسابع:** عائشة وقد سئلت عن قضاء الحائض الصوم دون الصلاة: «أحرورية أنت؟» (مق هامش ص٢٥٦)؛ وعمر لصاحب الحوض: «يا صاحب الحوض! لا تخبرنا، فإنا نرد على السباع، وترد علينا» (مق ص٢٥٦)؛ وفي معارضة الكتاب والسنة بالرأي: «قال سعيد: أعراقي أنت؟» (مق ص٢٥٧) في خبره مع ربيعة، وجواب ربيعة «بل عالم متثبت، أو جاهل متعلم» (مق هامش ص٢٥٧)، وقول مالك لمن سأله أيجادل عن السنة: «لا، ولكن يخبر بالسنة، فإن قبلت منه، وإلا سكت» (مق ص٢٥٧).
> **تحرير:** (أ) ضابط ابن عبد البر في الرأي المذموم لا في السؤال وحده؛ وقد جعل الجمهور فيه الاشتغال بحفظ المعضلات والأغلوطات وتفريع الوقائع قبل وقوعها، وهو يقابل ما عدّه الشاطبي في نقل المقدم رابعاً (الأغلوطات، مق ص٢٥٦) وثالثاً (من غير احتياج في الوقت)، ويرتبط بـم١٤ وم١٧. (ب) نقلا ابن حجر والشاطبي في المقدم متأخران عن ابن عبد البر وسائر المصادر المقروءة لهذا الباب، ولم أجد عند الخطيب ولا ابن جماعة قسمة بهذا العدد. (ج) الأمثلة المسوقة في المقدم لمواضع الشاطبي لم أراجع أصولها.

## ١٧ — الأغلوطات وعضل المسائل وشرار المسائل
**المسألة:** نُهي عن الأغلوطات، وفُسّرت بصعاب المسائل ودقيقها وما لا يحتاج إليه؛ وذُمّ من يغلّط العلماء بها ويُعنت عباد الله.
**المصادر:** خ‑ف ج٢ ص٢٠–٢٢، ٣٠٣ وع‑ب ج٢ ص١٠٥٥–١٠٥٦، ١٠٧٣ [٤٦٣]؛ ع ص١٧٥–١٧٦؛ مق ص٢٥٦، ٢٦٠.
**الزيادات:**
- **الخطيب (الفقيه):** حديث الصنابحي عن رجل من الصحابة سمّاه: «نهى رسول الله صلى الله عليه وسلم عن الأغلوطات»، قال الأوزاعي: «شداد المسائل» (خ‑ف ص٢٠)؛ وعن معاوية بلفظ «الغلوطات» وتفسيرها «يعني دقيق المسائل» (خ‑ف ص٢٠–٢١)؛ وعيسى بن يونس: «والأغلوطات: ما لا يحتاج إليه من كيف وكيف» (خ‑ف ص٢١)؛ وحديث ثوبان «سيكون أقوام من أمتي يغلطون فقهاءهم بعضل المسائل، أولئك شرار أمتي» (خ‑ف ص٢١)؛ والحسن: «شرار عباد الله ينتقون شرار المسائل يعمون بها عباد الله» (خ‑ف ص٢٢).
- **ابن عبد البر:** عن معاوية «نهى عن الأغلوطات» (ع‑ب ص١٠٥٥) وبلفظ «نهى رسول الله صلى الله عليه وسلم عن الأغلوطات» (ع‑ب ص١٠٥٦) وتفسير الأوزاعي «يعني صعاب المسائل» (ع‑ب ص١٠٥٦) وروايته «نهى عن عضل المسائل» (ع‑ب ص١٠٥٦)؛ وعن الأوزاعي نفسه «إذا أراد الله عز وجل أن يحرم عبده بركة العلم ألقى على لسانه الأغاليط» (ع‑ب ص١٠٧٣)، وعن الحسن «إن شرار عباد الله الذين يجيئون بشرار المسائل يعنتون بها عباد الله» (ع‑ب ص١٠٧٣).
- **العلموي:** معاوية «نهى رسول الله صلى الله عليه وسلم عن الأغلوطات» (ع ص١٧٥) وحديث ثوبان بلفظ «يتعاطون فقهاؤهم عضل المسائل» (ع ص١٧٦). **المقدم:** يذكر عن أحمد وأبي داود النهي عن الأغلوطات ويفسرها «صعاب المسائل» (مق ص٢٦٠).
> **تحرير:** (أ) اللفظ «الأغلوطات» و«الغلوطات» في روايتين متتاليتين عند خ‑ف (ص٢٠). (ب) تفسيرها: شداد المسائل (الأوزاعي عند خ‑ف)، صعاب المسائل (الأوزاعي عند ع‑ب)، دقيق المسائل (خ‑ف ص٢١)، ما لا يحتاج إليه من «كيف وكيف» (عيسى بن يونس)، وعضل المسائل (معاوية عند ع‑ب، وثوبان عند خ‑ف). (ج) حديث ثوبان: «يغلطون فقهاءهم بعضل المسائل» (خ‑ف) و«يتعاطون فقهاؤهم عضل المسائل» (ع) — الأول يجعل السائل مغلِّطاً للفقيه، والثاني يجعل الفقهاء أنفسهم متعاطين لها. (د) يورد الخطيب النهي عن الأغلوطات في باب السؤال عن الحادثة قبل وقوعها (خ‑ف ص٢٠–٢١)، ثم يقول في أدب السائل «وينبغي أن تكون مساءلته عما يكثر نفعه ويقل المساءلة عن العضل والأغلوطات» (خ‑ف ص٣٠٣).

## ١٨ — «أرأيت» في السؤال
**المسألة:** وردت عن السلف كراهة صيغة «أرأيت» في السؤال، وسُمّي أصحابها الرأيتيين؛ وفي الخطيب ما فيه أن ابن مسعود سأل بها.
**المصادر:** خ‑ف ج١ ص٤٦٢، وج٢ ص٣١، وع‑ب ج٢ ص١٠٧٦–١٠٧٧ [٤٦٣]؛ مق ص٢٥١ (معاصر، ينقل عن السلف).
**الزيادات:**
- **ابن عبد البر:** أبو وائل «لا تقاعد أصحاب: أرأيت» (ع‑ب ج٢ ص١٠٧٦)، والشعبي «ما كلمة أبغض إلي من: أرأيت» (ع‑ب ص١٠٧٦)، وعنه «احفظ عني ثلاثا لها شأن: إذا سألت عن مسألة فأجبت فيها فلا تتبع مسألتك: أرأيت» (ع‑ب ص١٠٧٦) مستشهداً بآية الفرقان في من اتخذ إلهه هواه، ثم «والثالثة: إذا سئلت عما لا تعلم فقل: لا أعلم وأنا شريكك» (ع‑ب ص١٠٧٧)، وعنه أيضاً «إنما هلك من كان قبلكم في: أرأيت» (ع‑ب ص١٠٧٧).
- **الخطيب (الفقيه):** في باب ذم القياس: الشعبي «لقد بغض إلي هؤلاء القوم هذا المسجد، حتى لهو أبغض إلي من كناسة داري» (خ‑ف ج١ ص٤٦٢) وجوابه لمن قال له «من هو يا أبا عمرو؟»: «هؤلاء الرأيتيون، أرأيت أرأيت» (خ‑ف ج١ ص٤٦٢)؛ وفي الاحتجاج للجواز يسوق «وعن ابن مسعود في مساءلته عبيدة السلماني: أرأيت، أرأيت» (خ‑ف ج٢ ص٣١).
- **المقدم:** «وقال أبو وائل: لاتقاعد أصحاب: أرأيت» (مق ص٢٥١)، والشعبي «ما كلمة أبغض إلي من: أرأيت» (مق ص٢٥١)، وأنه قال: «إذا سألت عن مسألة فأجبت فيها، فلا تتبع مسألتك: أرأيت» (مق ص٢٥١)؛ وزبيد «ما سألت إبراهيم عن شيء إلا عرفت الكراهية في وجهه» (مق ص٢٥١). وألفاظ أبي وائل والشعبي هنا هي بنصها في ع‑ب (ج٢ ص١٠٧٦)، فينفرد المقدم عنه بقول زبيد.
- **المقدم (هامش ص٢٥١):** يشرح: «الأرأيتيون: الذين يكثرون من قول: أرأيت في غير موضعها كأن يسأل عن علة الحكم في أمر تعبدي، أو يكون السائل غير أهل لذلك» (مق هامش ص٢٥١).
> **خلاف:** الخطيب يجعل «أرأيت» من ابن مسعود دليلاً على جواز السؤال عما لم يقع (ج٢ ص٣١)، وهو نفسه ينقل في ذم القياس ذمّ الشعبي للرأيتيين (ج١ ص٤٦٢)؛ وع‑ب يسوق الكراهة عن الشعبي وأبي وائل (ج٢ ص١٠٧٦)، والمقدم يأخذ عنهما بنصهما. وليس عندي نص يوفّق بينها، ولا أرجّح؛ ويلاحظ أن الشعبي يعلّل بآية الفرقان في «أرأيت من اتخذ إلهه هواه» (ع‑ب ص١٠٧٦)، وأن المقدم في هامشه يخصّ الذمّ بمن يكثر منها في غير موضعها أو وهو غير أهل للسؤال.

## القسم الثالث: صناعة الجواب وآداب المحاورة

## ١٩ — أضرب السؤال الأربعة وما يقابلها من الجواب
**المسألة:** يقسم الخطيب السؤال في مجلس النظر أربعة أضرب (عن المذهب، عن الدليل، عن وجه الدليل، على سبيل الاعتراض)، ويبين ما يجب على المسؤول وما لا يجب على السائل في كل ضرب.
**المصادر:** خ‑ف ج٢ ص٧٧–٨٢ [٤٦٣؛ لم أجد هذا التقسيم في بقية المصادر المقروءة لهذا الباب]. **موضعها الأوفى:** الباب العاشر، المسألة ٢٢ (أضرب السؤال الأربعة)، وتفصيل الضرب الرابع (الاعتراض) ووجوه الاعتراض على الأدلة في المسألتين ٢٣ و٢٤ منه (خ‑ف ج٢ ص٨٣–١١١).
> **إحالة:** نصوص الخطيب في الأضرب الأربعة وحكاية أبي حنيفة وأبي يوسف (مسألة القصّار) مثبتة في **الباب العاشر، المسألة ٢٢**؛ فلا أكررها هنا، وأثبت أدناه ما لم أجده بلفظه هناك.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** والرابع «السؤال على سبيل الاعتراض عليه، والطعن فيه» (خ‑ف ص٧٧). في الأول: إن كان عند المسؤول تفصيل كان «بالخيار بين أن يفصله في جوابه، وبين أن يقول للسائل: هذا مختلف عندي» (خ‑ف ص٧٧). وفي الثاني: «المسئول لا يلزمه أن يثبت مذهبه إلا بما هو دليل عنده» (خ‑ف ص٨٠)، و«جاز للمسئول أن يفرض المسألة حيث اختاره» (خ‑ف ص٨١)، ويكفيه إذا عارضه السائل بما ليس دليلاً عنده «أن يرده بأن يقول: هذا لا يصح على أصلي» (خ‑ف ص٨١). وفي الثالث: «فإن كان الدليل الذي استدل به غامضا يحتاج إلى بيان وجب السؤال عنه» (خ‑ف ص٨٢)، «وإن كان الدليل ظاهرا جليا لم يجز هذا السؤال، وكان السائل عنه متعنتا أو جاهلا» (خ‑ف ص٨٢).
> **تحرير:** هذا التقسيم في مناظرة الفقهاء، فلا يلزم نقله إلى سؤال المتعلم لشيخه، وقد عرضت في م٤ و٥ و١٠ ما يقابله من آداب السائل المتعلم (الإيجاز، وحسن السؤال، والتفقه لا التعنت). وتعنّت السائل عند الخطيب هنا فقهي المعيار (طلب ما لا يجب على المسؤول)، بخلاف تعليل الماوردي بنية السائل (م١٠).

## ٢٠ — الجواب المطابق: يسدّ موضع السؤال، ولا يجيب عما سُئل عنه غيره، ويفصّل ما احتمل
**المسألة:** ينبغي أن يكون الجواب بقدر السؤال لا ينقص ولا يزيد، وأن يُفصَّل إذا كان السؤال محتملاً لأقسام، وأن لا يتصدّى المرء للجواب عما سئل عنه غيره، وأن يخاطب السائل بما يحتمله.
**المصادر:** خ‑ف ج٢ ص٦٥، ٦٧، ٣٩٨–٣٩٩، ٤٠٧–٤٠٩ [٤٦٣]؛ ب‑ص ص١٣٣–١٣٥، ١٤١ [٦٤٣]؛ ج ص٤٢؛ ز‑ي ص٨؛ ع ص٢٠٣. وانظر الباب الثالث عشر، المسائل ٥٠ و٥١ و٧٢ (الاستفصال، والرفق بالمستفتي بعيد الفهم، وتنويع الجواب بحسب حال السائل)؛ وأحال الباب العاشر (م٢٠) إلى المسائل ٣ و٤ و٥ و٢٠ من هذا الباب.
**الزيادات:**
- **الخطيب:** «ويلزم المجيب أن يسد بالجواب موضع السؤال، ولا يتعدى مكانه» (خ‑ف ص٦٥) «ويقابل باللفظ المعنى، حتى يكون غير ناقص عن تمامه، ولا فاضل عن جملته» (خ‑ف ص٦٥)؛ ويقول بعد ذلك «ومن أدب العلم أن لا يجيب الرجل عما يسأل عنه غيره» (خ‑ف ص٦٧) ويسند إلى أبي عمرو بن العلاء «وليس من الأدب أن تجيب، من لا يسألك، أو تسأل من لا يجيبك، أو تحدث من لا ينصت لك» (خ‑ف ص٦٧)، وإلى ابن المقفع «ليس للعاقل أن يجيب عما يسأل عنه غيره» (خ‑ف ص٦٧). وفي المسألة ذات الأقسام: «ومتى كانت المسألة ذات أقسام لم تفصل في السؤال، لم يجز أن يضع جوابه على بعضها فقط» (خ‑ف ص٣٩٩)، ثم «بل يجب عليه أن يقسم المسألة» (خ‑ف ص٣٩٩)، «ويجب أن يكون جوابه محررا، وكلامه ملخصا» (خ‑ف ص٣٩٩).
- **الخطيب — حال السائل:** «وإذا رأى المفتي من المصلحة عندما تسأله عامة أو سوقة أن يفتي بما له فيه تأول» (خ‑ف ص٤٠٧)، «وإن كان لا يعتقد ذلك، بل لردع السائل، وكفه، فعل» (خ‑ف ص٤٠٧)، ومثاله ابن عباس لمن سأل عن توبة القاتل فقال «لا توبة له» ولآخر «له توبة»، قال: «أما الأول: فرأيت في عينيه إرادة القتل فمنعته» (خ‑ف ص٤٠٧)؛ وابن عمر للشاب «أنهاك عنها» وللشيخ «آمرك بها» في قبلة الصائم (خ‑ف ص٤٠٧)، وتعليل الخطيب «أراد ابن عمر، أن الشاب قوي الشهوة» (خ‑ف ص٤٠٨)، وحديث النبي «إن الشيخ يملك نفسه» (خ‑ف ص٤٠٩).
- **ابن الصلاح:** «الثانية: إذا كانت المسألة فيها تفصيل لم يطلق الجواب، فإنه خطأ» (ب‑ص ص١٣٥)، «ثم له أن يستفصل السائل إن حضر ويقيد السؤال في رقعة الاستفتاء ثم يجيب عنه» (ب‑ص ص١٣٥)؛ ويحكي كراهة التفصيل في الجواب: «وهذا قد كرهه أبو الحسن القابسي من أئمة المالكية» (ب‑ص ص١٣٥) لأن «هذا ذريعة إلى تعليم الناس الفجور» (ب‑ص ص١٣٥)، ويقول ابن الصلاح «ونحن نكرهه أيضا لما ذكره» (ب‑ص ص١٣٥) ثم «وإذا لم يجد المفتي من يستفسره في ذلك كان مدفوعا إلى التفصيل» (ب‑ص ص١٣٥). ويرى الاحتراز من الجواب المعمّى: «ولقد وقع ابن داود بعيدا عن مناهج المفتين في تعقيده» ثم «وتسجيعه وتحبيره من استرشده وتضييعه» (ب‑ص ص١٣٣)، وأن الجواب بـ«فيها خلاف» وحده «فقد عدل عن نهج الفتوى، ولم يفت أيضا بشيء» (ب‑ص ص١٣٣)؛ وللمستفتي بعيد الفهم «فينبغي للمفتي أن يكون رفيقا به صبورا عليه» (ب‑ص ص١٣٥)؛ ويحكي عن الماوردي صاحب الحاوي (كما سمّاه ب‑ص): «إن المفتي عليه أن يختصر جوابه فيكتفي فيه بأن يجوز أو لا يجوز» (ب‑ص ص١٤١) «ولكل مقام مقال» (ب‑ص ص١٤١)، ويعقّب ابن الصلاح: «الاقتصار على لا أو نعم لا يليق بغي العامة» (ب‑ص ص١٤١).
- **ابن جماعة:** المجيب «ويتروى فيما يجيب به» (ج ص٤٢). **الزيدي:** يزيد على ما في ج: «ويكلم كل أحد على قدر عقله وفهمه، فيجيب بما يحتمله حال السائل» (ز‑ي ص٨)، وهذه الجملة ليست في ج ص٤٢ ولا في ع ص١٢٨.
- **العلموي:** ينقل في آداب المفتي عن الصيمري وأبي إسحاق مثالاً على ملاءمة اللفظ للعامي: لا يقول في مسألة الإخوة والأخوات «للذكر مثل حظ الأنثيين» «فإن ذلك قدر يشكل على العامي» (ع ص٢٠٣)، ثم رد أبي إسحاق: «ونحن نجد في تعمد العدول عنه حزازة في النفس» (ع ص٢٠٣) (ينقله ع عن الصيمري وأبي إسحاق ولم أراجع أصليهما).
> **تحرير:** الخطيب يقيّد سدّ موضع السؤال بالمطابقة وبالتحرير، ويضيف ز‑ي مراعاة حال السائل، ويضيف ع (عن الصيمري) مراعاة فهم العامي حتى في لفظ القرآن ويذكر المخالفة في ذلك (أبو إسحاق). ولا ترجيح عندي. وفي ب‑ص (ص١٣٥، ١٤١) ثلاثة مواضع خلاف لم أجد نظيرها عند الخطيب في ما قرأت: كراهة القابسي للتفصيل (وب‑ص يوافقه ثم يستثني حالة تعذر الاستفسار)، واختصار الجواب عند الماوردي صاحب الحاوي مقابل رأي ب‑ص في أن الاقتصار على «لا» أو «نعم» لا يليق بالعامة، وجواب «فيها خلاف» الذي يعدّه ب‑ص غير إفتاء. (وسماع السؤال «من مورده على وجهه» في ٠٧ م٢٦.)

## ٢١ — لا جواب قبل فهم السؤال، والتثبّت وترك العجلة
**المسألة:** على المسؤول أن يفهم السؤال أولاً ثم يتأنى في الجواب؛ ولا يعجل حتى لو أبطأ عليه السائل.
**المصادر:** خ‑ف ج٢ ص٧١–٧٢، ٣٨٩، ٣٩٦–٤٠٠ [٤٦٣]؛ ج ص٤٢؛ ز‑ي ص٨.
**الزيادات:**
- **الخطيب:** يوصي المناظر «ولا يتكلم على ما لم يقع له علمه من كلامه، فإن الجواب لا يصح عما لم يفهمه» (خ‑ف ص٧١)؛ وقول لقمان لابنه «يا بني كن سريعا تفهم، بطيئا تكلم» (خ‑ف ص٧١)؛ ومالك «لا خير في جواب قبل فهم» (خ‑ف ص٧٢). وفي باب ما يفعله المفتي: علي «إذا سأل سائل فليعقل، وإذا سئل المسئول فليتثبت» (خ‑ف ص٣٨٩)، وابن عباس «فلينظر كيف يسأل» (خ‑ف ص٣٨٩). ثم أخبار التأني: أيوب لسائل قال له أأعيد عليك؟ «ولكني أفكر كيف أجيبك» (خ‑ف ص٤٠٠)، والخليل «قد عرفت مسألتك وجوابها، وإنما فكرت في جواب يكون أسرع لفهمك» (خ‑ف ص٤٠٠)، وحديث «البيان من الله، والعجلة من الشيطان» (خ‑ف ص٣٩٧)، و«كاد صاحب الأناة أن يصيب» (خ‑ف ص٣٩٨).
- **الخطيب (تعليقه):** يرى في حالتي علي وابن عمر أن السائل جاء فجأة وأراد الجواب في الحال، وأنهما لو أُخّر الاقتضاء بالجواب حتى ينظرا حق النظر «لأجاباه بالحكم» (خ‑ف ص٣٩٦–٣٩٧).
- **ابن جماعة:** «ويتروى فيما يجيب به» (ج ص٤٢)، وهو في المدرس عند السؤال. **الزيدي:** يكرره (ز‑ي ص٨).
> **تحرير:** الخطيب في تعليقه على علي وابن عمر يربط ما جاء من جوابهما بأن السؤال جاءهما فجأة وأن السائل أراد الجواب في الحال، ويقول إنهما لو أُخّر الاقتضاء لأجاباه بالحكم؛ وقد سبق في سياقه خبر علي مع ابن الكواء في الأختين المملوكتين (خ‑ف ص٣٩٥–٣٩٦) وخبر ابن عمر في نذر صوم الأربعاء (خ‑ف ص٣٩٦). أما «لا أدري» فهي ترك للجواب عند انتفاء العلم، وهي موضوع القسم الرابع؛ وليس في ما قرأته من هذا الموضع ربط صريح بين الأمرين. وانظر الباب العاشر، المسألة ١٩ (ألا يعجل بالجواب قبل الفهم)، والباب الثالث عشر، المسألتين ٥٢ و٦٩ (تأمل الرقعة وفهم السؤال قبل الجواب).

## ٢٢ — المحاور مقبل على خصمه، لا يقطع عليه ولا يغضب
**المسألة:** في مجلس النظر يُقبل كل من الخصمين على صاحبه ويستمع له حتى ينهي كلامه، ولا يحتد عليه إن أفحش.
**المصادر:** خ‑ف ج٢ ص٦٢–٧١ [٤٦٣]؛ ج ص١٠٧؛ ع‑ب ج١ ص٥٢١. **موضعها الأوفى:** الباب العاشر، المسألتان ١٧ (الحلم وترك الغضب) و١٩ (حسن الاستماع، وألا يقاطع، وألا يعجل بالجواب قبل الفهم).
> **إحالة:** نصوص الخطيب في إقبال الخصمين وعدم التقطيع والغضب (ابن المقفع وشريح وابن المعتز) مثبتة في **الباب العاشر، المسألتين ١٧ و١٩**؛ فلا أكررها هنا، وأثبت أدناه ما لم أجده بلفظه هناك.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** لأن الاستماع «طريق معرفته، والوقوف على حقيقته» (خ‑ف ص٦٢)؛ ومن الحكماء «إن من الأخلاق السيئة على كل حال مغالبة الرجل على كلامه والإعتراض فيه لقطع حديثه» (خ‑ف ص٦٨).
- **ابن عبد البر:** الحسين بن علي لابنه «ولا تقطع على أحد حديثا وإن طال حتى يمسك» (ع‑ب ص٥٢١). **ابن جماعة** (في الطالب مع الشيخ): «وينبغي أن لا يقطع على الشيخ كلامه؛ أي كلام كان» (ج ص١٠٧).
> **تحرير:** أدب الاستماع هنا مشترك بين الخصمين وبين الطالب والشيخ (م٦)، والخطيب يعلّله بأن في كلام الخصم «ما يدله على فساده» (خ‑ف ص٦٢) أي بمنفعة الجواب، وابن جماعة يعلله بالتأدب مع الشيخ؛ ولا تعارض.

## ٥٠ — حمقى السائلين وجواب العالم على السؤال السفيه
**المسألة:** من السائلين من يسأل بجهله عما يخطر له، أو عما لا نفع فيه، أو للهزل أو للإحراج؛ وللعالم في جوابه سبل متفاوتة: الإعراض، أو التعريض، أو الجواب بما يناسب السؤال.
**المصادر:** خ‑ف ج٢ ص٤١٩ [٤٦٣]؛ مق ص٢٧٢–٢٧٤ وهامش ص٢٧٤ (معاصر، ينقل عن السلف بلا أسانيد ولم أراجع أصوله). وانظر الماوردي (ماو ص٢٧٦–٢٧٧) في الباب الثامن، المسألة ٢١، وفي الباب الثالث (٠٥)، المسألة ١٢.
**الزيادات:**
- **الخطيب (الفقيه):** بعد حديث «لا تزالون تسألون، حتى يقال لكم: هذا الله خلقنا» (خ‑ف ص٤١٩) يقول: «وحمقى الناس يسألون بجهلهم عن جميع ما يعرض في قلوبهم» (خ‑ف ص٤١٩)، ويسوق: الشعبي لمن سأله بعد رعاف أيحتجم أم يفتصد: «الحمد لله الذي نقلنا من الفقه إلى الحجامة» (خ‑ف ص٤١٩)؛ وشريك لمن سأل أيما أطيب الطنبور أم العود: «أحسبك بايعت يا عدو الله» (خ‑ف ص٤١٩) ثم «كم على الطنبور من زبر؟» (خ‑ف ص٤١٩)؛ وحسين النجار لرجل متشادق: «أفكر، فإنا قد استرحنا» (خ‑ف ص٤١٩).
- **المقدم** (ص٢٧٢–٢٧٤، ينقل عن السلف): ابن سيرين إذا سئل عن مسألة فيها أغلوطة قال للسائل: «أمسكها حتى تسأل عنها أخاك إبليس» (مق ص٢٧٣)؛ والشعبي لمن قال له «إني خبأت لك مسائل»: «اخبأها لإبليس حتى تلقاه، فتسأله عنها» (مق ص٢٧٣)؛ وابن عمر لسائل عراقي عن دم البعوض: «ها انظروا إلى هذا! يسأل عن دم البعوض» (مق ص٢٧٤)، وفي هامشه رواية «ما أسالهم عن صغيرة، وأجرأهم على كبيرة» (مق هامش ص٢٧٤)؛ وعكرمة لمن قال «فلان قذفني في النوم»: «اضرب ظله ثمانين» (مق ص٢٧٤)؛ والشعبي لمن سأله ما اسم امرأة إبليس: «ذاك عرس ما شهدته» (مق ص٢٧٤).
- وفي نهي الأصحاب عن الضحك من سائل جاهل (خ‑ف ج٢ ص٢٨٧): الباب الثامن، المسألة ٢١؛ وجواب أبي حنيفة للسائل الهازل (خ‑ف ج٢ ص٦٠): الباب العاشر، المسألة ٢٢.
> **تحرير:** الخطيب يعلّق على حمقى السائلين بلا أمر بجواب معين؛ والمقدم يجمع أجوبة السلف عن الأسئلة السفيهة، وهي تتفاوت بين الإعراض والتعريض (ابن سيرين، والشعبي) والتوبيخ (ابن عمر) والسخرية (شريك)؛ وعند الماوردي جواب بما يناسب السؤال بلا زجر (الباب الثامن م٢١)، ولا ترجيح عندي. والفرق بين الأحمق والمتعنت: المتعنت قصده الإحراج (م١٠)، والأحمق يسأل بجهله. وتُنظر م١٢ في «سؤال الأحمق».

## القسم الرابع: قول «لا أدري»

## ٢٣ — «لا أدري» من العلم: القاعدة وأقدم من صرّح بها
**المسألة:** من سُئل عما لا يعلم قال «لا أعلم» أو «الله أعلم»، وليس في ذلك نقص بل هو من العلم، ويُذمّ من أبى أن يقولها. (مضت موجزة في ٠٧ م٢٨؛ وتبسط هنا بمصادر لم تُقرأ هناك.)
**المصادر:** ماو ص٧٣–٧٤ [٤٥٠: أقدم المقروء]؛ ع‑ب ج١ ص٧٥٣ وج٢ ص٨٢٦–٨٤٣ وخ‑ف ج٢ ص٣٦٠–٣٧٢ [٤٦٣]؛ غ ص٢٧، ٦٩–٧٠ [٥٠٥]؛ ب‑ص ص٧٤–٨٠ [٦٤٣]؛ ج ص٤٢–٤٣ [٧٣٣]؛ ع ص١٢٩–١٣٠، ١٧٤–١٧٧ [٩٨١]؛ ز‑ي ص٨؛ ش (OCR، بلا صفحة)؛ النراقي (جامع السعادات، OCR، بلا صفحة)؛ ح ص١٨٤ (معاصر).
**الزيادات:**
- **الماوردي:** «وإذا لم يكن في جهل بعضه عار لم يقبح به أن يقول لا أعلم فيما ليس يعلم» (ماو ص٧٤)؛ وعلي «وإن العالم من عرف أن ما يعلم فيما لا يعلم قليل» (ماو ص٧٤)؛ «وقال بعض العلماء: هلك من ترك لا أدري» (ماو ص٧٤)؛ و«من قال لا أدري علم فدرى، ومن انتحل مما لا يدري أهمل فهوى» (ماو ص٧٤)؛ ويحكي عن نفسه أنه ألّف كتاباً في البيوع ثم سأله أعرابيان عن بيع فلم يعرف جواباً، فقالا «ما عندك فيما سألناك جواب، وأنت زعيم هذه الجماعة؟» فقال «لا» (ماو ص٧٣).
- **ابن عبد البر:** ابن مسعود «فإن من علم المرء أن يقول لما لا يعلم: الله أعلم» (ع‑ب ج٢ ص٨٣١)؛ وأبو داود «قول الرجل فيما لا يعلم: لا أعلم نصف العلم» (ع‑ب ج٢ ص٨٤٢)؛ وأبو الذيال «تعلم لا أدري؛ فإنك إن قلت: لا أدري علموك حتى تدري» (ع‑ب ج٢ ص٨٤٢)؛ وأبيات الراجز «وقل إذا أعياك ذاك الأمر» (ع‑ب ج٢ ص٨٤٢)؛ وابن عمر «العلم ثلاثة أشياء: كتاب ناطق وسنة ماضية ولا أدري» (ع‑ب ج١ ص٧٥٣).
- **الخطيب (الفقيه):** علي «يا بردها على الكبد إذا سئل الرجل عما لا يعلم أن يقول: الله أعلم» (خ‑ف ص٣٦٢)؛ وابن مسعود «فإن من علم المرء أن يقول لما لا يعلم: الله أعلم» (خ‑ف ص٣٦٣)؛ وابن عمر «نعم ما قال ابن عمر سئل عما لا يعلم، فقال: لا أعلم» (خ‑ف ص٣٦٤)؛ وعن مالك عن نافع عن ابن عمر «العلم ثلاثة: كتاب ناطق، وسنة ماضية، ولا أدري» (خ‑ف ص٣٦٦)؛ والشعبي «لا أدري: نصف العلم» (خ‑ف ص٣٦٩)؛ وابن المقفع «من أنف من قول لا أدري تكلف الكذب» (خ‑ف ص٣٧١)؛ وقول بعض الأوائل «لقد حسنت عندي لا أدري، حتى أردت قولها فيما أدري» (خ‑ف ص٣٧١).
- **الغزالي:** يجعلها علامة إخلاص العالم: «ومن يرد غير وجه الله تعالى بعلمه فلا تسمح نفسه بأن يقر على نفسه بأنه لا يدري» (غ ص٢٧)؛ وفي علامات علماء الآخرة يضع للمسؤول ثلاث مراتب: «فإن سئل عما يعلمه تحقيقا بنص كتاب الله أو بنص حديث أو إجماع أو قياس جلي أفتى»، «وإن سئل عما يشك فيه قال لا أدري»، «وإن سئل عما يظنه باجتهاد وتخمين احتاط ودفع عن نفسه وأحال على غيره إن كان في غيره غنية» (غ ص٦٩)؛ ويروي «العلم ثلاثة كتاب ناطق وسنة قائمة ولا أدري» ثم «قال الشعبي لا أدري نصف العلم» (غ ص٦٩)؛ ويقول «ومن سكت حيث لا يدري لله تعالى فليس بأقل أجرا ممن نطق لأن الاعتراف بالجهل أشد على النفس» (غ ص٦٩).
- **ابن جماعة:** «فمن العلم أن يقول لا أعلم» (ج ص٤٢)، وعن بعضهم «لا أدري نصف العلم» (ج ص٤٢)، «وقيل: ينبغي للعالم أن يورث أصحابه لا أدري لكثرة ما يقولها» (ج ص٤٢)، وعن الشافعي في المتعة «والله ما ندري» (ج ص٤٢).
- **العلموي:** «ولا يستنكف عن ذلك» (ع ص١٢٩)، وعلي «إذا سئلتم عما لا تعلمون فاهربوا» (ع ص١٢٩)، وابن عمر «العلم ثلاثة: كتاب ناطق، وسنة ماضية، ولا أدري» (ع ص١٣٠)، وبيت ابن دريد «ومن كان يهوى أن يرى متصدرا ... ويكره لا أدري أصيبت مقاتله» (ع ص١٣٠). **الزيدي:** يكرر ج (ز‑ي ص٨). **أبو زيد:** «جنة العالم (لا أدرى)، ويهتك حجابه الاستنكاف منها» (ح ص١٨٤).
- **النراقي** (جامع السعادات، آداب التعليم، OCR، بلا صفحة؛ ت ١٢٠٩): من آداب المعلم أن يقول ما يعلم ويسكت عما لا يعلم، وهذا عنده يعمّ المفتي والقاضي؛ ويورد أحاديث عن الباقر والصادق في الوقوف عند ما لا يعلم، وفي أن يقول المسؤول لا أدري ولا يقول الله أعلم لئلا يوقع الشك في قلب السائل، وفي لعن الملائكة من أفتى بغير علم.
- **الشهيد الثاني** (ش، OCR، بلا صفحة): يعدّ من آداب المعلم في درسه الاعتراف بلا أدري، مع بيان أن ذلك لا يضع منزلة العالم.
> **تحرير:** (أ) عبارة علي: «ما أبردها على الكبد» عند ع‑ب (ص٨٣٦) وخ‑ف (ص٣٦١–٣٦٢)، وعند ماو «وما أبردها على القلب» (ص٧٤)؛ ولفظه عند ع «إذا سئلتم عما لا تعلمون فاهربوا» (ع ص١٢٩) صيغة أخرى لم أجدها عند ع‑ب ولا خ‑ف. (ب) «نصف العلم»: لأبي داود عند ع‑ب (ص٨٤٢)، وللشعبي عند خ‑ف (ص٣٦٩)، وبلا اسم «عن بعضهم» عند ج (ص٤٢). (ب‑١) «العلم ثلاثة»: لابن عمر عند خ‑ف (ص٣٦٦) وع (ص١٣٠)، وعند غ بلا نسبة وبلفظ «سنة قائمة» (ص٦٩) بدل «سنة ماضية» عند خ‑ف وع، وهو عند ع‑ب (ج١ ص٧٥٣) لابن عمر بلفظ «سنة ماضية»، برواية طاوس عنه وبرواية مالك عن نافع عنه. (ب‑٢) «لا أدري نصف العلم» للشعبي عند غ (ص٦٩) وخ‑ف (ص٣٦٩). (ج) أقدمية الماوردي بالوفاة (٤٥٠) في ما قرأت من هذه الصياغة العامة، لكن الأخبار نفسها عن الصحابة والتابعين عند ع‑ب وخ‑ف أوفر؛ وليس عندي ما يعين من أخذ عمن. (د) النراقي (OCR) يجعل المسؤول يقول لا أدري ولا يقول الله أعلم، وعند علي وابن مسعود في ع‑ب (ص٨٣١، ٨٣٦) وخ‑ف (ص٣٦٢–٣٦٣) أن يقول الله أعلم؛ ولا أرجّح. (هـ) أقدم من ماو في أن المفتي يتوقف عما لا يعلم ويحيل المستفتي على غيره: المرتضى (٤٣٦، OCR) في م٣١؛ وفي متى يسع لا أدري: م٥١.

## ٢٤ — «لا أدري» عن النبي وجبريل: أي البقاع خير؟
**المسألة:** رُوي أن النبي سُئل عن خير البقاع وشرها فقال «لا أدري»، وأن جبريل سُئل فقال مثله حتى سأل ربه؛ وفيه التأسي بالنبي في ترك الجواب قبل العلم.
**المصادر:** ماو ص٧٤ [٤٥٠]؛ ع‑ب ج٢ ص٨٢٦–٨٢٨ وخ‑ف ج٢ ص٣٦١ [٤٦٣]؛ غ ص٧٠ [٥٠٥]؛ ج ص٤٣ وع ص١٣٠ وز‑ي ص٨.
**الزيادات:**
- **الماوردي:** «أن رجلا قال: يا رسول الله أي البقاع خير، وأي البقاع شر؟ فقال: لا أدري حتى أسأل جبريل» (ماو ص٧٤)، وليس فيه قول جبريل «لا أدري».
- **ابن عبد البر:** حديث ابن عمر: سأل رجل «أي البقاع خير؟» فقال «لا أدري» ثم أتاه جبريل فقال مثل ذلك، وانتفض جبريل «انتفاضة كاد يصعق منها محمد» وجاء الجواب «أن خير البقاع المساجد، وأن شر البقاع الأسواق» (ع‑ب ص٨٢٦)؛ وحديث أبي هريرة «أحب البلاد إلى الله مساجدها وأبغض البلاد إلى الله أسواقها» (ع‑ب ص٨٢٧)؛ وحديث «ما أدري أعزير نبي أم لا؟ وما أدري أتبع ملعون أم لا؟» (ع‑ب ص٨٢٨)، وفي رواية «وما أدري ذو القرنين نبي أم لا؟ وما أدري الحدود كفارات لأهلها أم لا؟» (ع‑ب ص٨٢٨)؛ وتعقيب ابن عبد البر أن حديث عبادة في أن الحدود كفارة «أثبت وأصح إسنادا من حديث أبي هريرة هذا» (ع‑ب ص٨٢٨).
- **الخطيب (الفقيه):** رواية بلفظ «أي البلدان شر؟» ثم «لا أدري حتى أسأل ربي» عن جبريل (خ‑ف ص٣٦١)، والجواب «أسواقها» (خ‑ف ص٣٦١).
- **الغزالي:** «وقال صلى الله عليه وسلم ما أدري أعزير نبي أم لا وما أدري أتبع ملعون أم لا وما أدري ذو القرنين نبي أم لا» (غ ص٧٠)، ثم «ولما سئل رسول الله صلى الله عليه وسلم عن خير البقاع في الأرض وشرها قال لا أدري» ثم «حتى نزل عليه جبريل عليه السلام فسأله فقال لا أدري» (غ ص٧٠).
- **ابن جماعة:** يعدّ ذلك من تأديب الله للعلماء بقصة موسى والخضر: «وقد أدب الله تعالى العلماء بقصة موسى مع الخضر» (ج ص٤٣) إذ لم يرد موسى العلم إلى الله لما سُئل هل أحد أعلم منه؛ ويكرره ع (ص١٣٠) وز‑ي (ص٨).
> **تحرير:** (أ) لفظ السؤال: «أي البقاع» عند ماو وع‑ب، و«أي البلدان» عند خ‑ف. (ب) قول جبريل «لا أدري» ليس عند ماو، وهو عند ع‑ب وخ‑ف. (ج) حديث «ما أدري أعزير نبي» أورده ع‑ب في هذا الباب شاهداً، وتعقيبه إنما هو على «الحدود كفارات»؛ وأورده الغزالي أيضاً (غ ص٧٠) بلفظ «ما أدري أعزير نبي أم لا وما أدري أتبع ملعون أم لا وما أدري ذو القرنين نبي أم لا» بلا ذكر «الحدود». (د) ابن جماعة وتابعوه يستشهدون بموسى والخضر، ولا يذكرون خبر البقاع.

## ٢٥ — الصحابة والتابعون في «لا أدري» وفي القول بغير علم
**المسألة:** أخبار عن أبي بكر وعلي وابن مسعود وابن عمر والشعبي في الامتناع عن الجواب فيما لا يعلمون، والتحرج من القول في كتاب الله بغير علم.
**المصادر:** ع‑ب ج٢ ص٨٣٠–٨٣٦ وخ‑ف ج٢ ص٣٦٢–٣٧٠ [٤٦٣]؛ وع‑ب ج١ ص٧٧٨ وج٢ ص١٠٤٢، ١٠٧٧؛ وخ‑ف ج١ ص٤٥٨–٤٦١ (من القراءة الجديدة).
**الزيادات:**
- **ابن عبد البر:** ابن سيرين «لم يكن أحد بعد النبي صلى الله عليه وسلم أهيب لما لا يعلم من أبي بكر رضي الله عنه» (ع‑ب ص٨٣٠)؛ وأبو بكر «أي سماء تظلني؟ وأي أرض تقلني؟ إذا قلت في كتاب الله بغير علم» (ع‑ب ص٨٣٤) وعلي بمعناه (ع‑ب ص٨٣٤)؛ وابن مسعود يستشهد بآية «وما أنا من المتكلفين» (ع‑ب ص٨٣١–٨٣٣)؛ والشعبي لما قال لمسألة «هي زباء هلباء ذات وبر ولا أحسنها» (ع‑ب ص٨٣٢) فقال له أصحابه استحيينا منك، فقال «لكن الملائكة المقربين لم تستحي حين قالت» (ع‑ب ص٨٣٢)؛ وعلي «ما أبردها على الكبد، ما أبردها على الكبد» ثم «أن تقول للشيء لا تعلمه: الله أعلم» (ع‑ب ص٨٣٦)؛ وابن عمر «نعما قال عبد الله بن عمر، سئل عما لا يعلم فقال: لا علم لي به» (ع‑ب ص٨٣٥).
- **الخطيب (الفقيه):** ابن مسعود لسائل عن أمر الأمراء «ما أدري ما أقول لك» (خ‑ف ص٣٦٣)؛ وابن عمر بعد قوله «لا أدري»: «أتريدون أن تجعلوا ظهورنا لكم جسورا في جهنم» (خ‑ف ص٣٦٥)، وفي رواية مروان الأصفر «ونعم ما قال ابن عمر لما لا يدري: لا أدري» (خ‑ف ص٣٦٥)؛ والشعبي لمن قال له أما تستحي من «لا أدري» وأنت فقيه أهل العراقين: «لكن الملائكة لم تستحي حين قالت» (خ‑ف ص٣٧٠).
- **ابن عبد البر (من القراءة الجديدة):** الشعبي «والثالثة: إذا سئلت عما لا تعلم فقل: لا أعلم وأنا شريكك» (ع‑ب ج٢ ص١٠٧٧)؛ وبعض الصحابة وقد سُئل: «إني لأستحي من ربي أن أقول في أمة محمد صلى الله عليه وسلم برأيي» (ع‑ب ج١ ص٧٧٨)؛ وعمر في أصحاب الرأي «واستحيوا حين سئلوا أن يقولوا: لا نعلم» (ع‑ب ج٢ ص١٠٤٢) (وانظر م٢٩).
- **الخطيب (الفقيه، في باب ذم القياس):** مسروق وقد سُئل عن مسألة فقال «لا أدري» (خ‑ف ج١ ص٤٥٨) فلما قالوا له «قس لنا برأيك» (خ‑ف ج١ ص٤٥٨) أجاب «أخاف أن تزل قدمي» (خ‑ف ج١ ص٤٥٩)؛ والشعبي وقد سُئل عن مسألة فقال «لا أدري ولكن احفظ ثلاثا» (خ‑ف ج١ ص٤٦١) أولها «لا تقل لما لا تعلم إنك تعلم» (خ‑ف ج١ ص٤٦١).
> **تحرير:** (أ) ابن عمر في الجسر: «يريدون أن يجعلوا ظهورنا جسرا لهم إلى جهنم» عند ع‑ب (ص٨٤١) و«أتريدون أن تجعلوا ظهورنا لكم جسورا في جهنم» عند خ‑ف (ص٣٦٥). (ب) خبر الشعبي والملائكة عند ع‑ب (ص٨٣٢) جواب لأصحابه الذين قالوا «قد استحيينا منك»، وعند خ‑ف (ص٣٧٠) جواب لمن قال له «أما تستحي من قولك لا أدري» — القصة واحدة في المعنى والسائل مختلف في الصياغة. (ج) ابن عمر «نعم ما قال ابن عمر»: عند ع‑ب بصيغة «نعما قال عبد الله بن عمر» (ص٨٣٥)، وعند خ‑ف بصيغتين «نعم ما قال ابن عمر» (ص٣٦٤) «ونعم ما قال ابن عمر لما لا يدري» (ص٣٦٥).

## ٢٦ — القاسم بن محمد ومالك: «لا أحسنها» ولو رحل السائل
**المسألة:** أخبار في أن العالم يقول «لا أحسن» ولو جاء السائل من بعيد وقال له إنه لا يعرف غيره؛ وعدد ما قال فيه مالك «لا أدري».
**المصادر:** ع‑ب ج٢ ص٨٣٦–٨٤٠ وخ‑ف ج٢ ص٣٦٧–٣٧١ [٤٦٣]؛ غ ص٢٧، ٧٠ وغ ج٢ ص١٢٢ [٥٠٥]؛ ب‑ص ص٧٨–٨٠ [٦٤٣]؛ ع ص١٧٦–١٧٧ [٩٨١].
**الزيادات:**
- **ابن عبد البر:** القاسم: «أنا والله ما نعلم كل ما تسألونا عنه، ولو علمنا ما كتمناكم ولا حل لنا أن نكتمكم» (ع‑ب ص٨٣٦)؛ ولسائل قال إني لا أعرف غيرك: «لا تنظر إلى طول لحيتي وكثرة الناس حولي والله ما أحسنه» (ع‑ب ص٨٣٧)، ثم «والله لأن يقطع لساني أحب إلي من أن أتكلم بما لا علم لي به» (ع‑ب ص٨٣٧)؛ وسعيد بن جبير «ويل للذي يقول لما لا يعلم: إني أعلم» (ع‑ب ص٨٣٦)؛ وأيوب لسائل: «بلى» فهمت ولكن «لا أعلمه» (ع‑ب ص٨٣٨)؛ ومالك لرجل جاءه من مسيرة ستة أشهر: «تقول لهم: قال مالك: لا أحسن» (ع‑ب ص٨٣٨)، و«ينبغي للعالم أن يألف فيما أشكل عليه قول: لا أدري» (ع‑ب ص٨٣٩)، وابن وهب «لو كتبنا عن مالك: لا أدري لملأنا الألواح» (ع‑ب ص٨٣٩)، ومالك «كان رسول الله صلى الله عليه وسلم إمام المسلمين وسيد العالمين يسأل عن الشيء فلا يجيب حتى يأتيه الوحي» (ع‑ب ص٨٣٩).
- **الخطيب (الفقيه):** القاسم «لأن يعيش الرجل جاهلا خير له من أن يفتي بما لا يعلم» (خ‑ف ص٣٦٧)، وقوله لأمير سأله «إن من إكرام المرء نفسه أن لا يقول إلا ما أحاط به علمه» (خ‑ف ص٣٦٨)؛ وأيوب لمن قال له دلني على من يدري: ردّ «دلني على من يدري» ثم «ولا أدري من يدري» (خ‑ف ص٣٦٩–٣٧٠)؛ ومالك لرجل من أهل المغرب قال له تقول لا أدري: «نعم، فبلغ من وراءك أني لا أدري» (خ‑ف ص٣٧٠)؛ وأحمد «يستفتى فيكثر أن يقول: لا أدري» ثم تفسيره عند الخطيب: «وذلك فيما قد عرف الأقاويل فيه» «ومعنى قوله: لا أدري أي: لا أدري ما أختار من ذلك» (خ‑ف ص٣٧١).
- **الغزالي:** يروي عن الشافعي «إني شهدت مالكا وقد سئل عن ثمان وأربعين مسئلة فقال في اثنتين وثلاثين منها لا أدري» (غ ص٢٧)؛ ويقول في موضع آخر: «وكان في الفقهاء من يقول لا أدري أكثر مما يقول أدري» ويعدّ منهم «سفيان الثوري ومالك بن أنس وأحمد بن حنبل والفضيل ابن عياض وبشر بن الحرث» (غ ص٧٠)؛ وأن ابن عمر «يسئل عن عشر مسائل فيجيب عن واحدة ويسكت عن تسع» وابن عباس «يجيب عن تسع ويسكت عن واحدة» (غ ص٧٠)؛ ويطبّق على نفسه في مسألة اشتباه الذكية بالميتة «فهذا مما لا أدري ما أقول فيه» (غ ج٢ ص١٢٢)، ويحكي أن أحمد سئل عمن «رمى صيدا فوقع في ملك غيره أيكون الصيد للرامي أو لمالك الأرض» (غ ج٢ ص١٢٢) «فقال لا أدري فروجع فيه مرات فقال لا أدري» (غ ج٢ ص١٢٢)، ثم «فليقطع المفتي طمعه عن درك الحكم في جميع الصور» (غ ج٢ ص١٢٢) (من الجزء الثاني في ملف الإحياء، وسياقه الأموال المشتبهة).
- **ابن الصلاح:** ابن مهدي: جاء رجل مالكاً أياماً فقال «إني أريد الخروج» فقال مالك «إني إنما أتكلم فيما أحتسب فيه الخير، ولست أحسن مسألتك هذه» (ب‑ص ص٧٩)؛ والشافعي سئل فسكت «حتى أدري الفضل في سكوتي، أو في الجواب» (ب‑ص ص٧٩)؛ وأحمد «يستفتى فيكثر أن يقول: لا أدري» (ب‑ص ص٧٩)؛ وعن الهيثم بن جميل: «شهدت مالك بن أنس سئل عن ثمان وأربعين مسألة فقال في اثنتين وثلاثين منها: لا أدري» (ب‑ص ص٧٩)؛ ومالك «من أجاب في مسألة فينبغي من قبل أن يجيب فيها أن يعرض نفسه على الجنة أو النار» (ب‑ص ص٨٠)، وقوله لمن قال إنها خفيفة «ليس في العلم شيء خفيف» (ب‑ص ص٨٠).
- **العلموي:** يورد عدد مالك (اثنتان وثلاثون من ثمان وأربعين) بلا راوٍ (ع ص١٧٦)، ويزيد: «وسئل الإمام أبو حنيفة -رضي الله عنه- عن تسع مسائل فقال فيها: لا أدري» ثم يعددها (ع ص١٧٦–١٧٧).
> **تحرير:** (أ) خبر الثمان والأربعين: الشافعي عن مالك (غ ص٢٧)، والهيثم بن جميل عن مالك (ب‑ص ص٧٩)، وبلا راوٍ (ع ص١٧٦)؛ والعدد واحد. (ب) قول مالك «ليس في العلم شيء خفيف» (ب‑ص) و«ليس من العلم خفيف» (ع). (ج) الرجل القادم من بعيد: عند ع‑ب (عن ابن مهدي) «جئتك من مسيرة ستة أشهر» فقال مالك «لا أحسنها» (ص٨٣٨)، وعند خ‑ف (عن ابن مهدي) رجل من أهل المغرب قال له «تقول لا أدري؟» فقال «نعم» (ص٣٧٠)، وعند ب‑ص (عن ابن مهدي) تردد أياماً ثم أراد الخروج فقال «ولست أحسن مسألتك هذه» (ص٧٩)؛ والراوي في الثلاثة ابن مهدي والألفاظ مختلفة، ولا أجزم بأنها قصة واحدة. (ج‑١) تفسير الخطيب لقول أحمد بأنه عن اختياره بين أقوال معروفة (ص٣٧١) نقل ب‑ص عبارته الأولى ولم أجد عنده التفسير (ب‑ص ص٧٩). (د) ما زاده ع من مسائل أبي حنيفة لم أجده في ب‑ص ولا غ.

## ٢٧ — «إذا أخطأ العالم لا أدري أصيبت مقاتله»: الصيغ والنسبة
**المسألة:** عبارة شائعة منسوبة إلى ابن عباس أو ابن عجلان في أن العالم إذا ترك «لا أدري» هلك علمه؛ وصيغها متعددة.
**المصادر:** ماو ص٧٤ [٤٥٠]؛ ع‑ب ج٢ ص٨٣٩–٨٤١ وخ‑ف ج٢ ص٣٦٦–٣٦٧ [٤٦٣]؛ غ ص٦٩ [٥٠٥]؛ ب‑ص ص٧٦–٧٧ [٦٤٣]؛ ج ص٤٢؛ ع ص١٢٩؛ ز‑ي ص٨؛ ح ص١٨٤.
**الزيادات:**
- **الماوردي:** ابن عباس «إذا ترك العالم قول لا أدري أصيبت مقاتله» (ماو ص٧٤).
- **ابن عبد البر:** عن مالك: «كان ابن عباس يقول: إذا أخطأ العالم لا أدري أصيبت مقاتله» (ع‑ب ص٨٣٩)؛ وعن يحيى بن سعيد عن ابن عباس «إذا ترك العالم لا أعلم فقد أصيبت مقاتله» (ع‑ب ص٨٤٠)؛ وعن ابن عجلان «إذا أخطأ العالم لا أدري أصيبت مقاتله» (ع‑ب ص٨٤٠)، وفي رواية الشافعي عن مالك عن ابن عجلان «إذا أغفل العالم لا أدري أصيبت مقاتله» (ع‑ب ص٨٤١)، ثم «كان يقال: إذا لم يألف العالم لا أدري» (ع‑ب ص٨٤١).
- **الخطيب (الفقيه):** عن مالك عن ابن عباس «إذا أخطأ العالم أن يقول لا أدري، فقد أصيبت مقاتله» (خ‑ف ص٣٦٦)، وبإسناد أحمد عن الشافعي عن مالك عن ابن عجلان «إذا أخطأ العالم لا أدري أصيبت مقاتله» (خ‑ف ص٣٦٧).
- **الغزالي:** عقب قول ابن مسعود «إن الذي يفتي الناس في كل ما يستفتونه لمجنون»: «وقال جنة العالم لا أدري فإن أخطأها فقد أصيبت مقاتله» (غ ص٦٩) بلا تسمية القائل.
- **ابن الصلاح:** يسوق إسناد أحمد عن الشافعي عن مالك عن ابن عجلان بلفظ «إذا أغفل العالم لا أدري أصيبت مقاتله» ويقول «هذا إسناد جليل عزيز جدا لاجتماع أئمة المذاهب الثلاثة فيه بعضهم عن بعض» (ب‑ص ص٧٧)، ويضيف «وروى مالك مثل ذلك عن ابن عباس» (ب‑ص ص٧٧).
- **ابن جماعة:** «وعن ابن عباس رضي الله عنهما: إذا أخطأ العالم لا أدري أصيبت مقالته» (ج ص٤٢). **العلموي:** «وقال ابن عباس: إذا ترك العالم لا أدري أصيبت مقاتله» (ع ص١٢٩). **الزيدي:** «وعن ابن عباس: إذا أخطأ العالم لا أدري أصيبت مقاتله» (ز‑ي ص٨).
> **تحرير:** الصيغ المقروءة عند غير الغزالي أربع: «ترك» (ماو، وع‑ب ص٨٤٠ في رواية يحيى بن سعيد، وع)، و«أخطأ» (ع‑ب، وخ‑ف بصيغتين: «أخطأ العالم لا أدري» و«أخطأ العالم أن يقول لا أدري»، وج بلفظ «مقالته»، وز‑ي)، و«أغفل» (ع‑ب ص٨٤١ وب‑ص)، و«لم يألف» (ع‑ب ص٨٤١). والنسبة إلى ابن عباس (ماو، وع‑ب عن مالك وعن يحيى بن سعيد، وخ‑ف عن مالك، وج وع وز‑ي) أو إلى ابن عجلان (ع‑ب، وخ‑ف، وب‑ص). ولا أرجّح؛ وقد نبهت في ٠٧ م٢٨ على «مقالته» في ج. وعند غ صيغة خامسة «جنة العالم لا أدري فإن أخطأها فقد أصيبت مقاتله» بلا نسبة (غ ص٦٩)، وأبو زيد يوافقها في «جنة العالم (لا أدرى)» (ح ص١٨٤).

## ٢٨ — توريث «لا أدري» للجلساء
**المسألة:** يوصى العالم بأن يكثر من «لا أدري» حتى يصير أصلاً يرجع إليه من بعده إذا سئل عما لا يعلم.
**المصادر:** ع‑ب ج٢ ص٨٣٥ وخ‑ف ج٢ ص٣٦٧ [٤٦٣]؛ ج ص٤٢ [٧٣٣]؛ ز‑ي ص٨.
**الزيادات:**
- **ابن عبد البر:** عن عبد الله بن يزيد بن هرمز: «إني لأحب أن يكون من بقايا العالم بعده: لا أدري؛ ليأخذ به من بعده» (ع‑ب ص٨٣٥).
- **الخطيب (الفقيه):** عن ابن هرمز: «ينبغي للعالم أن يورث، جلساءه من بعده لا أدري، حتى يكون ذلك أصلا في أيديهم يفزعون إليه» (خ‑ف ص٣٦٧).
- **ابن جماعة:** «وقيل: ينبغي للعالم أن يورث أصحابه لا أدري لكثرة ما يقولها» (ج ص٤٢)، بلا نسبة إلى ابن هرمز. **الزيدي:** يكرره بلا نسبة (ز‑ي ص٨).
> **تحرير:** الأثر في المقروء لابن هرمز (ع‑ب وخ‑ف، وكلاهما عن مالك)، وابن جماعة يسقط الاسم ويحكيه «قيل»؛ وزاد ج التعليل «لكثرة ما يقولها»، وزاد الخطيب «يفزعون إليه»، ولم أجد التعليلين عند غيرهما.


## ٢٩ — علة الأنفة من «لا أدري» وما تُنتجه
**المسألة:** من أبى أن يقول «لا أدري» فعلته ضعف الدين أو العجب أو إرادة غير وجه الله، وعاقبته أن يقع فيما فرّ منه.
**المصادر:** ماو ص٧٣–٧٦ [٤٥٠]؛ ع‑ب ج٢ ص٨٢٠، ٨٨٠، ١٠٤٢ [٤٦٣]؛ غ ص٢٧ [٥٠٥]؛ خ‑ف ج٢ ص٣٧١ [٤٦٣]؛ ج ص٤٢–٤٣ [٧٣٣]؛ ع ص١٣٠؛ ز‑ي ص٨؛ ح ص١٨٤.
**الزيادات:**
- **الماوردي:** «وحق على من ترك العجب بما يحسن أن يدع التكلف لما لا يحسن» (ماو ص٧٣)؛ وقسمة الخليل: «رجل يدري ويدري أنه يدري فذلك عالم فاسألوه» (ماو ص٧٥)، و«فذلك مسترشد» لمن لا يدري ويدري أنه لا يدري (ماو ص٧٥)، و«فذلك جاهل فارفضوه» لمن لا يدري ولا يدري أنه لا يدري (ماو ص٧٦).
- **الغزالي:** تعليل الإخلاص (م٢٣).
- **الخطيب (الفقيه):** ابن المقفع «من أنف من قول لا أدري تكلف الكذب» (خ‑ف ص٣٧١).
- **ابن جماعة:** «لأنه دليل عظيم على عظم محله وقوة دينه وتقوى ربه» (ج ص٤٢)، «وإنما يأنف من قول لا أدري من ضعفت ديانته وقلت معرفته» (ج ص٤٣)، علته «لأنه يخاف من سقوطه من أعين الحاضرين» (ج ص٤٣)، وعاقبته «فيقع فيما فر منه ويتصف عندهم بما احترز عنه» (ج ص٤٣).
- **الزيدي:** يزيد على ج: «ولا يخاف من سقوطه من نظر رب العالمين» (ز‑ي ص٨).
- **العلموي:** عن النووي «كغيره»: «لأن المتمكن لا يضره عدم معرفته مسائل معدودة» (ع ص١٣٠)، وأنه «فإنه بإقدامه على الجواب فيما لا يعلمه يبوء بالإثم العظيم» (ع ص١٣٠)، ثم حديث «المتشبع بما لم يعط كلابس ثوبي زور» (ع ص١٣٠). **أبو زيد:** «ويهتك حجابه الاستنكاف منها» (ح ص١٨٤).
- **ابن عبد البر (من القراءة الجديدة):** عمر في أصحاب الرأي: «إن أصحاب الرأي أعداء السنن أعيتهم أن يحفظوها وتفلتت منهم أن يعوها، واستحيوا حين سئلوا أن يقولوا: لا نعلم، فعارضوا السنن برأيهم فإياكم وإياهم» (ع‑ب ج٢ ص١٠٤٢)؛ وحديث القضاة عن ابن بريدة عن أبيه: «وقاض قضى بغير علم واستحيا أن يقول: لا أعلم فهو من أهل النار» (ع‑ب ج٢ ص٨٨٠)، وقول أبي العالية فيمن اجتهد فأخطأ: «ذنبه ألا يكون قاضيا إذا لم يعلم» (ع‑ب ج٢ ص٨٨٠)؛ وقسمة الخليل: «رجل يدري ويدري أنه يدري فذلك عالم فاتبعوه وسلوه» (ع‑ب ج٢ ص٨٢٠)، «ورجل لا يدري ويدري أنه لا يدري فذلك جاهل فعلموه» (ع‑ب ج٢ ص٨٢٠)، «ورجل يدري ولا يدري أنه يدري فذلك عاقل فنبهوه» (ع‑ب ج٢ ص٨٢٠)، «ورجل لا يدري ولا يدري أنه لا يدري فذلك مائق فاحذروه» (ع‑ب ج٢ ص٨٢٠).
> **تحرير:** العلل المقروءة أربع: خوف سقوط المنزلة (ج، وز‑ي مع زيادة نظر الله)، والعجب والتكلف (ماو)، وإرادة غير وجه الله (غ)، والكذب (ابن المقفع عند خ‑ف). ولا تعارض بينها؛ ولا نص يرجح واحدة منها. وانظر خبر القاسم في م٢٦ حيث مدحه شيخ من قريش لما قال «لا أحسنه». (د) قسمة الخليل ثلاثية عند ماو (ص٧٥–٧٦) وأربعية عند ع‑ب (ص٨٢٠): من لا يدري ويدري أنه لا يدري «فذلك مسترشد» عند ماو (ص٧٥) و«جاهل فعلموه» عند ع‑ب؛ ومن لا يدري ولا يدري أنه لا يدري «فذلك جاهل فارفضوه» عند ماو (ص٧٦) و«مائق فاحذروه» عند ع‑ب؛ وينفرد ع‑ب بصنف «يدري ولا يدري أنه يدري»؛ ولا أرجّح. (هـ) وفي حديث القضاة ينقل الاستحياء من لا أعلم إلى الحكم بالنار (ع‑ب ص٨٨٠)، وفي قول عمر يجعله من خصال أصحاب الرأي (ص١٠٤٢).

## ٥١ — متى يسع العالمَ أن يقول «لا أدري»: بين النص والاجتهاد
**المسألة:** هل يسع العالم أن يقول «لا أدري» في كل ما لا يدريه، أم في موضع الرأي والاجتهاد دون ما فيه نصّ؟ وكيف يُجمع بين الأمر بالوقوف عند الإشكال والتحذير من الاستحياء من «لا أعلم»؟
**المصادر:** ع‑ب ج١ ص٧٥٨ وج٢ ص٨٤٨–٨٤٩، ٨٨٠، ١٠٤٢ [٤٦٣]؛ النراقي (جامع السعادات، OCR، بلا صفحة)؛ ش (OCR، بلا صفحة).
**الزيادات:**
- **ابن عبد البر:** سُئل سحنون «أيسع العالم أن يقول: لا أدري فيما يدري؟» فقال: «أما ما فيه كتاب الله قائم أو سنة ثابتة فلا يسعه ذلك» (ع‑ب ج١ ص٧٥٨) «وأما ما كان من هذا الرأي فإنه يسعه ذلك؛ لأنه لا يدري أمصيب هو أم مخطئ» (ع‑ب ج١ ص٧٥٨) (يرويه ابن عبد البر عن ابن وضاح). وابن مسعود في ترتيب مصادر القضاء: «فليجتهد رأيه، فإن لم يحسن فليقر ولا يستحي» (ع‑ب ج٢ ص٨٤٩) مع تعقيب ابن عبد البر: «ومن أشكل عليه شيء لزمه الوقوف» (ع‑ب ج٢ ص٨٤٨).
- **ابن عبد البر (في ذم الاستحياء منها):** حديث القضاة: «وقاض قضى بغير علم واستحيا أن يقول: لا أعلم فهو من أهل النار» (ع‑ب ج٢ ص٨٨٠)؛ وعمر في أصحاب الرأي «واستحيوا حين سئلوا أن يقولوا: لا نعلم» (ع‑ب ج٢ ص١٠٤٢).
- **النراقي** (جامع السعادات، آداب التعليم، OCR، بلا صفحة): يجعل من آداب المعلم أن يقول ما يعلم ويسكت عما لا يعلم، ويعمّ بذلك المفتي والقاضي، وأن يقول المسؤول لا أدري ولا يقول الله أعلم.
- **الشهيد الثاني** (ش، OCR، بلا صفحة): يعدّ الاعتراف بلا أدري من آداب المعلم في درسه، ويبيّن أنه لا يضع منزلة العالم.
> **تحرير:** (أ) سحنون يقيّد لا أدري بغير موضع النص، وابن مسعود يقرنها بعد الاجتهاد، وحديث القضاة يذم الاستحياء منها لمن قضى بغير علم؛ ولا تعارض بينها، وكلها في القضاء والفتيا. (ب) النراقي (OCR) يمنع الله أعلم بدل لا أدري لئلا يوقع الشك في قلب السائل، وعلي وابن مسعود يقولان الله أعلم (م٢٣)؛ ولا أرجّح. (ج) لم أجد عند سحنون تفسيراً لقوله ما كان من هذا الرأي غير ما نقله ابن عبد البر. وانظر م٢٣ في القاعدة، وم٣١ في الإحالة على الأعلم.

## ٣٠ — «لا أعلم» في التفسير وشرح الحديث: «إنما أنا زاملة»
**المسألة:** لا يفسّر الراوي أو المملي إلا ما عرف معناه، ويسكت عما لا يعرفه، ولا يعدّ ذلك نقصاً.
**المصادر:** خ‑ج ج٢ ص١١١–١١٢ وع‑ب ج٢ ص١٠٢٠ [٤٦٣]؛ غ ص٧١ [٥٠٥]؛ سم ص٦٢–٦٣ [٥٦٢]. **موضعها الأوفى:** الباب الحادي عشر، المسألة ١٧ (تفسير الغريب، وألا يفسّر إلا ما عرف).
> **إحالة:** نصوص الخطيب (ج٢ ص١١١–١١٢) والسمعاني (ص٦١–٦٣) في بيان الغريب ثم السكوت عما لا يُعرف، وفيها ابن مهدي وأبو أسامة والزهري ومطر وأيوب وأحمد وزاهر الشحامي، مثبتة بنصوصها في **الباب الحادي عشر، المسألة ١٧**؛ فلا أكررها هنا، وأثبت أدناه ما انفرد به هذا الباب.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب (الجامع):** أحمد «ومن يتعاطى معنى ذلك يخطئ كثيرا إلا بأثر» (خ‑ج ج٢ ص١١٢).
- **ابن عبد البر:** مطر: «لا أدري، إنما أنا زاملة» وجواب الرجل «جزاك الله من زاملة خيرا، فإن عليك من كل حلو وحامض» (ع‑ب ج٢ ص١٠٢٠).
- **الغزالي:** يحكي أن صحابياً «روى» في حضرة الحسن عشرين حديثاً «فسئل عن تفسيرها فقال ما عندي إلا ما رويت» فأخذ الحسن في تفسيرها حديثاً حديثاً (غ ص٧١).
- **السمعاني:** وقول أحمد «بغم كثير ومن يتعاطى معنى ذلك يخطىء كثيرا إلا بأثر» (سم ص٦٣).
> **تحرير:** (أ) عبارة الخطيب «للراوي» وعبارة السمعاني «للمملي» في الجملة نفسها (خ‑ج ص١١١، سم ص٦٢)، والأخبار التي عند السمعاني (الزهري، مطر، أحمد) هي في الخطيب، وإسناد السمعاني لخبر مطر يمر بالخطيب: «أن أبا بكر أحمد بن علي البغدادي أخبرهم» (سم ص٦٣). (ب) جواب أحمد لمن سأله «في الحديث ما لا ندري أيش معناه»: عند خ‑ج «نعم كثير ومن يتعاطى معنى ذلك يخطئ كثيرا إلا بأثر» (ص١١٢)، وعند سم «قال بغم كثير» (ص٦٣) — الخلاف في لفظ الجواب (كثير / بغم كثير) ولا أرجّح. (ج) الخطيب يجعل الأصل التفسير ثم يقيّده بالمعرفة، وأبو زيد وابن جماعة إنما تكلما عن الفتوى والدرس، فلا يظهر عندهما حكم التفسير.

## ٣١ — الإحالة على الأعلم، ووجوب الجواب إذا انفرد المفتي
**المسألة:** إذا أشكلت المسألة على المسؤول وعرف من يعلمها دلّ السائل عليه، وإذا لم يكن في البلد مفتٍ سواه لزمه الجواب.
**المصادر:** المرتضى (الذريعة، OCR، بلا صفحة) [٤٣٦، أقدم مقروء]؛ خ‑ف ج٢ ص٢٣–٢٤، ٣٣، ٣٦٠–٣٦١، ٣٨٦–٣٨٧ [٤٦٣]؛ ع‑ب ج٢ ص١١٢٠–١١٢٢، ١١٢٧ [٤٦٣]؛ خ‑ج ج١ ص٣١٧؛ غ ص٦٩–٧١ [٥٠٥]؛ ب‑ص ص٧٥، ١٠٩ [٦٤٣]؛ ع ص١٧٤.
**الزيادات:**
- **الخطيب (الفقيه):** «فإذا سئل المفتي عن حكم نازلة فأشكل عليه، وهناك من هو عارف به، لزمه أن يرشد السائل إليه، ويدله عليه» (خ‑ف ص٣٦٠)، وعائشة لشريح بن هانئ: «سل عليا، فإنه أعلم مني بهذا» (خ‑ف ص٣٦٠)؛ «فإن لم يكن هناك من يستفتى غيره لزمه الإمساك عنه» (خ‑ف ص٣٦٠). وفي موضع آخر: «إذا لم يكن بالموضع الذي هو فيه مفت سواه لزمه فتوى من استفتاه» (خ‑ف ص٣٨٦)، واستشهد بحديث «من سئل عن علم فكتمه ألجمه الله يوم القيامة بلجام من نار» (خ‑ف ص٣٨٦)، وقتادة «هذا ميثاق أخذه الله على أهل العلم، فمن علم علما فليعلمه» (خ‑ف ص٣٨٧)، ويحيى بن آدم في «وأما السائل فلا تنهر»: «هو الرجل يسألك عن شيء من أمر دينه، فلا تنهره وأجبه» (خ‑ف ص٣٨٧).
- **الخطيب — أخبار الإحالة:** عبد الرحمن بن أبي ليلى «يسأل أحدهم عن المسألة فيردها هذا إلى هذا، وهذا إلى هذا حتى ترجع إلى الأول» (خ‑ف ص٢٣)، وعلقمة لعمير بن سعيد «ائت عبيدة فسله» (خ‑ف ص٢٤).
- **المزني (في نقل الخطيب):** يعارض من يستشهد بذلك: «توجب في قولكم أنهم تركوا ما فرض الله عليهم» (خ‑ف ص٣٣).
- **الغزالي:** يجعل الإحالة جزءاً من القاعدة: «وأحال على غيره إن كان في غيره غنية» (غ ص٦٩)، ويورد ابن عمر «إذا سئل عن الفتيا قال اذهب إلى هذا الأمير الذي تقلد أمور الناس» (غ ص٦٩)، وخبر ابن أبي ليلى «ما منهم أحد يسئل عن حديث أو فتيا إلا ود أن أخاه كفاه ذلك» (غ ص٧٠) ثم «كانت المسئلة تعرض على أحدهم فيردها إلى الآخر ويردها الآخر إلى الآخر حتى تعود إلى الأول» (غ ص٧٠)؛ ويحكي في الإحالة على الأعلم: «وكان أنس رضي الله عنه إذا سئل يقول سلوا مولانا الحسن» (غ ص٧١)، وابن عباس «سلوا حارثة ابن زيد» (غ ص٧١)، وابن عمر «سلوا سعيد بن المسيب» (غ ص٧١).
- **ابن الصلاح:** يكرر خبر ابن أبي ليلى (ب‑ص ص٧٥)، ثم في الفتوى: «وإن لم يحضر غيره فعند الحليمي: تعين عليه بسؤاله جوابه، وليس له أن يحيله على غيره» (ب‑ص ص١٠٩) ويرجّح خلافه: «والأظهر أنه لا يتعين عليه بذلك» (ب‑ص ص١٠٩).
- **العلموي:** يورد خبر ابن أبي ليلى وكلمة البراء «ما فيهم من أحد إلا وهو يحب أن يكفيه صاحبه الفتيا» (ع ص١٧٤).
- **المرتضى** (الذريعة، صفات المفتي، OCR، بلا صفحة): المفتي إذا سُئل عن مسألة ولم يعلم لا يحل له أن يفتي، بل يصدّق المستفتي عن حاله ويرجعه إلى غيره؛ ويجب عليه أن يفتي بما يقطع عليه بعينه ويتوقف عما لا يعلم.
- **ابن عبد البر (من القراءة الجديدة):** ابن أبي ليلى «أدركت عشرين ومائة من أصحاب رسول الله صلى الله عليه وسلم أراه قال: في المسجد فما كان منهم محدث إلا ود أن أخاه كفاه الحديث ولا مفت إلا ود أن أخاه كفاه الفتيا» (ع‑ب ج٢ ص١١٢٠)؛ وابن الزبير لمن سأله في طلاق ثلاثاً قبل الدخول: «إن هذا الأمر ما لنا فيه قول فاذهب إلى عبد الله بن عباس، وأبي هريرة» (ع‑ب ج٢ ص١١٢٢)، وابن عباس لأبي هريرة: «أفته يا أبا هريرة فقد جاءتك معضلة» (ع‑ب ج٢ ص١١٢٢)؛ وزيد بن أرقم والبراء لأبي المنهال في الصرف: «سل الآخر؛ فإنه خير مني وأعلم مني» (ع‑ب ج٢ ص١١٢٧).
- **الخطيب (الجامع):** في النية في التحديث: «وإن كان في بلده أو بغيره من هو أعلى إسنادا منه، دل عليه، وأرشد الطلبة إليه» (خ‑ج ج١ ص٣١٧).
> **خلاف:** (أ) الخطيب عند عدم وجود مفتٍ آخر في موضعين: يلزمه الإمساك ما لم يتضح له (ص٣٦٠)، ويلزمه الإفتاء إذا انفرد (ص٣٨٦)؛ ولا يصرّح بالجمع بينهما. (ب) ابن الصلاح ينقل الإيجاب عن الحليمي ويرجّح عدم التعيّن ويستدل بخبر ابن أبي ليلى. (ج) المزني يعارض الاستدلال بإحالة السلف بأنه يلزم منه ترك الفرض عند من يقيس الجواب على أكل الميتة للمضطر. (د) المرتضى (OCR) يوجب على المفتي التوقف والإحالة عند عدم العلم، ويقابله عند الخطيب ما بين إمساك وإفتاء بحسب انفراده (أ). وانظر الباب الثالث عشر، المسألتين ٢٩ و٤٦ (وجوب الإفتاء إذا انفرد، والإفتاء مع وجود من هو أفضل).

## ٣٢ — «أجسر الناس على الفتيا أقلهم علما»، وتهيّب المفتي
**المسألة:** كثرة الجرأة على الفتيا علامة قلة العلم، والمتورع يتهيّبها.
**المصادر:** ع‑ب ج٢ ص٨١٥–٨١٧، ٨٤٣، ١١٢٠–١١٢٨ وخ‑ف ج٢ ص٢٤، ٣٥٠، ٣٥٣–٣٥٤ [٤٦٣]؛ غ ص٦٩–٧٠ [٥٠٥]؛ ب‑ص ص٧٤–٧٨ [٦٤٣]؛ ع ص١٧٤.
**الزيادات:**
- **ابن عبد البر:** ابن مسعود «إن من يفتي في كل ما يستفتونه لمجنون» (ع‑ب ص٨٤٣)، وقول الحكم «لو سمعت هذا منك قبل اليوم ما كنت أفتي في كل ما أفتي» (ع‑ب ص٨٤٣)، وابن عيينة «أجسر الناس على الفتيا أقلهم علما» (ع‑ب ص٨٤٣). ويقول: «وقد أفردنا بابا في تدافع الفتوى وذم من سارع إليها» (ع‑ب ص٨٤٣) (وهو الباب الذي في ج٢ ص١١٢٠–١١٢٨، وقد قرأته وأثبتُّ منه أدناه). وسفيان بن عيينة عن بشر: «من أحب أن يسأل، وليس بأهل أن يسأل فما ينبغي أن يسأل» (ع‑ب ج٢ ص١٠٦٦).
- **الخطيب (الفقيه):** علقمة «كان يقال: أجرأ القوم على الفتيا أدناهم علما» (خ‑ف ص٢٤)؛ وابن عيينة «أعلم الناس بالفتوى أسكتهم فيه، وأجهل الناس بالفتوى أنطقهم فيه» (خ‑ف ص٣٥٠)، وقول الشافعي فيه «أسكت عن الفتيا منه» (خ‑ف ص٣٥٠)؛ وبشر بن الحارث «من أحب أن يسأل، فليس بأهل أن يسأل» (خ‑ف ص٣٥٣)؛ وعطاء بن السائب «أدركت أقواما إن كان أحدهم ليسأل عن الشيء، فيتكلم وإنه ليرعد» (خ‑ف ص٣٥٣)؛ ومالك «كأنه واقف بين الجنة والنار» (خ‑ف ص٣٥٤)؛ وابن المنكدر «إن العالم بين الله وبين خلقه، فلينظر كيف يدخل عليهم» (خ‑ف ص٣٥٤).
- **الغزالي:** «منها أن لا يكون مسارعا إلى الفتيا بل يكون متوقفا ومحترزا ما وجد إلى الخلاص سبيلا» (غ ص٦٩)؛ وابن مسعود «إن الذي يفتي الناس في كل ما يستفتونه لمجنون» (غ ص٦٩)؛ و«إنما العالم الذي إذا سئل عن المسئلة فكأنما يقلع ضرسه» (غ ص٦٩)؛ وأبو حفص النيسابوري «العالم هو الذي يخاف عند السؤال أن يقال له يوم القيامة من أين أجبت» (غ ص٧٠)؛ وإبراهيم التيمي «لم تجدوا غيري حتى احتجتم إلي» (غ ص٧٠)؛ وقول بعضهم «كان أسرعهم إلى الفتيا أقلهم علما وأشدهم دفعا لها أورعهم» (غ ص٧٠)، و«كان الصحابة يتدافعون أربعة أشياء الإمامة والوصية والوديعة والفتيا» (غ ص٧٠)، و«فلم يزل السكوت دأب أهل العلم إلا عند الضرورة» (غ ص٧٠).
- **ابن الصلاح:** «روي عن مكحول، ومالك رضي الله عنهما: أنهما كانا لا يفتيان حتى يقولا: لا حول ولا قوة إلا بالله» (ب‑ص ص١٤٠)، ويستحب للمفتي أن يقول عند الإفتاء {سبحانك لا علم لنا إلا ما علمتنا إنك أنت العليم الحكيم} (ب‑ص ص١٤٠)؛ وعن سعيد بن المسيب «أنه كان لا يكاد يفتي فتيا، ولا يقول شيئا إلا قال: اللهم سلمني، وسلم مني» (ب‑ص ص٨٠)، ومثله عند ع «كان ابن المسيب لا يفتي فتيا إلا قال: اللهم سلمني وسلم مني» (ع ص١٧٦).
- **ابن الصلاح:** يعلّل: «ولما ذكرناه هاب الفتيا من هابها من أكابر العلماء» (ب‑ص ص٧٤)، وابن مسعود «من أفتى الناس في كل ما يستفتونه فهو مجنون» (ب‑ص ص٧٥)، وأبو حصين «إن أحدكم ليفتي في المسألة، ولو وردت على عمر بن الخطاب رضي الله عنه لجمع لها أهل بدر» (ب‑ص ص٧٦)، وابن عيينة وسحنون «أجسر الناس على الفتيا أقلهم علما» (ب‑ص ص٧٨).
- **العلموي:** ابن عباس «من أفتى الناس في كل ما يسألونه فهو مجنون» (ع ص١٧٤)، وابن المنكدر «فلينظر كيف يدخل بينهم» (ع ص١٧٤).
- **ابن عبد البر (باب تدافع الفتوى وذم من سارع إليها، ج٢ ص١١٢٠–١١٢٨):** ابن عباس «إن من أفتى الناس في كل ما يسألونه عنه لمجنون» (ص١١٢٣) وقال مالك «وبلغني عن ابن مسعود مثل ذلك» (ص١١٢٣)؛ وأبو إسحاق: «كنت أرى الرجل في ذلك الزمان وإنه ليدخل يسأل عن الشيء فيدفعه الناس من مجلس إلى مجلس حتى يدفع إلى مجلس سعيد بن المسيب كراهية للفتوى» (ص١١٢٣)؛ وابن عيينة «أجسر الناس على الفتيا أقلهم علما» (ص١١٢٤) وبيت أبي العتاهية «أشد الناس للعلم إدعاء ... أقلهم بما هو فيه علما» (ص١١٢٤)؛ وسحنون «أجرأ الناس على الفتيا أقلهم علما، يكون عند الرجل الباب الواحد من العلم يظن أن الحق كله فيه» (ص١١٢٥) و«إني لأحفظ مسائل، منها ما فيه ثمانية أقوال من ثمانية أئمة من العلماء فكيف ينبغي أن أعجل بالجواب» (ص١١٢٥)؛ وحذيفة «إنما يفتي الناس أحد ثلاثة» (ص١١٢٦) وابن سيرين «فلست بواحد من هذين وما أحب أن أكون الثالث» (ص١١٢٦)؛ وعمر لأبي مسعود «ألم أنبأ أنك تفتي الناس، ول حارها من تولى قارها» (ص١١٢٦)؛ وأبو حنيفة في المنام «ما أضر الفتيا على أهلها» (ص١١٢٧)؛ وسحنون «إنا لله ما أشقى المفتي والحاكم» (ص١١٢٧)؛ وأبو عثمان بن الحداد «القاضي أيسر مأثما وأقرب إلى السلامة من الفقيه» (ص١١٢٨).
- **ابن عبد البر (معرفة الاختلاف شرط الفتوى، ج٢ ص٨١٥–٨١٧):** أيوب «أجسر الناس على الفتيا أقلهم علما باختلاف العلماء، وأمسك الناس عن الفتيا أعلمهم باختلاف العلماء» (ص٨١٦)، ومثله لابن عيينة «أجسر الناس على الفتيا أقلهم علما باختلاف العلماء» (ص٨١٧)؛ وعطاء «لا ينبغي لأحد أن يفتي الناس، حتى يكون عالما باختلاف الناس» (ص٨١٦).
- **ابن عبد البر (حديث انتزاع العلم):** عن عبد الله بن عمرو: «فيبقى ناس جهال يستفتون فيفتون برأيهم فيضلون ويضلون» (ع‑ب ج٢ ص١٠٣٧).
- وانظر الباب الثالث عشر، المسائل ٣–٩ (تصفح الإمام للمفتين، وبكاء ربيعة، وتدافع الصحابة والتابعين، والنسبة واللفظ في المجنون، وأسكتهم، والرعدة، وهمّ تخليص النفس)، وفيها شهادة السبعين لمالك (خ‑ف ج٢ ص٣٢٥)؛ فلا أكرر ما فيها.
> **تحرير:** (أ) ابن مسعود في المجنون: «إن من يفتي...» عند ع‑ب، «من أفتى الناس...» عند ب‑ص، وينسبه ع إلى ابن عباس، وب‑ص أيضاً «وعن ابن عباس رضي الله عنهما نحوه» (ب‑ص ص٧٥). (ب) ابن المنكدر: «يدخل عليهم» (خ‑ف) و«يدخل بينهم» (ع)، و«فلينظر كيف يدخل بينهم» أيضاً عند ب‑ص (ص٧٤). (ج) «من أحب أن يسأل»: لبشر بن الحارث عند خ‑ف، وبشر ينقله عن سفيان بن عيينة عند ع‑ب (ص١٠٦٦). (د) «أجسر الناس...»: صيغتها عند غ «كان أسرعهم إلى الفتيا أقلهم علما» بلا نسبة (ص٧٠)، ولابن عيينة عند ع‑ب، ولعلقمة «كان يقال» عند خ‑ف ولم يعزه، وعند ب‑ص لابن عيينة وسحنون (عن ع‑ب). (هـ) صيغ «أجسر» في ع‑ب ثلاث: «أجسر الناس على الفتيا أقلهم علما» لابن عيينة (ص٨٤٣، ١١٢٤)، و«أجسر الناس على الفتيا أقلهم علما باختلاف العلماء» لأيوب ثم لابن عيينة (ص٨١٦–٨١٧)، و«أجرأ الناس على الفتيا أقلهم علما» لسحنون (ص١١٢٥)؛ ولا أرجّح. (و) يورد ب‑ص (ص٨٤) «قل من حرص على الفتوى، وسابق إليها، وثابر عليها إلا قل توفيقه» ويقول: «قال ذلك الصيمري أولا، ثم تلقاه عنه الخطيب فقاله في بعض تصانيفه» (ب‑ص ص٨٤)، ثم يروي بإسناده قول بشر «من أحب أن يسأل فليس بأهل أن يسأل» (ب‑ص ص٨٤)؛ وهي في الخطيب (ج٢ ص٣٥٠) بعد قوله «قلت:» بنص «وقل من حرص على الفتوى، وسابق إليها، وثابر عليها إلا قل توفيقه» (خ‑ف ص٣٥٠).

## ٣٣ — زلة العالم، وهمّ المفتي أن يخلّص نفسه
**المسألة:** المفتي مسؤول عن فتواه، وزلته تتعدى إلى أتباعه؛ فعليه أن يتفكر ويقدّم تخليص نفسه على تخليص السائل.
**المصادر:** خ‑ف ج٢ ص٢٥–٢٩، ٣٥٦–٣٥٩ وخ‑ج ج١ ص٢١١–٢١٢ وع‑ب ج٢ ص١٠٧١ [٤٦٣]؛ ب‑ص ص٨٠، ١٤٠ [٦٤٣]؛ ع ص١٧٦.
**الزيادات:**
- **الخطيب (الفقيه):** «أشد ما أتخوف على أمتي ثلاثة: زلة عالم» (خ‑ف ص٢٦)؛ وعيسى «زلة العالم، إذا زل العالم زل بزلته عالم كثير» (خ‑ف ص٢٧)؛ وابن المعتز «زلة العالم كانكسار السفينة» (خ‑ف ص٢٧)؛ وابن عباس «ويل للأتباع من عثرات العالم» (خ‑ف ص٢٧)؛ وأبو حنيفة «لولا الفرق من الله أن يضيع العلم ما أفتيت أحدا» (خ‑ف ص٣٥٦)؛ ومحمد بن واسع «أول من يدعى إلى الحساب يوم القيامة: الفقهاء» (خ‑ف ص٣٥٧)؛ ويرويان قول ابن خلدة لربيعة: «ولكن لتكن همتك أن تخلص نفسك» (خ‑ف ص٣٥٨) «وليكن همتك أن تتخلص مما سألك عنه» (خ‑ف ص٣٥٨)؛ وابن هرمز يرسل من يرد السائل ليقول له «إني قد عجلت فلا تقبل شيئا مما قلت لك حتى ترجع إلي» (خ‑ف ص٣٥٨)؛ وابن خلدة لمالك «إذا سئلت عن شيء فتفكر فيه، فإن وجدت لنفسك مخرجا فتكلم، وإلا فاسكت» (خ‑ف ص٣٥٩)؛ ومالك «ما علمت فقله ودل عليه، وما لم تعلم فاسكت عنه، وإياك أن تتقلد للناس قلادة سوء» (خ‑ف ص٣٥٩).
- **الخطيب (الجامع):** خبر عمر وتميم الداري: «اتقوا زلة العالم»، وتفسير تميم: «العالم يزل بالناس فيؤخذ به» (خ‑ج ص٢١١).
- **ابن عبد البر:** مالك في الرواية الأخرى: «يا عبد الله ما علمته فقل به ودل عليه، وما لم تعلم فاسكت عنه وإياك أن تتقلد للناس قلادة سوء» (ع‑ب ج٢ ص١٠٧١).
> **تحرير:** (أ) قول ابن خلدة لربيعة له روايتان عند الخطيب في صفحتين متتاليتين: «فلا يكن همتك أن تخلصه، ولكن لتكن همتك أن تخلص نفسك» (ص٣٥٧–٣٥٨) و«فلا يكن همك أن تخرجه مما وقع فيه، وليكن همتك أن تتخلص مما سألك عنه» (ص٣٥٨). (ب) قول مالك: «ما علمت فقله» (خ‑ف) و«ما علمته فقل به» (ع‑ب)، والمخاطب في الأول والثاني «يا عبد الله» (ابن وهب)، وهما لراوٍ واحد. وانظر الباب الثالث عشر، المسألة ٩ (همّ المفتي أن يخلّص نفسه)، وهي تحيل إلى هذه المسألة.

## القسم الخامس: هيبة السائل، وابتداء العالم، و«سلوني»

## ٣٤ — هيبة المتعلم للعالم، وقول الشيخ «لا تهبني»
**المسألة:** قد تمنع هيبةُ العالم المتعلمَ من سؤاله زماناً؛ ويرفعها الشيخ بإذنه وبقوله إنه يجيب أو يقول لا أعلم.
**المصادر:** ع‑ب ج١ ص٤٥٥–٤٥٩، ٤٧٤ [٤٦٣]؛ خ‑ف ج٢ ص٣٠، ٣٨٠ [٤٦٣].
**الزيادات:**
- **ابن عبد البر** (باب «هيبة المتعلم للعالم» ص٤٥٥): ابن عباس يريد أن يسأل عمر عن المتظاهرتين، قال ابن عبد البر: «لم يمنع ابن عباس من سؤال عمر عن ذلك إلا هيبته» (ع‑ب ص٤٥٥)؛ وفي رواية الزهري «مكثت سنتين أريد أن أسأل عمر بن الخطاب رضي الله عنه عن حديث ما منعني منه إلا هيبته» (ع‑ب ص٤٥٦)، وقول عمر له: «فلا تفعل إذا أردت أن تسأل فسلني فإن كان عندي منه أخبرتك وإلا قلت: لا أعلم، فسألت من يعلم» (ع‑ب ص٤٥٦)؛ وسعيد بن المسيب لسعد بن مالك: «إني أريد أن أسألك عن شيء وإني أهابك» فقال سعد «لا تهبني يا ابن أخي إذا علمت أن عندي علما، فسألني عنه» (ع‑ب ص٤٥٧)؛ وطاوس «إن من السنة، أن توقر العالم» (ع‑ب ص٤٥٩)؛ وابن عباس «ذللت طالبا فعززت مطلوبا» (ع‑ب ص٤٧٤).
- **الخطيب (الفقيه):** في نقل المزني: عمر لابن عباس «سل عما بدا لك، فإن كان عندنا، وإلا سألنا عنه غيرنا» (خ‑ف ص٣٠)؛ وفي أدب المستفتي: «من السنة أن يوقر، أربعة: العالم، وذو الشيبة، والسلطان والوالد» (خ‑ف ص٣٨٠).
> **تحرير:** رواية خبر ابن عباس وعمر اختلفت عند ع‑ب نفسه: في الأولى مكث «سنة، وأنا أشك في سنتين» وأراد السؤال في الطريق فأجابه عمر قبل أن يتم كلامه (ع‑ب ص٤٥٥)، وفي الثانية «سنتين» وبدأ بالمصارحة بهيبته فأذن له عمر وقال له إنه يقول «لا أعلم» إن لم يعلم (ع‑ب ص٤٥٦). والجملة المتعلقة بـ«لا أعلم» ليست في الأولى. (وفي ٠٤ تفصيل هيبة المتعلم لشيخه.)

## ٣٥ — ابتداء العالم الجلساء بالفائدة، أو انتظاره السؤال، أو تمنّعه
**المسألة:** هل يبتدئ العالم بالتحديث والفائدة، أم ينتظر أن يُسأل، أم يتمنع وإن سُئل؟ تعرض المصادر المقروءة مذاهب العلماء الثلاثة بلا ترجيح.
**المصادر:** ع‑ب ج١ ص٤٦٠–٤٧٥ [٤٦٣] وخ‑ج ج١ ص٢٠٢–٢٠٦ [٤٦٣] وخ‑ف ج٢ ص٢٩٢، ٢٩٥؛ لا أحكم بأقدمية؛ غ ص٦٩–٧٠ [٥٠٥].
**الزيادات:**
- **ابن عبد البر** (ترجمة الباب: «باب في ابتداء العالم جلساءه بالفائدة وقوله: سلوني وحرصهم على أن يؤخذ ما عندهم» ص٤٦٠): يبدؤه بحديث النبي «خذوا عني خذوا عني قد جعل الله لهن سبيلا» (ع‑ب ص٤٦٠) و«خذوا عني مناسككم؛ فإني لا أدري لعلي لا أحج بعد حجتي هذه» (ع‑ب ص٤٦١)، ثم نهيه عن التبشير لمعاذ: «إذا يتكلوا» (ع‑ب ص٤٦٢)، وعلي «ألا رجل يسأل فينتفع وينتفع جلساؤه؟» (ع‑ب ص٤٦٣)؛ ثم المذاهب: الحسن «كان يبتدئ الناس بالعلم ويقول: سلوني» (ع‑ب ص٤٦٩)، وابن سيرين وإبراهيم «لا يبتدئان أحدا حتى يسألا» (ع‑ب ص٤٦٩)؛ والحسن «أتى على الحسن زمان وهو يعجب ممن يدعو إلى نفسه» ثم «فما مات حتى دعا إلى نفسه» (ع‑ب ص٤٦٩)؛ ولقمان «إن العالم يدعو الناس إلى علمه بالصمت والوقار» (ع‑ب ص٤٧٠).
- **الخطيب (الجامع):** يفتتح الباب بقوله «مذاهب المحدثين في الرواية تختلف، فمنهم من يبتدئ بها احتسابا من غير أن يسأل» (خ‑ج ص٢٠٢)، ومنه أبو سعيد الخدري «مرحبا بوصية رسول الله صلى الله عليه وسلم» (خ‑ج ص٢٠٢)، وعطاء الخراساني «إذا لم يجد أحدا يحدثه أتى المساكين فحدثهم» (خ‑ج ص٢٠٣)؛ ثم «ومن المحدثين من لا يروي شيئا إلا بعد أن يسأل» (خ‑ج ص٢٠٤) ومنهم إبراهيم «كان إبراهيم لا يحدث حتى يسأل» (خ‑ج ص٢٠٤) وابن طاوس «إن سألتموني عن شيء ذكرته، وإلا فأهدر عليكم» (خ‑ج ص٢٠٤)؛ ثم «ومنهم من يتمنع وإن سئل، اعتمادا على قول شعبة بن الحجاج» (خ‑ج ص٢٠٤) وشعبة «تمنع أشهى لك» (خ‑ج ص٢٠٥)؛ وكان بعضهم «يتمنع من التحديث إذا كان السامع ليس من أهل العلم» (خ‑ج ص٢٠٥) كالأعمش «من يعلق الدر على الخنازير» (خ‑ج ص٢٠٥)؛ ومالك «من إهانة العلم أن تحدث كل من سألك» (خ‑ج ص٢٠٥).
- **الغزالي:** يوافق المذهب الثاني في وصف الأبدال: «كلامهم ضرورة أي لا يتكلمون حتى يسألوا»، وأنهم «يعدون الابتداء قبل السؤال من الشهوة الخفية للكلام» (غ ص٦٩)؛ ومرّ علي وعبد الله برجل يتكلم على الناس «فقال هذا يقول اعرفوني» (غ ص٦٩)؛ وإبراهيم بن أدهم «ليس شيء أشد على الشيطان من عالم يتكلم بعلم ويسكت بعلم» (غ ص٦٩)؛ وأبو العالية وإبراهيم بن أدهم والثوري «يتكلمون على الاثنين والثلاثة والنفر اليسير فإذا كثروا انصرفوا» (غ ص٧٠).
- **الخطيب (الفقيه):** في تخصيص النجيب من الطلبة: «وإذا بان للفقيه نفاذ أحد أصحابه في العلم وحسن بصيرته بالفقه، جاز له تخصيصه دونهم وأثرته عليهم» (خ‑ف ص٢٩٢)، ويورد عروة لبنيه «لا تغشوني مع الناس، إذا خلوت فسلوني» (خ‑ف ص٢٩٥).
> **تحرير:** (أ) ثلاثة مذاهب صرّح بها الخطيب (الابتداء، وانتظار السؤال، والتمنع) وذكر ع‑ب أولين منها صراحة (الحسن / ابن سيرين وإبراهيم). (ب) ورد خبر عروة «لا تغشوني مع الناس» عند الخطيب (خ‑ف) في سياق التخصيص، وعند ع‑ب (ص٤٧٢) في سياق حرص العلماء على أن يؤخذ عنهم (م٣٦)؛ وهو واحد. (ج) أبو سعيد الخدري عند الخطيب «فكان يجيبنا بمسائلنا» (خ‑ج ص٢٠٢) وهو في الباب الذي صدّره الخطيب بمن يبتدئ؛ ولا أفسر عبارة «فإذا نفدت مسائلنا نا بعد حتى نمل» لاضطراب نصها في المطبوع. (د) عند الغزالي (٥٠٥) أن الأبدال «يعدون الابتداء قبل السؤال من الشهوة الخفية للكلام» (غ ص٦٩)، وفي المقابل يبتدئ الحسن عند ع‑ب بـ«سلوني» (ص٤٦٩) ويحثّ علي «ألا رجل يسأل فينتفع وينتفع جلساؤه؟» (ص٤٦٣). (هـ) لا أجد في ما قرأت ترجيحاً بين المذاهب الثلاثة؛ والخطيب يعرضها ثم ينتقل إلى أدب الطالب مع المتمنع (م٧).

## ٣٦ — حرص العلماء على أن يؤخذ عنهم
**المسألة:** رُوي عن جماعة من العلماء تأسفهم على قلة من يسألهم وحرصهم على أن يُحمل عنهم علمهم.
**المصادر:** ع‑ب ج١ ص٤٦٩–٤٧٥ [٤٦٣].
**الزيادات:**
- **ابن عبد البر:** سعيد بن جبير «إن مما يهمني أني وددت أن الناس قد أخذوا ما معي من العلم» (ع‑ب ص٤٦٩) و«أما أحد يسألني؟» (ع‑ب ص٤٧١)؛ وعكرمة «ما لكم لا تسألوننا أفلستم؟» (ع‑ب ص٤٧١)؛ وعروة «ائتوني فتلقوا مني» (ع‑ب ص٤٧٠)، وقوله «والله، ما يسألني الناس عن شيء، حتى لقد نسيت» (ع‑ب ص٤٧٢)، وقوله لبنيه «لا تغشوني مع الناس وإذا خلوت فاسألوني» (ع‑ب ص٤٧٢)؛ والثوري «والله لو لم يأتوني لأتيتهم في بيوتهم يعني أصحاب الحديث» (ع‑ب ص٤٧٣)، ثم لما قيل إنهم يطلبونه بغير نية: «إن طلبهم إياه نية» (ع‑ب ص٤٧٥)؛ والشافعي للربيع: «يا ربيع، لو قدرت أن أطعمك العلم لأطعمتك إياه» (ع‑ب ص٤٧٣)؛ وهشام عن أبيه «إنا كنا أصاغر قوم ثم نحن اليوم أكابر» (ع‑ب ص٤٧٢).
> **تحرير:** قول عروة «ائتوني فتلقوا مني» يرويه مصعب الزبيري عن ابن عيينة عن عمرو بن دينار ويضم إليه «وكان عروة يستألف الناس على حديثه»، فقال أحمد بن زهير «كذا قال مصعب بن عبد الله، أدخل حديث الزهري في حديث عمرو بن دينار» (ع‑ب ص٤٧٠)، ثم يروي ابن عبد البر كلاً من الحديثين على حدة (ع‑ب ص٤٧٠–٤٧١). والخبر في الباب كله من الحرص على التعليم، فهو يقابل تمنّع من تمنّع (م٣٥).

## ٣٧ — «سلوني»: من قالها، وكيف عُلّل قولها
**المسألة:** قول العالم لجلسائه «سلوني» ثابت عن النبي وعن علي وابن عباس والشافعي، وقيل إنه لم يقلها من الصحابة غير علي؛ وعلّل الخطيب قول علي بتعين الفتوى عليه.
**المصادر:** ع‑ب ج١ ص٤٦٣–٤٦٦ وخ‑ف ج٢ ص٢٩٧، ٣٥١–٣٥٣ وخ‑ف ج١ ص٤٤٥ [٤٦٣]؛ سم ص١٣٣ [٥٦٢]؛ ع ص٢٧١ [٩٨١]؛ مق ص٢٤٢–٢٤٣ وهامش ص٢٤٢ (معاصر).
**الزيادات:**
- **ابن عبد البر:** سعيد بن المسيب «ما كان أحد من الناس يقول: سلوني غير علي بن أبي طالب رضي الله عنه» (ع‑ب ص٤٦٣)؛ وخطبة علي في رواية أبي الطفيل: «سلوني فوالله لا تسألوني عن شيء يكون إلى يوم القيامة إلا حدثتكم به» (ع‑ب ص٤٦٤) وبعدها سؤال ابن الكواء (م١٠)؛ وابن عباس «سلوني؛ فإني أصبحت طيبة نفسي» (ع‑ب ص٤٦٦) ثم «وسلوني عن سورة البقرة وسورة يوسف» (ع‑ب ص٤٦٦).
- **الخطيب (الفقيه):** النبي لما أكثر عليه السائلون «غضب، ثم قال للناس: سلوني عما شئتم» فقام رجل فقال «من أبي؟» (خ‑ف ص٢٩٧)؛ وفي باب الزجر عن التسرع إلى الفتوى يورد اعتراضاً: «فإن قال قائل: فقد قال علي بن أبي طالب: سلوني قبل أن تفقدوني؟ قيل له: الخبر عنه بذلك معروف» (خ‑ف ص٣٥١)، ثم خطبة علي «سلوني، والله لا تسألوني عن شيء يكون إلى يوم القيامة إلا حدثتكم به» (خ‑ف ص٣٥٢)، وروايته «لم يكن أحد من أصحاب النبي صلى الله عليه وسلم يقول: سلوني إلا علي بن أبي طالب» عن سعيد بن المسيب بصيغة «أراه عن سعيد بن المسيب» (خ‑ف ص٣٥٢)، ثم تعليله: «وإنما كان يقول هذا القول وقد انتهى الأمر إليه، وتعينت الفتوى عليه، وانقرضت الفقهاء من الصحابة سواه» (خ‑ف ص٣٥٢)، و«ثم من أين بعد علي مثله، حتى يقول هذا القول» (خ‑ف ص٣٥٣). وفي ترجمة الشافعي: «سلوني عما شئتم أخبركم من كتاب الله ومن سنة رسول الله صلى الله عليه وسلم» ثم قال الراوي «فقلت في نفسي: إن هذا لرجل جريء» (خ‑ف ج١ ص٤٤٥)، وسأله عن محرم قتل زنبوراً فأجابه بآية الحشر.
- **السمعاني:** حديث أنس «ثم أكثر النبي صلى الله عليه وسلم إن يقول سلوني فبرك عمر على ركبتيه» (سم ص١٣٣). **العلموي:** «كان بمكة يقول: سلوني عما شئتم أخبركم عنه من كتاب الله» (ع ص٢٧١) وسؤاله عن الزنبور. **المقدم:** النبي «من أحب أن يسأل عن شيء فليسأل عنه، فوالله! لا تسألوني عن شيء إلا أخبرتكم به ما دمت في مقامي هذا» (مق ص٢٤٢)، «وأكثر رسول الله صلى الله عليه وسلم أن يقول: سلوني» (مق ص٢٤٣).
- **الخطيب (تتمة):** يعلّل أن علياً لم يقل ذلك في عهد أبي بكر وعمر: «لأنه قد كان في ذلك الوقت جماعة يكفون أمر الفتوى» (خ‑ف ص٣٥٣).
- **المقدم (هامش ص٢٤٢):** ينقل عن الشاطبي: «وظاهر هذا المساق يقتضي أنه إنما قال: سلوني في معرض الغضب، تنكيلا بهم في السؤال، حتى يروا عاقبة ذلك» (مق هامش ص٢٤٢، عن الموافقات ٤/٣١٦)؛ وفي متنه عن ابن عباس في سبب نزول آية النهي عن السؤال عن أشياء: «كان قوم يسألون رسول الله صلى الله عليه وسلم استهزاء، فيقول الرجل: من أبي؟ ويقول الرجل، تضل ناقته: أين ناقتي؟» (مق ص٢٤٢).
> **تحرير:** (أ) خبر سعيد بن المسيب «ما كان أحد من الناس يقول: سلوني غير علي» (ع‑ب ص٤٦٣) وعند الخطيب «لم يكن أحد من أصحاب النبي صلى الله عليه وسلم يقول: سلوني إلا علي» مع الشك «أراه عن سعيد بن المسيب» (ص٣٥٢)؛ ثم يورد ع‑ب بعد صفحات قول ابن عباس «سلوني» (ص٤٦٦)، فالأخبار متعارضة في الظاهر (ابن عباس قالها)، وليس في المقروء توفيق بينها ولا ترجيح. (ب) رواية أبي موسى عند الخطيب (الغضب) ورواية أنس عند السمعاني ومق (ذكر الساعة) تذكران كلتاهما سؤال «من أبي؟» وجواب «أبوك حذافة»، وليس في المقروء نص على أنهما حادثة واحدة. (ج) الشافعي بمكة: عند الخطيب (ج١ ص٤٤٥) بإسناد عن الفريابي، وعند ع (ص٢٧١) بصيغة «وروي» بلا إسناد؛ والقصة واحدة في الجملة (سلوني، ثم سؤال الزنبور). (د) توجيه الشاطبي (في نقل المقدم) لقول النبي سلوني بأنه في معرض الغضب تنكيلاً بالسائلين يوافق رواية الغضب عند الخطيب (ج٢ ص٢٩٧)، وهو غير تعليل الخطيب لقول علي بتعين الفتوى عليه (ص٣٥٢).

## ٣٨ — فتوى الصغير بين يدي الكبير بإذنه، والحدث في حلقة النظر
**المسألة:** هل يفتي الأصغر والأحدث في مجلس فيه من هو أكبر منه؟ تقيّده المصادر بالإذن والمنزلة.
**المصادر:** ع‑ب ج١ ص٤٨٥–٤٨٧ وخ‑ف ج٢ ص٢٧٥–٢٨١ [٤٦٣].
**الزيادات:**
- **ابن عبد البر** (باب «فتوى الصغير بين يدي الكبير بإذنه» ص٤٨٥): حديث معاذ وقول عبد الرحمن بن غنم له: «فللرجل العالم أن يقول ومعه عداده من الناس في الأمر لا بد به؟» فقال معاذ «إن شاء قال وإن شاء أمسك حتى يكفيه أصحابه فذلك أحب إلي» (ع‑ب ص٤٨٥)، ويحكم ابن عبد البر على حديثه: «وهذا حديث لا يحتج بمثله لضعف إسناده، ولكنه حديث حسن رجاله معرفون» (ع‑ب ص٤٨٥)؛ ثم سالم بن عبد الله مع الحجاج وابن عمر وكيف نظر الحجاج إلى ابن عمر «فلما رأى ذلك عبد الله قال: صدق» (ع‑ب ص٤٨٦)؛ وقصة حجاج بن عمرو مع زيد: «أفته يا حجاج» فقال حجاج «إنما نجلس إليك لنتعلم منك» ثم «هو حرثك إن شئت سقيته وإن شئت أعطشته» (ع‑ب ص٤٨٧) فقال زيد «صدقت».
- **الخطيب (الفقيه):** أخبار الحدث الذكي بين الصمت والإذن في الكلام (خ‑ف ص٢٧٥–٢٧٩: إقبال الفقيه عليه، وصمته في حلقة النظر، وعمر لابن عباس، وإذنه إذا علت منزلته) مثبتة بنصوصها في **الباب الثامن، المسألة ٢٥**؛ فلا أكررها هنا. ومما لم أجده بلفظه هناك: «وإذا حضر هذا الصبي حلقة النظر، فينبغي له أن يصغي إلى كلام المتناظرين، ويكون من جملة الصامتين» (خ‑ف ص٢٧٥)، وعمر لابن عباس «إني كنت نهيتك أن تتكلم، فإذا دعوتك معهم فتكلم» (خ‑ف ص٢٧٨)؛ وعلى الفقيه: «وإذا أجاب المسئول بالصواب فعلى الفقيه أن يعرفه إصابته» (خ‑ف ص٢٧٩)، «وإذا أخطأ المسئول في الجواب، فعلى الفقيه أن يعلمه ذلك» (خ‑ف ص٢٨١).
> **تحرير:** ابن عبد البر يضعّف إسناد حديث معاذ ثم يذكره «لتقف على ذلك» (ع‑ب ص٤٨٥)؛ والخطيب لا يذكر حديث معاذ في ما قرأت، ويقيّد كلام الحدث بالمنزلة والإذن. وفي أخبار ع‑ب يفتي الأصغر بحضرة الأكبر (سالم مع أبيه ابن عمر، وحجاج بن عمرو بأمر زيد: «أفته يا حجاج»)، وفي أخبار الخطيب (ابن عباس مع عمر) يؤذن للأصغر بالكلام حيث يُدعى. ولم أجد عند أي منهما حداً للسن.

## القسم السادس: آداب المستفتي

## ٣٩ — وجوب الاستفتاء، وطلب المفتي والرحلة إليه
**المسألة:** من نزلت به نازلة وجب عليه أن يسأل من يفتيه، ويرحل إليه إن لم يكن في بلده؛ وكل من لم يبلغ رتبة الفتوى فهو في الشرعيات مستفتٍ مقلِّد.
**المصادر:** المرتضى (الذريعة، OCR، بلا صفحة) [٤٣٦، أقدم مقروء]؛ خ‑ف ج٢ ص٣٧٥–٣٧٦ [٤٦٣]؛ ب‑ص ص١٥٨ [٦٤٣]؛ ع ص٢٠٩، ٢١٦ [٩٨١]. **موضعها الأوفى:** الباب الثالث عشر، المسألتان ٧٥ و٧٦.
> **إحالة:** نصوص الخطيب في وجوب طلب المفتي والرحلة إليه، وتعريف ابن الصلاح للمستفتي وحدّ التقليد، ونقل العلموي عنه، مثبتة بنصوصها في **الباب الثالث عشر، المسألتين ٧٥ و٧٦**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **المرتضى** (الذريعة، فصل صفة المفتي والمستفتي، OCR، بلا صفحة): لا يجب على العامي الاستدلال والاجتهاد، ويجوز له قبول قول المفتي؛ ودليله إجماع الأمة قديماً وحديثاً على رجوع العامي إلى المفتي، وأن تجويز الخطأ على المفتي لا يمنع قبول قوله كما في الشاهدين؛ ويحكي عن قوم منعوا الاستفتاء وأوجبوا على العامي العلم بأحكام الفروع، ويفرّق بين أصول الدين (لا تقليد فيها) والفروع (لا تنحصر حوادثها فلا يتمكن العامي من تفصيل أحكامها).
- **الخطيب:** يستدل برحلة رجل إلى أبي الدرداء في أمر زوجته التي أمرته أمه بفراقها، فقال له «ما أنا بالذي آمرك أن تطلق، وما أنا بالذي آمرك أن تمسك» (خ‑ف ص٣٧٥) ثم روى حديث «الوالد أوسط أبواب الجنة» (خ‑ف ص٣٧٥)؛ وبرحلة سعيد بن جبير إلى ابن عباس بعد أن «اختلف أهل الكوفة في هذه الآية» (خ‑ف ص٣٧٦).
- **العلموي:** قيداً في آخر الباب: «وإذا لم يجد صاحب الواقعة مفتيا ولا من ينقل له حكمها لا في بلده ولا في غيره، فالصحيح أنه غير مكلف فلا يؤاخذ بشيء يصنعه فيها» (ع ص٢١٦).
> **تحرير:** (أ) الخطيب لا يعرّف المستفتي ولا يذكر التقليد في الموضع المقروء، والتعريف من ب‑ص ثم ع. (ب) الخطيب يوجب الرحيل «وإن بعدت داره»، وع يضيف حكم من لم يجد مفتياً لا في بلده ولا في غيره فيسقط عنه التكليف؛ وهذه الإضافة لم أجد نظيرها عند الخطيب ولا في ب‑ص. (ج) خبر أبي الدرداء وسعيد بن جبير عند الخطيب فقط في المقروء.

## ٤٠ — اختيار المفتي والتثبّت في أهليته
**المسألة:** لا يستفتي المرء كل من ادّعى العلم أو انتسب إليه؛ بل يسأل من يثق به عن أعلم أهل المحلة.
**المصادر:** خ‑ف ج٢ ص٣٧٦–٣٧٩ [٤٦٣]؛ ب‑ص ص١٥٨–١٥٩ [٦٤٣]؛ ع ص٢٠٩ [٩٨١]. **موضعها الأوفى:** الباب الثالث عشر، المسألة ٧٧.
> **إحالة:** نصوص الخطيب (عمر بن عبد العزيز، وابن سيرين، ويزيد بن هارون، وابن أبي الزناد) وابن الصلاح والعلموي في البحث عن أهلية المفتي والسؤال عن أعلم أهل البلد مثبتة بنصوصها في **الباب الثالث عشر، المسألة ٧٧**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** «وإذا قصد أهل محلة للاستفتاء عما نزل به، فعليه أن يسأل من يثق بدينه ويسكن إلى أمانته عن أعلمهم وأمثلهم» (خ‑ف ص٣٧٦).
- **ابن الصلاح:** «فإنه يجب عليه قطعا البحث الذي يعرف به صلاحية من يستفتيه للإفتاء» (ب‑ص ص١٥٨)، ثم «وقد أطلق الشيخ أبو إسحاق الشيرازي وغيره: أنه يقبل فيه خبر العدل الواحد» (ب‑ص ص١٥٩) ويقيّد «ولا يعتمد في ذلك على خبر آحاد العامة» (ب‑ص ص١٥٩).
- **العلموي:** «يلزم المستفتي أن يستفتي من عرف علمه وعدالته، فإن جهلت فالأصح الاكتفاء بستارته» (ع ص٢٠٩).
> **تحرير:** (أ) الخطيب يجعل التحقق بسؤال الثقة، وب‑ص يجعله بحثاً واجباً مع جواز الاعتماد على الاستفاضة أو خبر العدل، ويرد خبر آحاد العامة. (ب) قول ع «فإن جهلت فالأصح الاكتفاء بستارته» لم أجده في ب‑ص ص١٥٨–١٥٩ ولا عند الخطيب؛ فهو من ع أو من مصدر لم أقرأه. (ج) خلاف في أن المتواتر لا يفيد (ب‑ص ص١٥٨ ينقل عن «بعض أصحابنا المتأخرين» أنه إنما يعتمد قوله «أنا أهل للفتوى» لا شهرته)، ولم أجد هذا الخلاف عند ع.

## ٤١ — الأعلم والأسن، وتعدد المفتين، واختلاف الفتوى
**المسألة:** إذا تعدد المفتون هل يلزم المستفتي تقليد الأعلم، ومن يبدأ بالسؤال، وماذا يفعل إذا اختلفت فتواهم؟
**المصادر:** المرتضى (الذريعة، OCR، بلا صفحة) [٤٣٦، أقدم مقروء]؛ خ‑ف ج٢ ص٣٧٩–٣٨٠، ٣٨٤–٣٨٥ [٤٦٣]؛ غ ج٢ ص١١٥ [٥٠٥]؛ ب‑ص ص١٥٩–١٦٦ [٦٤٣]؛ ع ص٢٠٩–٢١٤ [٩٨١]. **موضعها الأوفى:** الباب الثالث عشر، المسائل ٧٨ و٧٩ و٨٠ و٨١ و٨٢.
> **إحالة:** وجوه ابن الصلاح في وجوب البحث عن الأعلم والأورع، وتقليد الميت، والتخير بين المذاهب وتتبع الرخص، وأوجه اختلاف المفتين على المستفتي (الأغلظ والأخف والأوثق وسؤال مفتٍ آخر والتخيير)، ونقل العلموي لها، مثبتة بنصوصها في **الباب الثالث عشر، المسائل ٧٨–٨٢**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **المرتضى** (الذريعة، OCR، بلا صفحة): للعامي طريق إلى معرفة من يستفتيه بالمخالطة والأخبار المتواترة عن حال العلماء في بلده ورتبتهم في العلم والديانة؛ فإن كانت الصفات عنده لعالم واحد لزمه استفتاؤه تعييناً، وإن تساووا عنده كان مخيراً، وبعضهم أوجب استفتاء الأقدم في العلم والدين وهو الأولى لأن الثقة أوكد؛ وفي اختلاف المفتين عليه (طلاق الثلاث، ورؤية الهلال) الأولى كونه مخيراً بين القولين لأنه لا طريق له إلى معرفة الأحق منهما.
- **الخطيب:** «وإن ذكر له اثنان، أو أكثر بدأ بالأسن والأكثر منهم رياضة ودربة» (خ‑ف ص٣٧٩)، ويستشهد بقول النبي في قصة حويصة ومحيصة «كبر كبر» (خ‑ف ص٣٨٥) وبحديث ابن عمر «يستن فأعطاه أكبر القوم» ثم «إن جبريل أمرني أن أكبر» (خ‑ف ص٣٨٥)؛ وفي الرقعة الواحدة تقدَّم الأسن والأعلم، وفي الرقاع المفردة «فلا عليه أن يبدأ بأيهم شاء» (خ‑ف ص٣٨٥).
- **الغزالي** (من مبحث مثارات الشبهة في الجزء الثاني من ملف الإحياء): «وإن كان المقلد يجوز له أن يأخذ بما أفتى له مقلده الذي يظن أنه أفضل علماء بلده» (غ ج٢ ص١١٥)، «ويعرف ذلك بالتسامع كما يعرف أفضل أطباء البلد بالتسامع والقرائن» (غ ج٢ ص١١٥)، «وليس للمستفتي أن ينتقد من المذاهب أوسعها عليه بل عليه أن يبحث حتى يغلب على ظنه الأفضل ثم يتبعه» (غ ج٢ ص١١٥).
- **ابن الصلاح:** في الوجهين في وجوب البحث عن الأعلم: «يجب عليه ذلك، وهو قول ابن سريج، واختيار القفال المروزي» (ب‑ص ص١٦٠).
> **تحرير:** (أ) صيغة «تقليد الأورع من العالمين، والأعلم من الورعين» عند ب‑ص و«تقليد أورع العالمين، وأعلم الورعين» عند ع بإسقاط «من». (أ‑١) الغزالي (٥٠٥) أقدم من ب‑ص وع في منع انتقاء المستفتي أوسع المذاهب عليه، وفي جواز الأخذ بما يفتيه من يظنه أفضل علماء بلده؛ ولم أجد في ما قرأت من الغزالي ذكر الخلاف الذي حكاه ب‑ص في وجوب البحث عن الأعلم. (ب) يتوسع ع بعد ص٢١٠ في ترجيح مذهب الشافعي ونسب قريش (ص٢١١–٢١٣)، وهذا الجزء الزائد على ب‑ص ليس من آداب الاستفتاء فلا أنقله، وأكتفي بأنه من زيادات ع (مثل «وهذا ومن قواعده: إذا صح الحديث فهو مذهبي» ع ص٢١١). (ج) هذه المسألة (تقليد الميت والتمذهب واختلاف الفتوى) تقع في المصادر المقروءة في باب المستفتي عند ب‑ص وع فقط، ولم أجدها عند الخطيب في ص٣٧٥–٣٨٩ المقروءة.

## ٤٢ — لزوم الفتوى، وتجديد السؤال، والسؤال بالوكيل والخط
**المسألة:** متى يلزم المستفتيَ العملُ بالفتوى؟ وهل يجدد السؤال إذا تكررت الحادثة؟ وهل يرسل غيره أو يعتمد على خط المفتي؟
**المصادر:** غ ج٢ ص١٠٣ [٥٠٥]؛ ب‑ص ص١٦٦–١٦٨ [٦٤٣]؛ ع ص٢١٤–٢١٥ [٩٨١]؛ خ‑ف ج٢ ص٣٨٦ [٤٦٣]. **موضعها الأوفى:** الباب الثالث عشر، المسائل ٨٣ و٨٤ و٨٥.
> **إحالة:** نصوص ابن الصلاح في لزوم الفتوى وتجديد السؤال والوكالة والخط، ونقل العلموي لها، مثبتة بنصوصها في **الباب الثالث عشر، المسائل ٨٣–٨٥**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **الغزالي:** في موضع الشك: «وما وقع الشك فيه استفتى فيه القلب فإن الإثم حزاز القلوب» (غ ج٢ ص١٠٣) ويستشهد بقول النبي لوابصة «استفت قلبك وإن أفتوك وأفتوك وأفتوك» (غ ج٢ ص١٠٣)، ثم يقول «فالمفتي يفتي بالظن» (غ ج٢ ص١٠٣).
- **ابن الصلاح:** يحكي عن أبي المظفر السمعاني: «إذا سمع المستفتي جواب المفتي لم يلزمه العمل به إلا بالتزامه» (ب‑ص ص١٦٦)، ويقول: «قلت: لم أجد هذا لغيره» (ب‑ص ص١٦٦).
> **تحرير:** (أ) العبارة التي ينسبها ع إلى الخطيب لم أجدها في الخطيب الذي قرأته؛ وأقرب ما فيه عنوان باب ما يفعله المفتي: «إذا لم يكن بالموضع الذي هو فيه مفت سواه لزمه فتوى من استفتاه» (خ‑ف ص٣٨٦)، والملزَم فيها المفتي بالإجابة لا المستفتي بالعمل (م٣١)؛ فنسبة ع تحتاج مراجعة موضع آخر من الخطيب. (ب) ب‑ص يقول عن قول السمعاني «لم أجد هذا لغيره» ثم يفصل بين وجود مفتٍ آخر وعدمه، وع يحكي السمعاني ويسكت عن ذلك التفصيل في هذا الموضع. (ج) «ويكفي ترجمان واحد» زيادة من ع ليست في ب‑ص ص١٦٨.


## ٤٣ — أدب المستفتي في خطاب المفتي
**المسألة:** يبجّل المستفتي المفتي في خطابه وسؤاله، ولا يقول له ما يشعر بالتحقق منه أو بمقابلته بغيره، ولا يسأله في حال شغل أو قيام. (وقت السؤال وحال الشيخ في م٦.)
**المصادر:** خ‑ف ج٢ ص٣٧٩–٣٨٢ [٤٦٣ أقدم المقروء]؛ ب‑ص ص١٦٨–١٦٩ [٦٤٣]؛ ع ص٢١٥ [٩٨١]؛ مق ص٢٢٧ (معاصر)؛ ج ص١٠١ وح ص١٨٨ (في الطالب مع شيخه). **موضعها الأوفى:** الباب الثالث عشر، المسألة ٨٦.
> **إحالة:** نصوص الخطيب وابن الصلاح والعلموي في تبجيل المستفتي للمفتي في خطابه وسؤاله، وألا يسأله قائماً أو مستوفزاً، مثبتة بنصوصها في **الباب الثالث عشر، المسألة ٨٦**؛ ووقت السؤال وحال الشيخ في م٦ هنا؛ فلا أكرر نصوص المستفتي.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** «ومن أدب المستفتي أن لا يقول عند جواب المفتي: هكذا قلت أنا، أو هكذا وقع لي، أو بهذا أجبت» (خ‑ف ص٣٨٢).
- **العلموي:** «ولا يقل ما تحفظه في كذا؟ أو ما مذهب إمامك في كذا؟» (ع ص٢١٥).
- **ابن جماعة وأبو زيد (في الطالب مع الشيخ):** «ولا يقول له لم، ولا لا نسلم، ولا من نقل هذا» (ج ص١٠١)، «وإياك إذا حصل الجواب أن تقول: لكن الشيخ فلان قال لي كذا» (ح ص١٨٨).
> **تحرير:** (أ) «ما تقول أيها الفقيه؟» عند الخطيب، وكراهته «ما يقول صاحبك؟» صريحة، وصار عند ب‑ص «ما مذهب إمامك الشافعي»، وعند ع «ما مذهب إمامك» بإسقاط الشافعي. (ب) ب‑ص يضيف ما ليس عند الخطيب في المقروء: الإيماء باليد، و«أفتاني فلان»، وشرط الرقعة، و«مستوفز» و«ضجر». (ج) أبو زيد يوافق نهي «أفتاني فلان» لكنه في الطالب مع شيخه وبلفظ «لكن الشيخ فلان قال لي».

## ٤٤ — ترك مطالبة المفتي بالحجة، وخلاف السمعاني
**المسألة:** لا يطالب العامي المفتي بالدليل في الحال ولا يقول «لم» و«كيف»؛ فإن أحب سكون نفسه سأل في وقت آخر.
**المصادر:** خ‑ف ج٢ ص٣٨٢–٣٨٣، ٤٠٥–٤٠٦ [٤٦٣ أقدم المقروء بنصه؛ وينقل ب‑ص ص١٥٢ عن الماوردي (٤٥٠) المنعَ من المطالبة بالحجة ولم أقف على نصه]؛ ب‑ص ص١٥١–١٥٢، ١٧١ [٦٤٣]؛ ع ص٢١٥ [٩٨١]؛ مق ص٢٢٨–٢٣٠ (معاصر)؛ ج ص١٠١. **موضعها الأوفى:** الباب الثالث عشر، المسألتان ٨٨ و٧٠.
> **إحالة:** نصوص الخطيب وابن الصلاح والعلموي في ترك مطالبة العامي المفتي بالحجة، وفي ذكر المفتي الحجة في فتواه، مثبتة بنصوصها في **الباب الثالث عشر، المسألتين ٨٨ و٧٠**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **ابن الصلاح:** «وذكر السمعاني: أنه لا يمنع من أن يطالب المفتي بالدليل لأجل احتياطه لنفسه» (ب‑ص ص١٧١)، وأنه «يلزمه أن يذكر له الدليل إن كان مقطوعا به، ولا يلزمه ذلك إن لم يكن مقطوعا به» (ب‑ص ص١٧١)؛ و«وهذا التفصيل أولى مما سبق قريبا ذكره عن القاضي الماوردي من إطلاقه القول: بالمنع من تعرضه للاحتجاج» (ب‑ص ص١٥٢).
- **المقدم:** ينقل الخطيب بنصه (مق ص٢٢٩–٢٣٠) ويقدّم له: «وليحذر طالب العلم عند استفتاء العالم أن يتعنت عند طلب الدليل على فتواه» (مق ص٢٢٨).
> **خلاف:** الخطيب وب‑ص وع على المنع من المطالبة في الحال مع الإذن بالسؤال لاحقاً؛ والسمعاني (في نقل ب‑ص وع) لا يمنع الطلب لأجل الاحتياط، ويلزم المفتي بالدليل المقطوع به، ويرجّح النووي (في نقل ع) قول المنع «الأول». وأقدم من صرّح بالمنع من المقروء الخطيب (٤٦٣)؛ أما السمعاني المخالف فهو «أبو المظفر السمعاني» عند ب‑ص (ص١٦٦) و«ابن السمعاني» عند ع (ص٢١٥)، ولم أستطع من المقروء ربطه بصاحب «أدب الإملاء والاستملاء» (سم) ولا تعيين وفاته، فلا أحكم بأقدميته ولا أصف كتابه الذي ينقلان عنه (ولم أراجعه). وعبارة ج أخص (في الطالب مع الشيخ): «عن بعض السلف: من قال لشيخه لم، لم يفلح أبدا» (ج ص١٠١).

## ٤٥ — كتابة الرقعة: سعتها وضبطها والدعاء فيها وكاتبها
**المسألة:** إذا كتب المستفتي سؤاله فينبغي أن يوسع الورق، ويضبط الخط واللفظ، ويدعو للمفتي، ويحسن كاتب السؤال؛ فرب سؤال مصحّف ينتج فتوى خاطئة.
**المصادر:** خ‑ف ج٢ ص٣٨٣–٣٨٥ [٤٦٣ أقدم المقروء]؛ ب‑ص ص١٦٩–١٧١ [٦٤٣]؛ ع ص٢١٥–٢١٦ [٩٨١]. **موضعها الأوفى:** الباب الثالث عشر، المسألة ٨٧.
> **إحالة:** نصوص الخطيب وابن الصلاح والعلموي في سعة الرقعة وضبطها والدعاء فيها وكاتبها، مثبتة بنصوصها في **الباب الثالث عشر، المسألة ٨٧**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** حكاية الطبري إذ ورد من السلطان استفتاء بلا دعاء فكتب «لا يجوز، أو كتب: يجوز ولم يزد على ذلك» (خ‑ف ص٣٨٣)؛ وقصة المرأة التي جاءت أبا مجالد برقعة فيها «أنت طالق إن تم وقف عبدان» فقال لأصحابه «انظروا في رقعة المرأة» (خ‑ف ص٣٨٤).
- **ابن الصلاح:** يحكي قصة وقف عبدان: «فتصحف ذلك على الفقهاء لكون السؤال عريا عن الضبط» (ب‑ص ص١٧٠) «أبي الحسن الكرخي الحنفي، وقيل إلى أبي مجالد الضرير» (ب‑ص ص١٧٠).
> **خلاف:** (أ) «أفتونا» يمنعها الخطيب (ص٣٨٣) في الرقعة، ويجيزها ع بزيادة «مأجورين أو مثابين» (ص٢١٦)؛ ولا نص جامعاً بينهما في المقروء. (ب) راوي قصة وقف عبدان: أبو مجالد أحمد بن الحسين عند الخطيب (جلست عنده فجاءته امرأة)، وعند ب‑ص «أبي الحسن الكرخي الحنفي، وقيل إلى أبي مجالد الضرير» بالشك. (ج) الأسن والأعلم في الرقعة الواحدة: عند الخطيب (ص٣٨٤–٣٨٥) وب‑ص (ص١٦٩).

## ٤٦ — ما يفعله المفتي بالسؤال: التأمل والاستفصال والتفصيل
**المسألة:** هذه الآداب تُنقل في الكتب قرينةً لآداب المستفتي لأن جودة الجواب مرتبطة بجودة فهم السؤال؛ وأذكر منها ما يتعلق بالسؤال فقط.
**المصادر:** خ‑ف ج٢ ص٣٨٧–٣٩٩ [٤٦٣]؛ ب‑ص ص١٣٧، ١٤٤–١٤٥، ١٥٠–١٥١، ١٥٣ [٦٤٣]؛ ع ص٢٠٣–٢٠٤ [٩٨١]. **موضعها الأوفى:** الباب الثالث عشر، المسائل ٥٠–٥٢ و٥٤–٥٥ و٦٩.
> **إحالة:** نصوص الخطيب وابن الصلاح والعلموي في تأمل المفتي للرقعة كلمة بعد كلمة، والاستفصال، والمشاورة، وفهم السؤال، مثبتة بنصوصها في **الباب الثالث عشر، المسائل ٥٠–٥٢ و٥٤–٥٥ و٦٩**؛ ويلتقي هذا الموضع مع م٢١ هنا؛ فلا أكررها.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** «فأول ما يجب على المفتي أن يتأمل رقعة الاستفتاء تأملا شافيا، ويقرأ ما فيها كله، كلمة بعد كلمة، حتى ينتهي إلى آخره» (خ‑ف ص٣٨٧)؛ «وإن مر بشبه كلمة غريبة أو لفظة تحتمل عدة معان، سأل عنها المستفتي» (خ‑ف ص٣٨٨)؛ «وإن سئل عن قوم شهدوا على رجل بالزنا فينبغي أن يستفهم السائل كيف رأى الشهود المشهود عليه» (خ‑ف ص٣٩٤) ويستشهد بقول النبي لماعز «لعلك قبلت أو غمزت أو نظرت» (خ‑ف ص٣٩٤)؛ «وإذا تفكر في مسألة متعارضة الأدلة، لم يجب فيها حتى يثبت عنده ما يرجح به أحد الأدلة» (خ‑ف ص٣٩٥).
- **ابن الصلاح:** يصرّح بالنقل عن الخطيب: «وقال الخطيب أبو بكر أحمد بن علي الحافظ: رأيت القاضي أبا الطيب الطبري يفعل هذا في الرقاع التي ترفع إليه للاستفتاء» (ب‑ص ص١٣٧)؛ وإذا اشتملت الرقعة على مسائل فهم بعضها «سكت عن ذلك البعض وأجاب عن البعض الآخر» (ب‑ص ص١٥١).
- **العلموي:** عن الصيمري وابن الصلاح: «وإذا ظهر للمفتي أن الجواب خلاف غرض المستفتي فليقتصر على مشافهته بالجواب بلا كتابة» (ع ص٢٠٤) (نقل ع عن الصيمري وابن الصلاح، ولم أراجع أصليهما).
> **تحرير:** لم أنقل بقية أبواب ما يفعله المفتي (الرجوع عن الفتوى، المشاورة، كتابة الخط)، وهي خارج موضوع السؤال. ويلتقي هذا الموضع مع م٢١ في التثبت قبل الجواب.

## ٤٧ — من سُئل عما ليس من فنّه: يردّه إلى أهله
**المسألة:** المفتي لا يجيب عن علم لا يحسنه؛ وهو لون من «لا أدري» يتعلق بالتخصص.
**المصادر:** خ‑ف ج٢ ص٤٠١–٤٠٢ [٤٦٣]؛ ماو ص٧٣–٧٤ [٤٥٠]؛ ب‑ص ص١٥٣–١٥٧ [٦٤٣]. **موضعها الأوفى:** الباب الثالث عشر، المسألتان ٧٣ و٧٤.
> **إحالة:** نصوص الخطيب وابن الصلاح في الإمساك عن الفتوى في المسائل الكلامية، وفي أسئلة التفسير وما ليس من فنّ الفقيه، مثبتة بنصوصها في **الباب الثالث عشر، المسألتين ٧٣ و٧٤**؛ فلا أكررها هنا.
**الزيادات التي انفرد بها هذا الباب:**
- **الخطيب:** «وأما إذا سئل عن تفسير: الزقوم والغسلين، والفتيل، والنقير، والقطمير، والحنان، رد ذلك إلى أهله ووكله إلى من نصب نفسه له» (خ‑ف ص٤٠٢)؛ وأبو حنيفة «كان أبو حنيفة إذا سئل عن شيء من اللغة، يقول: إنها ليست من شأني» (خ‑ف ص٤٠٢)، ويتمثل بشعر منه «من تحلى بغير ما هو فيه فضحته شواهد الامتحان» (خ‑ف ص٤٠٢).
- **ابن الصلاح:** ويستشهد بتعزير عمر صبيغاً «الذي كان يسأل عن المتشابهات» (ب‑ص ص١٥٤)؛ وينقل عن الصيمري «إن مما أجمع عليه أهل الفتوى أن من كان موسوما بالفتوى في الفقه، لم ينبغ أن يضع» (ب‑ص ص١٥٦) ثم «في مسألة من الكلام، كالقضاء والقدر، والرؤية، وخلق القرآن» (ب‑ص ص١٥٧)؛ ويحكي «وحكى أبو عمر ابن عبد البر الفقيه الحافظ الأندلسي: الامتناع من الكلام في كل ذلك عن الفقهاء والعلماء قديما وحديثا» (ب‑ص ص١٥٧).
- **الماوردي:** «وحق على من ترك العجب بما يحسن أن يدع التكلف لما لا يحسن» (ماو ص٧٣) ثم استعاذة الجاحظ «ونعوذ بك من التكلف لما لا نحسن» (ماو ص٧٣).
> **تحرير:** الخطيب لا يستعمل عبارة «لا أدري» في هذا الموضع؛ والمعنى (رد ما ليس من الفن إلى أهله) قريب منها عندي. وعبارة الصيمري في ب‑ص «موسوما بالفتوى في الفقه... يضع» تقابل عبارة الخطيب «مرسوما بالفتوى في الفقه... يطلق خطه» (خ‑ف ص٤٠١)، فهما متقاربتان لفظاً (موسوما/مرسوما، يضع/يطلق)؛ وعزو ب‑ص إلى ابن عبد البر الامتناع عن الكلام لم أجده في المواضع المقروءة من ع‑ب. وليس عند الماوردي ذكر للفتوى في هذا الموضع بل هو في الجهل والتكلف.

## القسم السابع: مراجعة الجواب وبيان مأخذه

## ٤٨ — استفهام المتعلم عن مأخذ الجواب: أثر هو أم رأي؟
**المسألة:** يجوز للمتعلم أن يسأل عن جواب الفقيه أهو عن أثر أم عن رأي، ويحسن بالمجيب أن يصرّح بأن الجواب رأيه إذا كان رأياً، وأن يبيّن علة فعله لأتباعه.
**المصادر:** خ‑ف ج٢ ص٣١١–٣١٥ وع‑ب ج١ ص٧٧٣–٧٧٨ وج٢ ص٨٣٠، ١٠٦٩–١٠٧١، ١٠٧٥، ١١٢٤ [٤٦٣]؛ ب‑ص ص١٣٩–١٤٠ [٦٤٣].
**الزيادات:**
- **الخطيب (الفقيه):** «وإذا أجابه الفقيه عن مسألة جاز أن يستفهمه عن جوابه، أقاله عن أثر، أو عن رأي» (خ‑ف ص٣١١)؛ ومثاله ابن مسعود حين لبّى على الصفا وقال «يا لسان، قل خيرا تغنم»، فقيل له «هذا شيء أنت تقوله، أو شيء سمعته؟» فقال «لا بل شيء سمعته» (خ‑ف ص٣١١)؛ وجابر في الضبع: «فقلت: سمعته من رسول الله صلى الله عليه وسلم؟ قال: نعم» (خ‑ف ص٣١٢)؛ وقول ربيعة لابن شهاب: «إذا حدثت الناس برأيك فأخبرهم أنه رأيك، وإذا حدثت الناس بشيء من السنة فأخبرهم أنه سنة لا يظنون أنه رأيك» (خ‑ف ص٣١٢)؛ ثم «ويحق على الفقيه إذا فعل فعلا من علة أن يخبر بها الأتباع» (خ‑ف ص٣١٣) ومثاله جهر ابن عباس بالفاتحة في صلاة الجنازة: «إني لم أجهر إلا لتعلموا» (خ‑ف ص٣١٥).
- **ابن عبد البر:** أبو بكر «هذا رأيي فإن يكن صوابا فمن الله وإن يكن خطأ فمني وأستغفر الله» (ع‑ب ج٢ ص٨٣٠)؛ وقول ربيعة لابن شهاب بلفظ «لا يظنوا أنه رأيك» (ع‑ب ج٢ ص١٠٧١)؛ وزيد لقوم كتبوا عنه أجوبته فاعتذر إليهم «عذرا لعل كل شيء حدثتكم خطأ، إنما اجتهدت لكم رأيي» (ع‑ب ج٢ ص١٠٦٩)؛ والقاسم لرجل أجابه: «لا تقل إن القاسم يزعم أن هذا هو الحق، ولكن إن اضطررت إليه عملت به» (ع‑ب ج٢ ص١٠٧٠)؛ ومالك «ما كانوا يجترئون على ذلك وإنما كانوا يقولون: نكره هذا ونرى هذا حسنا» (ع‑ب ج٢ ص١٠٧٥) ويشرحه ابن عبد البر: «معنى قول مالك هذا أن ما أخذه من العلم رأيا واستحسانا لم يقل فيه حلال ولا حرام» (ع‑ب ج٢ ص١٠٧٥).
- **ابن عبد البر (من القراءة الجديدة، ج١ ص٧٧٣–٧٧٨):** سليمان بن موسى لعطاء بعد أن أفتاه في المستحاضة: «أرأي أم علم؟» فقال: «بلى سمعنا أنها إذا صلت وصامت حل لزوجها أن يصيبها» (ص٧٧٣)؛ وابن جريج لعطاء في الغريب القادم معتمراً: «أرأي أم علم؟» فقال: «بل علم» (ص٧٧٣)؛ وابن سيرين في المتعة: «فإن يكن علما فهما أعلم مني وإن يكن رأيا فرأيهما أفضل» (ص٧٧٣)؛ وحفص بن غياث: «إنما هو رأيي ليس فيه كتاب ولا سنة» (ص٧٧٤)؛ ومالك «إنما أنا بشر، أخطئ وأصيب فانظروا في رأيي فكلما وافق الكتاب والسنة فخذوا به» (ص٧٧٥)؛ وابن سيرين إذا سئل: «ليس عندي فيه إلا رأي أتهمه» (ص٧٧٧)؛ وسالم لمن رضي برأيه: «لعلي أن أخبرك برأيي ثم تذهب فأرى بعدك رأيا غيره فلا أجدك» (ص٧٧٧)؛ وابن عمر «إن شئتم أخبرتكم بالظن» (ص٧٧٧)؛ والعنبري «إنما قولي لا تلزمه نفقتهم رأي ولا أدري لعله خطأ» (ص٧٧٨)؛ والقاسم وقد قال لسائل «ما أدري ما هذا» ثم لما ذُكر له قضاء عمر بن عبد العزيز: «ما أرى رأيه إلا معتدلا وهذا رأيي وما أقول إنه الحق» (ج٢ ص١١٢٤).
- **ابن الصلاح (عن الصيمري):** «وكان بعض السلف إذا أفتى يقول: إن كان هذا صوابا فمن الله» (ب‑ص ص١٣٩) «كان خطأ فمني» (ب‑ص ص١٤٠)، ثم «وهذا معنى كره في هذا الزمان لأن إضعاف نفس السائل، وإدخال قلبه الشك في الجواب» (ب‑ص ص١٤٠)، «وليس يقبح منه أن يقول: الجواب عندنا، أو الذي عندنا، أو يقول: الذي نراه، كذا وكذا» (ب‑ص ص١٤٠).
> **تحرير:** (أ) أثر ربيعة لابن شهاب: «لا يظنون أنه رأيك» (خ‑ف) و«لا يظنوا أنه رأيك» (ع‑ب)، والمعنى واحد. (ب) الخطيب يذكر الاستفهام عن المأخذ جائزاً للمتعلم، وابن عبد البر يسوق أخبار الصحابة والتابعين في التنبيه على أن الجواب رأي؛ والموضعان مختلفان في السياق (الأول آداب المتعلم مع الفقيه، والثاني ذم الرأي في الدين)، فلا أنسب إلى ابن عبد البر حكماً في استفهام المتعلم. (ج) يلتقي هذا مع م٤٤ (ترك مطالبة الحجة) في أن الخطيب يمنع «لم» و«كيف» من العامي، ويجيز السؤال عن كون الجواب أثراً أو رأياً؛ ولم أجد في المقروء توفيقاً صريحاً بين الأمرين. (هـ) خلاف: قول أبي بكر «هذا رأيي فإن يكن صوابا فمن الله وإن يكن خطأ فمني» (ع‑ب ص٨٣٠) يقابله عند ب‑ص كراهة الصيمري هذه الصيغة «في هذا الزمان» لأنها تضعف نفس السائل (ب‑ص ص١٤٠) مع إجازته «الذي نراه»؛ ولا أرجّح. (د) أبو بكر قال «هذا رأيي» عن قضية «فاجتهد رأيه» (ع‑ب ج٢ ص٨٣٠) وهو في الباب الخاص بـ«لا أدري».
> (و) في ع‑ب يصدر السؤال «أرأي أم علم؟» عن السائل أو الطالب (سليمان بن موسى، وابن جريج)، وجواب العالم إما بإسناد الجواب إلى سماع أو بتسميته رأياً (ابن سيرين، وحفص، وسالم، والقاسم)؛ وهو نظير استفهام المتعلم عن المأخذ الذي أجازه الخطيب (خ‑ف ص٣١١)، ولا أنسب إلى ع‑ب في هذه المواضع حكماً بالجواز أو بالمنع.

## ما لم يدخل هذا الباب وسببه
- **باب طرح العالم المسألة على المتعلم** (ع‑ب ج١ ص٤٧٩–٤٨٤): قرأتُ نصه (سؤال النبي لمعاذ «هل تدري يا معاذ ما حق الله على الناس؟»، وسؤال النبي عن الشجرة، وأسئلة سعيد بن المسيب لجلسائه) ولم أدرجه هنا لأنه من آداب العالم مع طلبته؛ وهو مثبت في **الباب الثامن، المسألة ١٩**. وباب **تدافع الفتوى وذم من سارع إليها** (ع‑ب ج٢ ص١١٢٠–١١٢٨): قرأته كاملاً وأدرجتُ منه ما يخص هذا الباب في م٣١ وم٣٢ وم٤٨، وقد أحال عليه ع‑ب في ج٢ ص٨٤٣ (م٣٢)؛ وما بعده (باب رتب الطلب من ص١١٢٩) خارج موضوع الباب.
- **رجوع المفتي عن فتواه** (خ‑ف ج٢ ص٤٢١–٤٢٦، وب‑ص ص١٠٩): من أحكام الفتوى لا من آداب السؤال والجواب في الدرس، وقد قرأت موضعيهما ولم أنقل منهما.
- **شروط المفتي وطبقاته** (ب‑ص قبل ص١٣١، وع ص١٧٨–٢٠٢): خارج موضوع الباب ولم أقرأها لهذا الباب.
- **تفصيل مسألة تمذهب العامي ومناقب المذهب** (ب‑ص ص١٦١–١٦٤، وع ص٢١٠–٢١٤): لم أنقل منها إلا ما يخص اختيار المستفتي (م٤١).
- **آداب الكتابة والإملاء عند المحدث** وما يجري مجرى «أدب السؤال للمحدث» في غير ما ذُكر (خ‑ج ج١ بعد ص٢٢٧): خارج المقروء.
- **الزرنوجي والمنسوب للطوسي** (OCR، بلا صفحة): لم أجد فيهما شيئاً مستقلاً في هذا الباب من المقروء، فلا أذكر منهما شيئاً.
- **منية المريد** (ش، OCR، بلا صفحة): لم أقرأ تفصيل أمور الباب الثاني منه (آداب الفتوى والمفتي والمستفتي، وهو في أربعة أنواع)، وأخذتُ من بنية الباب الأول ما يخص السؤال في الدرس (م٥ وم١٠ وم٢٣ وم٥١). **الكافي** (OCR): باب سؤال العالم وتذاكره (ص٩٧) وباب النهي عن القول بغير علم (ص١٠١) لم يُقرأ منهما إلا العنوان.
